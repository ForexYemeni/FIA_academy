    39→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    40→// Show only the human-readable text in the UI, but preserve the extra data on save.
    41→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    42→const splitBio = (raw: string) => {
    43→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    44→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    45→}
    46→const joinBio = (display: string, original?: string) => {
    47→  if (!original) return display
    48→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    49→  if (i === -1) return display
    50→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    51→}

    53→// Returns a normalized status info object for an enrollment's payment_status.
    54→// payment_status can be one of: 'pending' | 'confirmed' | 'completed' | 'free'.
    55→// - 'pending'    -> student paid, waiting for admin approval (yellow)
    56→// - 'confirmed'  -> admin approved the payment, course in progress (green)
    57→// - 'completed'  -> payment confirmed AND course finished (green)
    58→// - 'free'       -> free enrollment (blue)
    59→export function getPaymentStatusInfo(paymentStatus: string | undefined | null): {
    60→  label: string
    61→  badgeClass: string
    62→} {
    63→  switch (paymentStatus) {
    64→    case 'completed':
    65→      return { label: 'مكتمل', badgeClass: 'bg-green-500 text-white' }
    66→    case 'confirmed':
    67→      return { label: 'تم تأكيد الدفع', badgeClass: 'bg-green-500 text-white' }
    68→    case 'free':
    69→      return { label: 'مجانية', badgeClass: 'bg-blue-500 text-white' }
    70→    case 'pending':
    71→    default:
    72→      return { label: 'بانتظار تأكيد الدفع', badgeClass: 'bg-yellow-500 text-white' }
    73→  }
    74→}

    76→// Returns a status info object for a raw payment record's `status` field
    77→// (used in the Payments section). Raw status can be: 'مكتمل' | 'completed' |
    78→// 'قيد الانتظار' | 'pending' | 'بانتظار' | 'ملغي' | 'failed' | 'فشل' | etc.
    79→export function getPaymentRecordStatusInfo(status: string | undefined | null): {
    80→  label: string
    81→  badgeClass: string
    82→} {
    83→  const s = (status || '').trim()
    84→  const norm = s.toLowerCase()
    85→  if (s === 'مكتمل' || norm === 'completed' || norm === 'paid' || norm === 'confirmed') {
    86→    return { label: s || 'مكتمل', badgeClass: 'bg-green-500/15 text-green-600 border-green-500/30' }
    87→  }
    88→  if (s === 'ملغي' || norm === 'failed' || s === 'فشل' || norm === 'cancelled' || norm === 'canceled') {
    89→    return { label: s || 'فشل', badgeClass: 'bg-red-500/15 text-red-600 border-red-500/30' }
    90→  }
    91→  if (s === 'قيد الانتظار' || s === 'بانتظار' || norm === 'pending' || s === '') {
    92→    return { label: s || 'بانتظار', badgeClass: 'bg-yellow-500/15 text-yellow-600 border-yellow-500/30' }
    93→  }
    94→  // Unknown status — show as neutral
    95→  return { label: s, badgeClass: 'bg-muted text-muted-foreground border-border' }
    96→}

   326→function BrowseCourses({ courses, enrollments, userId, onEnrolled }: any) {

   328→  const [payDialog, setPayDialog] = useState<{ course: any; open: boolean } | null>(null)
   329→  const [enrolling, setEnrolling] = useState<string | null>(null)

   333→  }, [])
   334→
   335→  const isEnrolled = (courseId: string) => enrollments?.some((e: any) => e.course_id === courseId)
   336→
   337→  const enrollFree = async (course: any) => {
   338→    // Double-check client-side before calling API
   339→    if (isEnrolled(course.id)) {
   340→      toast.info('أنت مسجّل في هذه الدورة')
   341→      return
   342→    }
   343→    setEnrolling(course.id)
   344→    try {
   345→      const res = await fetch('/api/enrollments', {
   346→        method: 'POST',
   347→        headers: { 'Content-Type': 'application/json' },
   348→        body: JSON.stringify({ user_id: userId, course_id: course.id, payment_status: 'free' }),
   349→      })
   350→      const data = await res.json()
   351→      if (data.already_enrolled) {
   352→        toast.info('أنت مسجّل في هذه الدورة')
   353→      } else if (res.ok) {
   354→        toast.success(`تم التسجيل في دورة: ${course.title}`)
   355→        onEnrolled()
   356→      } else toast.error('فشل التسجيل')
   357→    } catch {
   358→      toast.error('خطأ')
   359→    } finally {
   360→      setEnrolling(null)
   361→    }
   362→  }
   363→
   364→  const enrollPaid = async (course: any, method: any, reference: string, proofUrl?: string | null) => {

   366→    if (isEnrolled(course.id)) {
   367→      toast.info('أنت مسجّل في هذه الدورة')
   368→      return
   369→    }
   370→    setEnrolling(course.id)
   371→    try {
   372→      // Combine reference with proof URL
   373→      const fullReference = proofUrl ? `${reference} | PROOF: ${proofUrl}` : reference
   374→
   375→      // 1. Create enrollment with pending payment
   376→      const res = await fetch('/api/enrollments', {
   377→        method: 'POST',
   378→        headers: { 'Content-Type': 'application/json' },
   379→        body: JSON.stringify({
   380→          user_id: userId,
   381→          course_id: course.id,
   382→          payment_status: 'pending',
   383→          amount_paid: course.price,
   384→          payment_method: method.name,
   385→          payment_reference: fullReference,
   386→        }),
   387→      })
   388→      const data = await res.json()
   389→      if (data.already_enrolled) {
   390→        toast.info('أنت مسجّل في هذه الدورة')
   391→      } else if (res.ok) {
   392→        // 2. Create payment record with proof URL
   393→        await fetch('/api/payments', {
   394→          method: 'POST',
   395→          headers: { 'Content-Type': 'application/json' },
   396→          body: JSON.stringify({
   397→            amount: course.price,
   398→            currency: course.currency || 'ر.ي',
   399→            method: method.name,
   400→            status: 'قيد الانتظار',
   401→            reference: fullReference,
   402→            user_id: userId,
   403→            course_id: course.id,
   404→          }),
   405→        })
   406→        toast.success('تم تسجيل طلبك! سيتم تفعيل الدورة بعد تأكيد الدفع')
   407→        onEnrolled()
   408→        setPayDialog(null)
   409→      } else toast.error('فشل التسجيل')
   410→    } catch {
   411→      toast.error('خطأ')
   412→    } finally {
   413→      setEnrolling(null)
   414→    }
   415→  }
   416→
   417→  return (
   418→    <SectionCard title="استكشاف الدورات" description="تصفّح جميع الدورات المتاحة">
   419→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">

   421→          const enrolled = isEnrolled(c.id)
   422→          const isFree = c.price_type === 'مجانية' || c.price === 0
   423→          return (
   424→            <Card key={c.id} className="overflow-hidden card-hover">

   426→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}

   428→                {isFree ? (
   429→                  <Badge className="absolute top-2 left-2 bg-green-500 text-white">مجانية</Badge>

   431→                  <Badge className="absolute top-2 left-2 bg-gold text-navy">مدفوعة</Badge>

   879→function Payments({ payments, userId, courses }: any) {
   880→  // Filter by the current user's id so we never display someone else's payments.
   881→  const myPayments = (payments || []).filter((p: any) => p.user_id === userId)
   882→  // Sort by date DESC (most recent first) for display
   883→  const sortedPayments = [...myPayments].sort((a: any, b: any) => {
   884→    const da = a.date ? new Date(a.date).getTime() : 0
   885→    const db = b.date ? new Date(b.date).getTime() : 0
   886→    return db - da
   887→  })
   888→
   889→  const getCourseTitle = (courseId: string | null | undefined) => {
   890→    if (!courseId) return null
   891→    return courses?.find((c: any) => c.id === courseId)?.title || null
   892→  }
   893→
   894→  const fmtAmount = (p: any) => {
   895→    const amt = typeof p.amount === 'number' ? p.amount : Number(p.amount) || 0
   896→    return amt.toLocaleString('en-US')
   897→  }
   898→
   899→  const fmtDate = (d: string | null | undefined) => {
   900→    if (!d) return null
   901→    try {
   902→      return new Date(d).toLocaleDateString('ar-EG', {
   903→        year: 'numeric',
   904→        month: 'long',
   905→        day: 'numeric',
   906→      })
   907→    } catch {
   908→      return null
   909→    }
   910→  }
   911→
   912→  // Strip the trailing "| PROOF: https://..." suffix from a reference so we
   913→  // show the user-friendly reference number (e.g., "JB-123456789") in the UI.
   914→  const cleanReference = (ref: string | null | undefined) => {
   915→    if (!ref) return null
   916→    return ref.replace(/\s*\|\s*PROOF:.*$/i, '').trim() || null
   917→  }
   918→
   919→  return (
   920→    <SectionCard title="مدفوعاتي" description={`${sortedPayments.length} معاملة`}>
   921→      {sortedPayments.length === 0 ? (
   922→        <div className="text-center py-16">
   923→          <div className="w-20 h-20 mx-auto mb-4 rounded-full bg-muted/40 flex items-center justify-center">
   924→            <CreditCard className="w-10 h-10 text-muted-foreground/50" />
   925→          </div>
   926→          <p className="text-base font-semibold text-navy mb-1">لا توجد مدفوعات بعد</p>
   927→          <p className="text-xs text-muted-foreground">
   928→            عند تسجيلك في دورة مدفوعة ورفع إيصال الدفع، ستظهر تفاصيل المعاملة هنا.
   929→          </p>
   930→        </div>
   931→      ) : (
   932→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   933→          {sortedPayments.map((p: any) => {
   934→            const statusInfo = getPaymentRecordStatusInfo(p.status)
   935→            const courseTitle = getCourseTitle(p.course_id)
   936→            const ref = cleanReference(p.reference)
   937→            const dateStr = fmtDate(p.date)
   938→            return (
   939→              <Card key={p.id} className="p-0 overflow-hidden border-border card-hover">
   940→                {/* Amount header — large bold */}
   941→                <div className="bg-gradient-to-br from-navy to-navy-light p-4 text-white">
   942→                  <div className="flex items-start justify-between gap-2">
   943→                    <div className="min-w-0">
   944→                      <div className="text-[11px] text-gold/80 font-semibold mb-0.5">المبلغ</div>
   945→                      <div className="text-2xl font-extrabold tracking-tight" dir="ltr">
   946→                        {fmtAmount(p)} <span className="text-base font-bold text-gold">{p.currency || 'ر.ي'}</span>
   947→                      </div>
   948→                    </div>
   949→                    <Badge className={`${statusInfo.badgeClass} border whitespace-nowrap`}>
   950→                      {statusInfo.label}
   951→                    </Badge>
   952→                  </div>
   953→                </div>
   954→
   955→                {/* Details body */}
   956→                <div className="p-4 space-y-3">
   957→                  {courseTitle && (
   958→                    <div>
   959→                      <div className="text-[11px] text-muted-foreground mb-0.5">الدورة</div>
   960→                      <div className="text-sm font-semibold text-navy line-clamp-2">{courseTitle}</div>
   961→                    </div>
   962→                  )}
   963→
   964→                  <div className="grid grid-cols-2 gap-3">
   965→                    <div>
   966→                      <div className="text-[11px] text-muted-foreground mb-0.5">طريقة الدفع</div>
   967→                      <div className="text-sm font-medium text-navy">{p.method || '-'}</div>
   968→                    </div>
   969→                    <div>
   970→                      <div className="text-[11px] text-muted-foreground mb-0.5">التاريخ</div>
   971→                      <div className="text-sm font-medium text-navy" dir="ltr">{dateStr || '-'}</div>
   972→                    </div>
   973→                  </div>
   974→
   975→                  <div>
   976→                    <div className="text-[11px] text-muted-foreground mb-0.5">رقم المرجع</div>
   977→                    <div className="text-sm font-mono text-navy break-all" dir="ltr">{ref || '-'}</div>
   978→                  </div>
   979→                </div>
   980→              </Card>
   981→            )
   982→          })}
   983→        </div>
   984→      )}
   985→    </SectionCard>
   986→  )
