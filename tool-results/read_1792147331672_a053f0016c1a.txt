    39→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    40→// Show only the human-readable text in the UI, but preserve the extra data on save.
    41→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    42→const splitBio = (raw: string) => {
    43→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    44→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    45→}
    46→const joinBio = (display: string, original?: string) => {
    47→  if (!original) return display
    48→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    49→  if (i === -1) return display
    50→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    51→}

    53→export function StudentDashboard() {
    54→  const [active, setActive] = useState('overview')
    55→  const [viewingCourse, setViewingCourse] = useState<any | null>(null)
    56→  const [viewingEnrollment, setViewingEnrollment] = useState<any | null>(null)
    57→  const user = useAuth((s) => s.user)
    58→  const updateUser = useAuth((s) => s.updateUser)
    59→  const [courses, setCourses] = useState<any[]>([])
    60→  const [certificates, setCertificates] = useState<any[]>([])
    61→  const [payments, setPayments] = useState<any[]>([])

    63→  const [profile, setProfile] = useState({ name: user?.name || '', phone: user?.phone || '', title: user?.title || '', bio: splitBio(user?.bio || '') })
    64→
    65→  const loadEnrollments = () => {
    66→    if (!user?.id) return
    67→    fetch(`/api/enrollments?user_id=${user.id}`).then((r) => r.json()).then((d) => setEnrollments(Array.isArray(d) ? d : [])).catch(() => {})

    69→
    70→  useEffect(() => {
    71→    fetch('/api/courses').then((r) => r.json()).then((d) => setCourses(Array.isArray(d) ? d : [])).catch(() => {})

    74→    loadEnrollments()
    75→  }, [user?.id])
    76→
    77→  const onEnrolled = () => {
    78→    loadEnrollments()
    79→  }
    80→
    81→  return (
    82→    <DashboardShell
    83→      nav={NAV}
    84→      active={active}
    85→      onNavigate={(id) => { setActive(id); if (id !== 'course-view') setViewingCourse(null) }}

    91→          course={viewingCourse}
    92→          enrollment={viewingEnrollment}
    93→          onBack={() => { setViewingCourse(null); setActive('courses'); loadEnrollments() }}
    94→        />

    99→          {active === 'courses' && <MyCourses courses={courses} enrollments={enrollments} studentName={user?.name} onContinue={(course, enrollment) => { setViewingCourse(course); setViewingEnrollment(enrollment); setActive('course-view') }} />}

   170→function MyCourses({ courses, enrollments, onContinue, studentName }: any) {
   171→  const [requesting, setRequesting] = useState<string | null>(null)
   172→
   173→  const requestCertificate = async (course: any) => {
   174→    setRequesting(course.id)
   175→    try {
   176→      const res = await fetch('/api/certificate-requests', {
   177→        method: 'POST',
   178→        headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({
   180→          user_id: course.enrollment.user_id,
   181→          course_id: course.id,
   182→          student_name: studentName || course.enrollment.user_id || 'متدرب',
   183→          course_title: course.title,
   184→          fee: 0,
   185→        }),
   186→      })
   187→      const data = await res.json()
   188→      if (res.ok) {
   189→        toast.success('تم إرسال طلب الشهادة! سيتم مراجعته من قبل المدرب والإدارة')
   190→      } else {
   191→        toast.error(data.error || 'فشل الطلب')
   192→      }
   193→    } catch {
   194→      toast.error('خطأ في الاتصال')
   195→    } finally {
   196→      setRequesting(null)
   197→    }
   198→  }
   199→
   200→  const enrolled = (enrollments || [])
   201→    .map((e: any) => {
   202→      const course = courses.find((c: any) => c.id === e.course_id)
   203→      return course ? { ...course, enrollment: e } : null

   205→    .filter(Boolean)

   208→    <SectionCard title="دوراتي المسجّلة" description={`${enrolled.length} دورة`}>

   214→        </div>

   220→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   221→                <div className="absolute top-2 right-2">
   222→                  <Badge className={
   223→                    (c.enrollment.status === 'completed' || c.enrollment.status === 'مكتمل') ? 'bg-green-500 text-white'

   285→function BrowseCourses({ courses, enrollments, userId, onEnrolled }: any) {
   286→  const [paymentMethods, setPaymentMethods] = useState<any[]>([])
   287→  const [payDialog, setPayDialog] = useState<{ course: any; open: boolean } | null>(null)
   288→  const [enrolling, setEnrolling] = useState<string | null>(null)
   289→
   290→  useEffect(() => {
   291→    fetch('/api/payment-methods').then((r) => r.json()).then((d) => setPaymentMethods(Array.isArray(d) ? d.filter((p:any)=>p.enabled) : [])).catch(() => {})
   292→  }, [])
   293→
   294→  const isEnrolled = (courseId: string) => enrollments?.some((e: any) => e.course_id === courseId)
   295→
   296→  const enrollFree = async (course: any) => {
   297→    // Double-check client-side before calling API
   298→    if (isEnrolled(course.id)) {
   299→      toast.info('أنت مسجّل في هذه الدورة')
   300→      return
   301→    }
   302→    setEnrolling(course.id)
   303→    try {
   304→      const res = await fetch('/api/enrollments', {
   305→        method: 'POST',
   306→        headers: { 'Content-Type': 'application/json' },
   307→        body: JSON.stringify({ user_id: userId, course_id: course.id, payment_status: 'free' }),
   308→      })
   309→      const data = await res.json()
   310→      if (data.already_enrolled) {
   311→        toast.info('أنت مسجّل في هذه الدورة')
   312→      } else if (res.ok) {
   313→        toast.success(`تم التسجيل في دورة: ${course.title}`)
   314→        onEnrolled()
   315→      } else toast.error('فشل التسجيل')
   316→    } catch {
   317→      toast.error('خطأ')
   318→    } finally {
   319→      setEnrolling(null)
   320→    }
   321→  }
   322→
   323→  const enrollPaid = async (course: any, method: any, reference: string, proofUrl?: string | null) => {
   324→    // Double-check client-side before calling API
   325→    if (isEnrolled(course.id)) {
   326→      toast.info('أنت مسجّل في هذه الدورة')
   327→      return
   328→    }
   329→    setEnrolling(course.id)
   330→    try {
   331→      // Combine reference with proof URL
   332→      const fullReference = proofUrl ? `${reference} | PROOF: ${proofUrl}` : reference
   333→
   334→      // 1. Create enrollment with pending payment
   335→      const res = await fetch('/api/enrollments', {
   336→        method: 'POST',
   337→        headers: { 'Content-Type': 'application/json' },
   338→        body: JSON.stringify({
   339→          user_id: userId,
   340→          course_id: course.id,
   341→          payment_status: 'pending',
   342→          amount_paid: course.price,
   343→          payment_method: method.name,
   344→          payment_reference: fullReference,
   345→        }),
   346→      })
   347→      const data = await res.json()
   348→      if (data.already_enrolled) {
   349→        toast.info('أنت مسجّل في هذه الدورة')
   350→      } else if (res.ok) {
   351→        // 2. Create payment record with proof URL
   352→        await fetch('/api/payments', {
   353→          method: 'POST',
   354→          headers: { 'Content-Type': 'application/json' },
   355→          body: JSON.stringify({
   356→            amount: course.price,
   357→            currency: course.currency || 'ر.ي',
   358→            method: method.name,
   359→            status: 'قيد الانتظار',
   360→            reference: fullReference,
   361→            user_id: userId,
   362→            course_id: course.id,
   363→          }),
   364→        })
   365→        toast.success('تم تسجيل طلبك! سيتم تفعيل الدورة بعد تأكيد الدفع')
   366→        onEnrolled()
   367→        setPayDialog(null)
   368→      } else toast.error('فشل التسجيل')
   369→    } catch {
   370→      toast.error('خطأ')
   371→    } finally {
   372→      setEnrolling(null)
   373→    }
   374→  }
   375→
   376→  return (
   377→    <SectionCard title="استكشاف الدورات" description="تصفّح جميع الدورات المتاحة">
   378→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   379→        {courses.map((c: any) => {
   380→          const enrolled = isEnrolled(c.id)
   381→          const isFree = c.price_type === 'مجانية' || c.price === 0
   382→          return (
   383→            <Card key={c.id} className="overflow-hidden card-hover">
   384→              <div className="h-32 bg-navy relative">
   385→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   386→                <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   387→                {isFree ? (
   388→                  <Badge className="absolute top-2 left-2 bg-green-500 text-white">مجانية</Badge>
   389→                ) : (
   390→                  <Badge className="absolute top-2 left-2 bg-gold text-navy">مدفوعة</Badge>
   391→                )}
   392→              </div>
   393→              <div className="p-4">
   394→                <h3 className="font-bold text-navy text-sm mb-1 line-clamp-2 min-h-[2.5rem]">{c.title}</h3>
   395→                <p className="text-xs text-muted-foreground mb-3 line-clamp-2">{c.short_description}</p>
   396→                <div className="flex items-center justify-between text-xs text-muted-foreground mb-3">
   397→                  <span className="flex items-center gap-1">
   398→                    <Clock className="w-3 h-3" /> {c.duration_hours || 0} ساعة
   399→                  </span>
   400→                  <span className="font-bold text-navy">
   401→                    {isFree ? <span className="text-green-600">مجانية</span> : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}
   402→                  </span>
   403→                </div>
   404→                {enrolled ? (
   405→                  <Button size="sm" variant="outline" className="w-full border-green-500/30 text-green-600 bg-green-500/5">
   406→                    <CheckCircle2 className="w-4 h-4 ml-1" /> مسجّل
   407→                  </Button>
   408→                ) : isFree ? (
   409→                  <Button
   410→                    size="sm"
   411→                    className="w-full bg-green-600 hover:bg-green-700"
   412→                    onClick={() => enrollFree(c)}
   413→                    disabled={enrolling === c.id}
   414→                  >
   415→                    {enrolling === c.id ? 'جاري التسجيل...' : 'سجّل الآن'}
   416→                  </Button>
   417→                ) : (
   418→                  <Button
   419→                    size="sm"
   420→                    className="w-full bg-gold hover:bg-gold-light text-navy"
   421→                    onClick={() => setPayDialog({ course: c, open: true })}
   422→                  >
   423→                    <Lock className="w-3.5 h-3.5 ml-1" /> اشترك الآن
   424→                  </Button>
   425→                )}
   426→              </div>
   427→            </Card>
   428→          )
   429→        })}
   430→      </div>
   431→
   432→      {payDialog?.open && (
   433→        <PaymentDialog
   434→          course={payDialog.course}
   435→          methods={paymentMethods}
   436→          open={payDialog.open}
   437→          onOpenChange={(o) => setPayDialog({ ...payDialog, open: o })}
   438→          onPay={(method: any, reference: string, proofUrl?: string | null) => enrollPaid(payDialog.course, method, reference, proofUrl)}

   460→          const maxW = 1024, maxH = 1024

   463→          if (height > maxH) { width = (width * maxH) / height; height = maxH }
   464→          const canvas = document.createElement('canvas')
   465→          canvas.width = width; canvas.height = height
   466→          const ctx = canvas.getContext('2d')!
   467→          ctx.drawImage(img, 0, 0, width, height)
   468→          canvas.toBlob(async (blob) => {
   469→            if (!blob) { setUploadingProof(false); return }
   470→            const fd = new FormData()
   471→            fd.append('file', new File([blob], 'proof.jpg', { type: 'image/jpeg' }))
