    68→function fmtRelative(dateStr: string): string {
    69→  try {
    70→    const diff = Date.now() - new Date(dateStr).getTime()
    71→    const sec = Math.floor(diff / 1000)
    72→    if (sec < 60) return 'الآن'
    73→    const min = Math.floor(sec / 60)
    74→    if (min < 60) return `قبل ${min} دقيقة`
    75→    const hr = Math.floor(min / 60)
    76→    if (hr < 24) return `قبل ${hr} ساعة`
    77→    const day = Math.floor(hr / 24)
    78→    if (day < 30) return `قبل ${day} يوم`
    79→    return new Date(dateStr).toLocaleDateString('ar-EG', { day: 'numeric', month: 'short' })
    80→  } catch { return '' }
    81→}
