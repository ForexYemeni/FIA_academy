   106→  qualifications TEXT,

   108→  avatar TEXT,

   136→
   137→  const copySql = () => {
   138→    navigator.clipboard.writeText(SETUP_SQL).then(() => {
   139→      setSetupSqlCopied(true)
   140→      toast.success('تم نسخ SQL! الصقه في Supabase SQL Editor واضغط Run')
   141→      setTimeout(() => setSetupSqlCopied(false), 4000)
   142→    })
   143→  }
   144→
   145→  // Dialogs
   146→  const [moduleDialog, setModuleDialog] = useState<{ open: boolean; editing: Module | null }>({ open: false, editing: null })
   147→  const [lessonDialog, setLessonDialog] = useState<{ open: boolean; editing: Lesson | null; moduleId: string } | null>(null)
   148→  const [previewLesson, setPreviewLesson] = useState<Lesson | null>(null)
   149→
   150→  const loadModules = async () => {
   151→    setLoading(true)
   152→    setNeedsSetup(false)
   153→    try {
   154→      const res = await fetch(`/api/modules?course_id=${courseId}`)
   155→      const data = await res.json()
   156→      if (data.needs_setup) {
   157→        setNeedsSetup(true)
   158→        setModules([])
   159→        setLoading(false)
   160→        return
   161→      }
   162→      const mods = Array.isArray(data) ? data : []
   163→      setModules(mods)
   164→      if (mods.length > 0 && expanded.size === 0) {
   165→        setExpanded(new Set([mods[0].id]))
   166→      }
   167→      const lessonRes = await fetch(`/api/lessons?course_id=${courseId}`)
   168→      const allLessons = await lessonRes.json()
   169→      const byModule: Record<string, Lesson[]> = {}
   170→      ;(allLessons || []).forEach((l: Lesson) => {
   171→        const mid = l.module_id
   172→        if (!mid) return
   173→        if (!byModule[mid]) byModule[mid] = []
   174→        byModule[mid].push(l)
   175→      })
   176→      setLessonsByModule(byModule)
   177→    } catch {
   178→      // ignore
   179→    } finally {
   180→      setLoading(false)
   181→    }
   182→  }
   183→
   184→  useEffect(() => {
   185→    loadModules()
   186→  }, [courseId])
   187→
   188→  const toggleModule = (id: string) => {
   189→    setExpanded((prev) => {
   190→      const next = new Set(prev)
   191→      if (next.has(id)) next.delete(id)
   192→      else next.add(id)
   193→      return next
   194→    })
   195→  }
   196→
   197→  const saveModule = async (data: any) => {
   198→    try {
   199→      const res = await fetch('/api/modules', {
   200→        method: moduleDialog.editing ? 'PUT' : 'POST',
   201→        headers: { 'Content-Type': 'application/json' },
   202→        body: JSON.stringify(moduleDialog.editing ? { id: moduleDialog.editing.id, ...data } : { course_id: courseId, ...data }),
   203→      })
   204→      const result = await res.json()
   205→      if (res.ok) {
   206→        toast.success(moduleDialog.editing ? 'تم تحديث الوحدة' : 'تمت إضافة الوحدة')
   207→        setModuleDialog({ open: false, editing: null })
   208→        loadModules()
   209→      } else if (result.needs_setup) {
   210→        setNeedsSetup(true)
   211→        setModuleDialog({ open: false, editing: null })
   212→        toast.error('يجب تنفيذ SQL أولاً')
   213→      } else {
   214→        toast.error(result.error || 'فشل الحفظ')
   215→      }
   216→    } catch {
   217→      toast.error('خطأ في الاتصال بالخادم')
   218→    }
   219→  }
   220→
   221→  const deleteModule = async (id: string) => {
   222→    if (!confirm('حذف الوحدة سيحذف جميع دروسها. متابعة؟')) return
   223→    await fetch('/api/modules', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id }) })
   224→    toast.success('تم الحذف')
   225→    loadModules()
   226→  }
   227→
   228→  const saveLesson = async (data: any, moduleId: string, editing: Lesson | null) => {
   229→    try {
   230→      // The /api/lessons route requires { id, module_id, course_id } for PUT
   231→      // and { module_id, course_id, ... } for POST. Always include course_id
   232→      // to avoid an expensive full-table scan on the server.
   233→      const body = editing
   234→        ? { id: editing.id, module_id: moduleId, course_id: courseId, ...data }
   235→        : { module_id: moduleId, course_id: courseId, ...data }
   236→      const res = await fetch('/api/lessons', {
   237→        method: editing ? 'PUT' : 'POST',
   238→        headers: { 'Content-Type': 'application/json' },
   239→        body: JSON.stringify(body),
   240→      })
   241→      if (res.ok) {
   242→        toast.success(editing ? 'تم تحديث الدرس' : 'تمت إضافة الدرس')
   243→        setLessonDialog(null)
   244→        loadModules()
   245→      } else {
   246→        const result = await res.json().catch(() => ({}))
   247→        toast.error(result.error || 'فشل الحفظ')
   248→      }
   249→    } catch {
   250→      toast.error('خطأ في الاتصال')
   251→    }
   252→  }
   253→
   254→  const deleteLesson = async (id: string, moduleId: string) => {
   255→    if (!confirm('حذف الدرس؟')) return
   256→    try {
   257→      const res = await fetch('/api/lessons', {
   258→        method: 'DELETE',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({ id, module_id: moduleId, course_id: courseId }),
   261→      })
   262→      if (!res.ok) {
   263→        const result = await res.json().catch(() => ({}))
   264→        toast.error(result.error || 'فشل الحذف')
   265→        return
   266→      }
   267→      toast.success('تم الحذف')
   268→      loadModules()
   269→    } catch {
   270→      toast.error('خطأ في الاتصال')
   271→    }
   272→  }
   273→
   274→  if (loading) {
   275→    return (
   276→      <div className="text-center py-12">
   277→        <Loader2 className="w-8 h-8 animate-spin mx-auto text-navy mb-3" />
   278→        <p className="text-sm text-muted-foreground">جاري تحميل محتوى الدورة...</p>
   279→      </div>
   280→    )
   281→  }
   282→
   283→  // ===== Database setup needed =====
   284→  if (needsSetup) {
   285→    return (
   286→      <div className="space-y-4">
   287→        <Card className="p-8 text-center bg-amber-50 border-2 border-amber-300">
   288→          <div className="w-16 h-16 mx-auto rounded-2xl bg-amber-500 flex items-center justify-center mb-4">
   289→            <Database className="w-8 h-8 text-white" />
   290→          </div>
   291→          <h3 className="font-bold text-navy text-lg mb-2">تهيئة قاعدة البيانات مطلوبة</h3>
   292→          <p className="text-sm text-muted-foreground mb-5 max-w-md mx-auto">
   293→            لتفعيل نظام الوحدات والدروس، يجب إنشاء جداول في Supabase. اضغط زر النسخ ثم الصق في Supabase SQL Editor واضغط Run.
   294→          </p>
   295→          <div className="flex gap-2 justify-center flex-wrap">
   296→            <Button onClick={copySql} className="bg-navy hover:bg-navy-light">
   297→              {setupSqlCopied ? <CheckCircle2 className="w-4 h-4 ml-1" /> : <Copy className="w-4 h-4 ml-1" />}
   298→              {setupSqlCopied ? 'تم النسخ!' : 'نسخ SQL'}
   299→            </Button>
   300→            <a href="https://supabase.com/dashboard/project/tzquudmfazysjogorrxz/sql/new" target="_blank" rel="noopener noreferrer">
   301→              <Button variant="outline" className="border-navy/30 text-navy hover:bg-navy/5">
   302→                <ExternalLink className="w-4 h-4 ml-1" /> فتح SQL Editor
   303→              </Button>
   304→            </a>
   305→            <Button variant="outline" onClick={loadModules} className="border-navy/30 text-navy hover:bg-navy/5">
   306→              <RefreshCw className="w-4 h-4 ml-1" /> إعادة المحاولة
   307→            </Button>
   308→          </div>
   309→          <details className="mt-5 text-right">
   310→            <summary className="text-xs text-muted-foreground cursor-pointer hover:text-navy">عرض SQL الكامل</summary>
   311→            <pre className="mt-3 text-xs bg-navy text-white p-4 rounded-lg overflow-x-auto max-h-72 overflow-y-auto font-mono" dir="ltr">{SETUP_SQL}</pre>
   312→          </details>
   313→        </Card>
   314→      </div>
   315→    )
   316→  }
   317→
   318→  return (
   319→    <div className="space-y-4">
   320→      <div className="flex items-center justify-between">
   321→        <div>
   322→          <h3 className="font-bold text-navy text-lg">محتوى الدورة</h3>
   323→          <p className="text-sm text-muted-foreground">{modules.length} وحدة، {Object.values(lessonsByModule).flat().length} درس</p>
   324→        </div>
   325→        <Button
   326→          className="bg-navy hover:bg-navy-light"
   327→          onClick={() => setModuleDialog({ open: true, editing: null })}
   328→        >
   329→          <Plus className="w-4 h-4 ml-1" /> وحدة جديدة
   330→        </Button>
   331→      </div>
   332→
   333→      {modules.length === 0 ? (
   334→        <Card className="p-12 text-center">
   335→          <FileText className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   336→          <p className="text-muted-foreground mb-4">لا توجد وحدات في هذه الدورة بعد</p>
   337→          <Button className="bg-navy hover:bg-navy-light" onClick={() => setModuleDialog({ open: true, editing: null })}>
   338→            <Plus className="w-4 h-4 ml-1" /> أضف أول وحدة
   339→          </Button>
   340→        </Card>
   341→      ) : (
   342→        modules.map((m, idx) => {
   343→          const isExpanded = expanded.has(m.id)
   344→          const lessons = lessonsByModule[m.id] || []
   345→          return (
   346→            <Card key={m.id} className="overflow-hidden">
   347→              {/* Module header */}
   348→              <div className="flex items-center justify-between p-4 bg-muted/30 border-b border-border">
   349→                <button onClick={() => toggleModule(m.id)} className="flex items-center gap-3 flex-1 text-right">
   350→                  <GripVertical className="w-5 h-5 text-muted-foreground" />
   351→                  <div className="w-8 h-8 rounded-lg bg-navy text-gold flex items-center justify-center text-sm font-bold flex-shrink-0">
   352→                    {idx + 1}
   353→                  </div>
   354→                  <div className="flex-1">
   355→                    <div className="font-bold text-navy">{m.title}</div>
   356→                    <div className="text-xs text-muted-foreground">{m.description || `${lessons.length} درس`}</div>
   357→                  </div>
   358→                  {isExpanded ? <ChevronUp className="w-5 h-5" /> : <ChevronDown className="w-5 h-5" />}
   359→                </button>
   360→                <div className="flex gap-1">
   361→                  <Button size="icon" variant="ghost" onClick={() => setModuleDialog({ open: true, editing: m })}>
   362→                    <Edit className="w-4 h-4 text-navy" />
   363→                  </Button>
   364→                  <Button size="icon" variant="ghost" onClick={() => deleteModule(m.id)}>
   365→                    <Trash2 className="w-4 h-4 text-red-500" />
   366→                  </Button>
   367→                </div>
   368→              </div>
   369→
   370→              {/* Module lessons */}
   371→              {isExpanded && (
   372→                <div className="p-4 space-y-2">
   373→                  {lessons.length === 0 ? (
   374→                    <div className="text-center py-6 text-muted-foreground text-sm">
   375→                      لا توجد دروس في هذه الوحدة
   376→                    </div>
   377→                  ) : (
   378→                    lessons.map((l, li) => {
   379→                      const type = LESSON_TYPES.find((t) => t.value === l.lesson_type) || LESSON_TYPES[0]
   380→                      const Icon = type.icon
   381→                      return (
   382→                        <div key={l.id} className="flex items-center justify-between p-3 bg-white border border-border rounded-lg hover:shadow-sm transition-shadow">
   383→                          <div className="flex items-center gap-3 flex-1">
   384→                            <GripVertical className="w-4 h-4 text-muted-foreground/50" />
   385→                            <Badge className={type.color}><Icon className="w-3 h-3 ml-1" /> {type.label}</Badge>
   386→                            <div className="flex-1">
   387→                              <div className="font-medium text-navy text-sm flex items-center gap-2">
   388→                                {l.title}
   389→                                {l.is_preview && <Badge variant="outline" className="text-xs">معاينة مجانية</Badge>}
   390→                              </div>
   391→                              {l.duration_minutes ? (
   392→                                <div className="text-xs text-muted-foreground">{l.duration_minutes} دقيقة</div>
   393→                              ) : null}
   394→                            </div>
   395→                          </div>
   396→                          <div className="flex gap-1">
   397→                            <Button size="icon" variant="ghost" onClick={() => setPreviewLesson(l)} title="معاينة الدرس">
   398→                              <Eye className="w-4 h-4 text-gold" />
   399→                            </Button>
   400→                            <Button size="icon" variant="ghost" onClick={() => setLessonDialog({ open: true, editing: l, moduleId: m.id })}>
   401→                              <Edit className="w-4 h-4 text-navy" />
   402→                            </Button>
   403→                            <Button size="icon" variant="ghost" onClick={() => deleteLesson(l.id, m.id)}>
   404→                              <Trash2 className="w-4 h-4 text-red-500" />
   405→                            </Button>
   406→                          </div>
   407→                        </div>
   408→                      )
   409→                    })
   410→                  )}
   411→                  <Button
   412→                    variant="outline"
   413→                    size="sm"
   414→                    className="w-full mt-2 border-dashed"
   415→                    onClick={() => setLessonDialog({ open: true, editing: null, moduleId: m.id })}
   416→                  >
   417→                    <Plus className="w-4 h-4 ml-1" /> إضافة درس
   418→                  </Button>
   419→                </div>
   420→              )}
   421→            </Card>
   422→          )
   423→        })
   424→      )}
   425→
   426→      {/* Module dialog */}
   427→      {moduleDialog.open && (
   428→        <ModuleDialog
   429→          open={moduleDialog.open}
   430→          onOpenChange={(o) => setModuleDialog({ open: o, editing: moduleDialog.editing })}
   431→          editing={moduleDialog.editing}
   432→          onSave={saveModule}
   433→        />
   434→      )}
   435→
   436→      {/* Lesson dialog */}
   437→      {lessonDialog?.open && (
   438→        <LessonDialog
   439→          open={lessonDialog.open}
   440→          onOpenChange={(o) => setLessonDialog(o ? { ...lessonDialog, open: o } : null)}
   441→          editing={lessonDialog.editing}
   442→          moduleId={lessonDialog.moduleId}
   443→          onSave={(data: any) => saveLesson(data, lessonDialog.moduleId, lessonDialog.editing)}
   444→        />
   445→      )}
   446→
   447→      {/* Lesson preview modal */}
   448→      <LessonPreviewModal
   449→        open={!!previewLesson}
   450→        onOpenChange={(o) => { if (!o) setPreviewLesson(null) }}
   451→        lesson={previewLesson}
   452→        modules={modules}
   453→        onNavigate={(l) => setPreviewLesson(l)}
   454→      />
   455→    </div>
   456→  )
   457→}

   459→// ============== Module Dialog ==============
   460→function ModuleDialog({ open, onOpenChange, editing, onSave }: any) {
   461→  const [title, setTitle] = useState(editing?.title || '')
   462→  const [description, setDescription] = useState(editing?.description || '')
   463→  const [position, setPosition] = useState(editing?.position || 0)
   464→
   465→  useEffect(() => {
   466→    setTitle(editing?.title || '')
   467→    setDescription(editing?.description || '')
   468→    setPosition(editing?.position || 0)
   469→  }, [editing])
   470→
   471→  return (
   472→    <Dialog open={open} onOpenChange={onOpenChange}>
   473→      <DialogContent className="sm:max-w-md">
   474→        <DialogHeader>
   475→          <DialogTitle>{editing ? 'تعديل الوحدة' : 'وحدة جديدة'}</DialogTitle>
   476→        </DialogHeader>
   477→        <div className="space-y-4 py-4">
   478→          <div>
   479→            <Label>عنوان الوحدة *</Label>
   480→            <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="مثال: الوحدة الأولى - الأساسيات" className="mt-1" />
   481→          </div>
   482→          <div>
   483→            <Label>وصف مختصر</Label>

   506→function LessonDialog({ open, onOpenChange, editing, onSave }: any) {
   507→  const [title, setTitle] = useState(editing?.title || '')
   508→  const [lessonType, setLessonType] = useState<'video' | 'text' | 'exam' | 'pdf'>(editing?.lesson_type || 'video')

   510→  const [videoUrl, setVideoUrl] = useState(editing?.video_url || '')
   511→  const [pdfUrl, setPdfUrl] = useState(editing?.pdf_url || '')
   512→  const [duration, setDuration] = useState(editing?.duration_minutes || 0)
   513→  const [isPreview, setIsPreview] = useState(editing?.is_preview || false)
   514→  const [position, setPosition] = useState(editing?.position || 0)
   515→  const [savingPdf, setSavingPdf] = useState(false)
   516→
   517→  useEffect(() => {
   518→    setTitle(editing?.title || '')
   519→    setLessonType(editing?.lesson_type || 'video')
   520→    setContent(editing?.content || '')
   521→    setVideoUrl(editing?.video_url || '')
   522→    setPdfUrl(editing?.pdf_url || '')
   523→    setDuration(editing?.duration_minutes || 0)
   524→    setIsPreview(editing?.is_preview || false)
   525→    setPosition(editing?.position || 0)
   526→  }, [editing])
   527→
   528→  const handlePdfUpload = async (file: File) => {
   529→    setSavingPdf(true)
   530→    try {
   531→      const fd = new FormData()
   532→      fd.append('file', file)
   533→      fd.append('folder', 'pdfs')
   534→      fd.append('caption', `FIA Lesson PDF: ${title || ''}`)
   535→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   536→      const data = await res.json().catch(() => ({}))
   537→      if (data.url) {
   538→        setPdfUrl(data.url)
   539→        toast.success('تم رفع PDF')
   540→      } else {
   541→        toast.error('فشل رفع PDF: ' + (data.error || ''))
   542→      }
   543→    } catch {
   544→      toast.error('خطأ في الاتصال')
   545→    } finally {
   546→      setSavingPdf(false)
   547→    }
   548→  }
   549→
   550→  const [uploadingVideo, setUploadingVideo] = useState(false)
   551→
   552→  // Upload a video file to the Telegram bot via the server-side /api/upload/image
   553→  // endpoint (which uses Telegram's sendVideo internally) and store the resulting
   554→  // public URL in the lesson's video_url field. The trainer can also paste a
   555→  // Telegram file URL / direct link directly into the input below.

   557→    setUploadingVideo(true)
   558→    try {
   559→      const fd = new FormData()
   560→      fd.append('file', file)
   561→      fd.append('folder', 'videos')
   562→      fd.append('caption', `FIA Lesson Video: ${title || ''}`)
   563→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   564→      const data = await res.json().catch(() => ({}))
   565→      if (data.url) {
   566→        setVideoUrl(data.url)
   567→        toast.success('تم رفع الفيديو بنجاح')
   568→      } else {
   569→        toast.error('فشل رفع الفيديو: ' + (data.error || ''))
   570→      }
   571→    } catch {
   572→      toast.error('خطأ في الاتصال')
   573→    } finally {
   574→      setUploadingVideo(false)
   575→    }
   576→  }
   577→
   578→  return (
   579→    <Dialog open={open} onOpenChange={onOpenChange}>
   580→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   581→        <DialogHeader>
   582→          <DialogTitle>{editing ? 'تعديل الدرس' : 'درس جديد'}</DialogTitle>
   583→        </DialogHeader>
   584→        <div className="space-y-4 py-4">
   585→          <div>
   586→            <Label>عنوان الدرس *</Label>
   587→            <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="مثال: مقدمة في الإنعاش القلبي" className="mt-1" />
   588→          </div>
   589→
   590→          <div>
   591→            <Label>نوع الدرس</Label>
   592→            <div className="grid grid-cols-4 gap-2 mt-1">

   594→                const Icon = t.icon
   595→                const active = lessonType === t.value
   596→                return (
   597→                  <button
   598→                    key={t.value}
   599→                    type="button"
   600→                    onClick={() => setLessonType(t.value as any)}
   601→                    className={`flex flex-col items-center gap-1 p-2 rounded-lg border-2 transition-all ${
   602→                      active ? `${t.color} border-current` : 'border-border bg-muted/30 hover:border-muted'

   605→                    <Icon className="w-5 h-5" />

   619→                  accept="video/*"
   620→                  className="hidden"

   622→                  onChange={(e) => { const f = e.target.files?.[0]; if (f) handleVideoUpload(f); e.target.value = '' }}
   623→                />
   624→                <Button
   625→                  variant="outline"

   746→// ============== Visual Exam Builder (no JSON editing) ==============
   747→function ExamBuilder({ value, onChange }: { value?: string | null; onChange: (v: string) => void }) {
   748→  const [questions, setQuestions] = useState<any[]>(() => {
   749→    if (!value) return []
   750→    try {
   751→      const parsed = JSON.parse(value)
   752→      return Array.isArray(parsed) ? parsed : []
   753→    } catch {
   754→      return []
   755→    }
   756→  })
   757→
   758→  const updateJson = (qs: any[]) => {
   759→    setQuestions(qs)
   760→    onChange(JSON.stringify(qs))
   761→  }
   762→
   763→  const addQuestion = () => {
   764→    updateJson([...questions, { q: '', options: ['', '', '', ''], answer: 0 }])
   765→  }
   766→
   767→  const updateQuestion = (i: number, field: string, val: any) => {
   768→    const qs = [...questions]
   769→    qs[i] = { ...qs[i], [field]: val }
   770→    updateJson(qs)
   771→  }
   772→
   773→  const removeQuestion = (i: number) => {
   774→    updateJson(questions.filter((_, idx) => idx !== i))
   775→  }
   776→
   777→  return (
   778→    <div className="space-y-3">
   779→      <div className="flex items-center justify-between">
   780→        <Label className="flex items-center gap-1">
   781→          <HelpCircle className="w-4 h-4 text-purple-600" /> أسئلة الاختبار
   782→        </Label>
   783→        <Button type="button" size="sm" variant="outline" onClick={addQuestion}>
   784→          <Plus className="w-3.5 h-3.5 ml-1" /> سؤال جديد
   785→        </Button>
   786→      </div>
   787→
   788→      {questions.length === 0 ? (
   789→        <div className="text-center py-8 bg-purple-500/5 border border-purple-500/30 rounded-lg">
   790→          <HelpCircle className="w-10 h-10 text-purple-600/30 mx-auto mb-2" />
   791→          <p className="text-sm text-muted-foreground">لا توجد أسئلة بعد</p>
   792→          <p className="text-xs text-muted-foreground mt-1">اضغط "سؤال جديد" لإضافة أول سؤال</p>
   793→        </div>
   794→      ) : (
   795→        <div className="space-y-3 max-h-96 overflow-y-auto fia-scroll pr-1">
   796→          {questions.map((q, qi) => (
   797→            <Card key={qi} className="p-3 border-purple-500/20">
   798→              <div className="flex items-start gap-2 mb-2">
   799→                <Badge variant="outline" className="text-xs bg-purple-500/5 border-purple-500/30 text-purple-600 flex-shrink-0">
   800→                  سؤال {qi + 1}
   801→                </Badge>
   802→                <Input
   803→                  value={q.q || ''}
   804→                  onChange={(e) => updateQuestion(qi, 'q', e.target.value)}
   805→                  placeholder="اكتب السؤال هنا..."
   806→                  className="flex-1 text-sm"
   807→                />
   808→                <Button type="button" size="icon" variant="ghost" onClick={() => removeQuestion(qi)} className="flex-shrink-0">
   809→                  <Trash2 className="w-4 h-4 text-red-500" />
   810→                </Button>
   811→              </div>
   812→
   813→              <div className="space-y-1.5">
   814→                <div className="text-xs text-muted-foreground">الإجابات (اضغط الدائرة لتحديد الإجابة الصحيحة):</div>
   815→                {(q.options || []).map((opt: string, oi: number) => (
   816→                  <div key={oi} className="flex items-center gap-2">
   817→                    <button
   818→                      type="button"
   819→                      onClick={() => updateQuestion(qi, 'answer', oi)}
   820→                      className={`w-5 h-5 rounded-full border-2 flex-shrink-0 flex items-center justify-center transition-colors ${
   821→                        q.answer === oi ? 'bg-green-500 border-green-500' : 'border-muted-foreground/40 hover:border-purple-400'
   822→                      }`}
   823→                      title="تحديد كإجابة صحيحة"
   824→                    >
   825→                      {q.answer === oi && <CheckCircle2 className="w-3 h-3 text-white" />}
   826→                    </button>
   827→                    <Input
   828→                      value={opt}
   829→                      onChange={(e) => {
   830→                        const newOptions = [...(q.options || [])]
   831→                        newOptions[oi] = e.target.value
   832→                        updateQuestion(qi, 'options', newOptions)
   833→                      }}
   834→                      placeholder={`الخيار ${oi + 1}`}
   835→                      className={`text-sm flex-1 ${q.answer === oi ? 'border-green-500/40 bg-green-500/5' : ''}`}
   836→                      dir="rtl"
   837→                    />
   838→                  </div>
   839→                ))}
   840→              </div>
   841→            </Card>
   842→          ))}
   843→        </div>
   844→      )}
   845→    </div>
   846→  )
   847→}
