   747→   747→function ExamBuilder({ value, onChange }: { value?: string | null; onChange: (v: string) => void }) {
   748→   748→  const [questions, setQuestions] = useState<any[]>(() => {
   749→   749→    if (!value) return []
   750→   750→    try {
   751→   751→      const parsed = JSON.parse(value)
   752→   752→      return Array.isArray(parsed) ? parsed : []
   753→   753→    } catch {
   754→   754→      return []
   755→   755→    }
   756→   756→  })
   757→   757→
   758→   758→  const updateJson = (qs: any[]) => {
   759→   759→    setQuestions(qs)
   760→   760→    onChange(JSON.stringify(qs))
   761→   761→  }
   762→   762→
   763→   763→  const addQuestion = () => {
   764→   764→    updateJson([...questions, { q: '', options: ['', '', '', ''], answer: 0 }])
   765→   765→  }
   766→   766→
   767→   767→  const updateQuestion = (i: number, field: string, val: any) => {
   768→   768→    const qs = [...questions]
   769→   769→    qs[i] = { ...qs[i], [field]: val }
   770→   770→    updateJson(qs)
   771→   771→  }
   772→   772→
   773→   773→  const removeQuestion = (i: number) => {
   774→   774→    updateJson(questions.filter((_, idx) => idx !== i))
   775→   775→  }
   776→   776→
   777→   777→  return (
   778→   778→    <div className="space-y-3">
   779→   779→      <div className="flex items-center justify-between">
   780→   780→        <Label className="flex items-center gap-1">
   781→   781→          <HelpCircle className="w-4 h-4 text-purple-600" /> أسئلة الاختبار
   782→   782→        </Label>
   783→   783→        <Button type="button" size="sm" variant="outline" onClick={addQuestion}>
   784→   784→          <Plus className="w-3.5 h-3.5 ml-1" /> سؤال جديد
   785→   785→        </Button>
   786→   786→      </div>
   787→   787→
   788→   788→      {questions.length === 0 ? (
   789→   789→        <div className="text-center py-8 bg-purple-500/5 border border-purple-500/30 rounded-lg">
   790→   790→          <HelpCircle className="w-10 h-10 text-purple-600/30 mx-auto mb-2" />
   791→   791→          <p className="text-sm text-muted-foreground">لا توجد أسئلة بعد</p>
   792→   792→          <p className="text-xs text-muted-foreground mt-1">اضغط "سؤال جديد" لإضافة أول سؤال</p>
   793→   793→        </div>
   794→   794→      ) : (
   795→   795→        <div className="space-y-3 max-h-96 overflow-y-auto fia-scroll pr-1">
   796→   796→          {questions.map((q, qi) => (
   797→   797→            <Card key={qi} className="p-3 border-purple-500/20">
   798→   798→              <div className="flex items-start gap-2 mb-2">
   799→   799→                <Badge variant="outline" className="text-xs bg-purple-500/5 border-purple-500/30 text-purple-600 flex-shrink-0">
   800→   800→                  سؤال {qi + 1}
   801→   801→                </Badge>
   802→   802→                <Input
   803→   803→                  value={q.q || ''}
   804→   804→                  onChange={(e) => updateQuestion(qi, 'q', e.target.value)}
   805→   805→                  placeholder="اكتب السؤال هنا..."
   806→   806→                  className="flex-1 text-sm"
   807→   807→                />
   808→   808→                <Button type="button" size="icon" variant="ghost" onClick={() => removeQuestion(qi)} className="flex-shrink-0">
   809→   809→                  <Trash2 className="w-4 h-4 text-red-500" />
   810→   810→                </Button>
