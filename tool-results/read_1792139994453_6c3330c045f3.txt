    32→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    33→// Show only the human-readable text in the UI, but preserve the extra data on save.
    34→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    35→const splitBio = (raw: string) => {
    36→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    37→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    38→}
    39→const joinBio = (display: string, original?: string) => {
    40→  if (!original) return display
    41→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    42→  if (i === -1) return display
    43→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    44→}

    46→export function TrainerDashboard() {
    47→  const [active, setActive] = useState('overview')
    48→  const [contentCourseId, setContentCourseId] = useState<string | null>(null)
    49→  const user = useAuth((s) => s.user)
    50→  const updateUser = useAuth((s) => s.updateUser)
    51→  const logout = useAuth((s) => s.logout)
    52→  const [courses, setCourses] = useState<any[]>([])
    53→  const [students, setStudents] = useState<any[]>([])
    54→  const [editing, setEditing] = useState<any | null>(null)
    55→  const [isDialogOpen, setIsDialogOpen] = useState(false)
    56→  const [profile, setProfile] = useState({ name: user?.name || '', phone: user?.phone || '', title: user?.title || '', bio: splitBio(user?.bio || ''), specialization: user?.title || '' })
    57→
    58→  const loadCourses = () => {
    59→    fetch('/api/courses').then((r) => r.json()).then((d) => {
    60→      const all = Array.isArray(d) ? d : []
    61→      // Filter to ONLY this trainer's courses
    62→      const mine = all.filter((c: any) => c.trainer_id === user?.id)
    63→      setCourses(mine)
    64→    }).catch(() => {})

    67→    fetch('/api/students?role=student').then((r) => r.json()).then((d) => setStudents(Array.isArray(d) ? d : [])).catch(() => {})
    68→  }
    69→
    70→  useEffect(() => {
    71→    loadCourses()
    72→    loadStudents()

    76→    setContentCourseId(courseId)
    77→    setActive('content')
    78→  }
    79→
    80→  return (
    81→    <DashboardShell

    83→      active={active}
    84→      onNavigate={(id) => { setActive(id); if (id !== 'content') setContentCourseId(null) }}

    91→        <div className="space-y-4">

    98→      {active === 'students' && <TrainerStudents students={students} courses={courses} />}
    99→      {active === 'exams' && <TrainerExams />}
   100→      {active === 'profile' && <TrainerProfile profile={profile} setProfile={setProfile} user={user} updateUser={updateUser} />}
   101→      {active === 'notifications' && <TrainerNotifs />}
   102→      {active === 'settings' && <TrainerAccountSettings user={user} updateUser={updateUser} logout={logout} />}
   103→
   104→      {isDialogOpen && (
   105→        <CourseDialog
   106→          open={isDialogOpen}
   107→          onOpenChange={setIsDialogOpen}

   154→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   155→  const [deleting, setDeleting] = useState<string | null>(null)
   156→
   157→  const handleDelete = async (id: string) => {
   158→    if (!confirm('هل أنت متأكد من حذف هذه الدورة؟')) return
   159→    setDeleting(id)
   160→    try {
   161→      const res = await fetch('/api/courses', {
   162→        method: 'DELETE',
   163→        headers: { 'Content-Type': 'application/json' },
   164→        body: JSON.stringify({ id }),
   165→      })
   166→      if (res.ok) {
   167→        toast.success('تم حذف الدورة')
   168→        onRefresh()
   169→      } else {
   170→        toast.error('فشل الحذف')
   171→      }
   172→    } catch {
   173→      toast.error('خطأ في الاتصال')
   174→    } finally {
   175→      setDeleting(null)
   176→    }
   177→  }
   178→
   179→  return (
   180→    <SectionCard
   181→      title="دوراتي التدريبية"
   182→      description="إدارة الدورات التي تدرّسها"
   183→      action={
   184→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   185→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   186→        </Button>
   187→      }
   188→    >
   189→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   190→        {courses.map((c: any) => (
   191→          <Card key={c.id} className="overflow-hidden card-hover">
   192→            <div className="h-32 bg-navy relative">
   193→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   194→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   195→            </div>
   196→            <div className="p-4">
   197→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   198→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   199→                <span>{c.enrolled || 0} متدرب</span>
   200→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   201→              </div>
   202→              <div className="space-y-2">
   203→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   204→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   205→                </Button>
   206→                <div className="flex gap-2">
   207→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   208→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   209→                  </Button>
   210→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => handleDelete(c.id)} disabled={deleting === c.id}>
   211→                    <Trash2 className="w-3.5 h-3.5" />
   212→                  </Button>
   213→                </div>
   214→              </div>
   215→            </div>
   216→          </Card>
   217→        ))}
   218→      </div>
   219→    </SectionCard>
   220→  )
   221→}

   223→function CourseDialog({ open, onOpenChange, course, trainerId, onSaved }: any) {
   224→  const [form, setForm] = useState<any>(
   225→    course || {
   226→      title: '',
   227→      short_description: '',
   228→      description: '',
   229→      level: 'متوسط',
   230→      price_type: 'مجانية',
   231→      price: 0,
   232→      currency: 'ر.ي',
   233→      duration_hours: 0,
   234→      pass_score: 60,
   235→      cover: '',
   236→      status: 'منشورة',
   237→    }
   238→  )
   239→  const [saving, setSaving] = useState(false)
   240→
   241→  const handleUploadCover = async (file: File) => {
   242→    const fd = new FormData()
   243→    fd.append('file', file)
   244→    fd.append('folder', 'courses')
   245→    const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   246→    const data = await res.json()
   247→    if (data.url) {
   248→      setForm({ ...form, cover: data.url })
   249→      toast.success('تم رفع الصورة')
   250→    } else {
   251→      toast.error('فشل رفع الصورة')
   252→    }
   253→  }
   254→
   255→  const save = async () => {
   256→    if (!form.title) {
   257→      toast.error('أدخل عنوان الدورة')
   258→      return
   259→    }
   260→    setSaving(true)
   261→    try {
   262→      const slug = form.title.toLowerCase().replace(/\s+/g, '-').slice(0, 80)
   263→      const body = {
   264→        ...form,
   265→        slug,
   266→        trainer_id: trainerId,
   267→        modules_count: form.modules_count || 0,
   268→        lessons_count: form.lessons_count || 0,
   269→      }
   270→      const res = await fetch('/api/courses', {
   271→        method: course ? 'PUT' : 'POST',
   272→        headers: { 'Content-Type': 'application/json' },
   273→        body: JSON.stringify(course ? { id: course.id, ...body } : body),
   274→      })
   275→      if (res.ok) {
   276→        toast.success(course ? 'تم تحديث الدورة' : 'تم إنشاء الدورة')
   277→        onSaved()
   278→      } else {
   279→        toast.error('فشل الحفظ')
   280→      }
   281→    } catch {
   282→      toast.error('خطأ')
   283→    } finally {
   284→      setSaving(false)
   285→    }
   286→  }
   287→
   288→  return (
   289→    <Dialog open={open} onOpenChange={onOpenChange}>
   290→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   291→        <DialogHeader>
   292→          <DialogTitle>{course ? 'تعديل الدورة' : 'دورة جديدة'}</DialogTitle>
   293→        </DialogHeader>
   294→        <div className="space-y-4 py-4">
   295→          <div>
   296→            <Label>عنوان الدورة</Label>
   297→            <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} className="mt-1" />
   298→          </div>
   299→          <div>
   300→            <Label>وصف مختصر</Label>
   301→            <Input value={form.short_description || ''} onChange={(e) => setForm({ ...form, short_description: e.target.value })} className="mt-1" />
   302→          </div>
   303→          <div>
   304→            <Label>الوصف الكامل</Label>
   305→            <Textarea value={form.description || ''} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} className="mt-1" />
   306→          </div>
   307→          <div className="grid grid-cols-2 gap-3">
   308→            <div>
   309→              <Label>المستوى</Label>
   310→              <select
   311→                value={form.level}
   312→                onChange={(e) => setForm({ ...form, level: e.target.value })}
   313→                className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background"
   314→              >
   315→                <option value="مبتدئ">مبتدئ</option>
   316→                <option value="متوسط">متوسط</option>
   317→                <option value="متقدم">متقدم</option>
   318→              </select>
   319→            </div>
   320→            <div>
   321→              <Label>السعر</Label>
   322→              <select
   323→                value={form.price_type}
   324→                onChange={(e) => setForm({ ...form, price_type: e.target.value })}
   325→                className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background"
   326→              >
   327→                <option value="مجانية">مجانية</option>
   328→                <option value="مدفوعة">مدفوعة</option>
   329→              </select>
   330→            </div>
   331→          </div>
   332→          {form.price_type === 'مدفوعة' && (
   333→            <div>
   334→              <Label>المبلغ</Label>
   335→              <Input type="number" value={form.price || 0} onChange={(e) => setForm({ ...form, price: parseInt(e.target.value) || 0 })} className="mt-1" dir="ltr" />
   336→            </div>
   337→          )}
   338→          <div className="grid grid-cols-2 gap-3">
   339→            <div>
   340→              <Label>عدد الساعات</Label>
   341→              <Input type="number" value={form.duration_hours || 0} onChange={(e) => setForm({ ...form, duration_hours: parseInt(e.target.value) || 0 })} className="mt-1" dir="ltr" />
   342→            </div>
   343→            <div>
   344→              <Label>درجة النجاح (%)</Label>
   345→              <Input type="number" value={form.pass_score || 60} onChange={(e) => setForm({ ...form, pass_score: parseInt(e.target.value) || 60 })} className="mt-1" dir="ltr" />
   346→            </div>
   347→          </div>
   348→          <div>
   349→            <Label>صورة الغلاف</Label>
   350→            <div className="flex items-center gap-3 mt-1">
   351→              {form.cover && <img src={form.cover} alt="cover" className="w-16 h-16 object-cover rounded-md" />}
   352→              <label className="cursor-pointer">
   353→                <Button variant="outline" type="button" onClick={() => document.getElementById('cover-upload')?.click()}>
   354→                  <FileUp className="w-4 h-4 ml-1" /> رفع صورة
   355→                </Button>
   356→                <input
   357→                  id="cover-upload"
   358→                  type="file"
   359→                  accept="image/*"
   360→                  className="hidden"
   361→                  onChange={(e) => { const f = e.target.files?.[0]; if (f) handleUploadCover(f) }}
   362→                />
   363→              </label>
   364→            </div>
   365→          </div>
   366→        </div>
   367→        <DialogFooter>
   368→          <Button variant="outline" onClick={() => onOpenChange(false)}>إلغاء</Button>
   369→          <Button onClick={save} disabled={saving} className="bg-navy hover:bg-navy-light">
   370→            {saving ? 'جاري الحفظ...' : 'حفظ'}
   371→          </Button>
   372→        </DialogFooter>
   373→      </DialogContent>
   374→    </Dialog>

   376→}

   378→function TrainerStudents({ students, courses }: any) {

   382→    fetch('/api/enrollments').then((r) => r.json()).then((d) => {
   383→      const all = Array.isArray(d) ? d : []
   384→      // Filter to ONLY enrollments in this trainer's courses
   385→      const trainerCourseIds = courses.map((c: any) => c.id)
   386→      const mine = all.filter((e: any) => trainerCourseIds.includes(e.course_id))
   387→      setEnrollments(mine)
   388→    }).catch(() => {})
   389→  }, [courses])
   390→
   391→  // Helper: get all enrollments for a student
   392→  const studentEnrollments = (studentId: string) => enrollments.filter((e) => e.user_id === studentId)
   393→
   394→  return (
   395→    <SectionCard title="متدربوّ" description={`${students.length} متدرب${students.length === 1 ? '' : 'ين'}`}>
   396→      {courses.length === 0 ? (

   402→        <div className="text-center py-12">
   403→          <Users className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   404→          <p className="text-muted-foreground">لا يوجد متدربون بعد</p>
   405→        </div>
   406→      ) : (
   407→        <div className="space-y-4">

   424→                        <span dir="ltr">{s.email}</span>
   425→                        {s.phone && <span dir="ltr">{s.phone}</span>}

   428→                    </div>
   429→                    <Badge className="bg-blue-500/15 text-blue-600 border-blue-500/30">
   430→                      {studentEnrs.length} دورة
   431→                    </Badge>
   432→                  </div>
   433→                </div>

   436→                    <p className="text-sm text-muted-foreground text-center py-4">لم يسجّل في أي دورة بعد</p>

   441→                        if (!course) return null
   442→                        const isFree = course.price_type === 'مجانية' || course.price === 0
   443→                        return (
   444→                          <div key={e.id} className="flex items-center justify-between p-3 bg-white border border-border rounded-lg">

   566→function TrainerAccountSettings({ user, updateUser, logout }: any) {
   567→  const [currentPassword, setCurrentPassword] = useState('')
   568→  const [newEmail, setNewEmail] = useState('')
   569→  const [newPassword, setNewPassword] = useState('')
   570→  const [confirmPassword, setConfirmPassword] = useState('')
   571→  const [showCurrent, setShowCurrent] = useState(false)
   572→  const [showNew, setShowNew] = useState(false)
   573→  const [loading, setLoading] = useState(false)
   574→
   575→  const save = async () => {
   576→    if (!currentPassword) {
   577→      toast.error('أدخل كلمة المرور الحالية للتأكيد')
   578→      return
   579→    }
   580→    if (newPassword && newPassword !== confirmPassword) {
   581→      toast.error('كلمتا المرور الجديدتان غير متطابقتين')
   582→      return
   583→    }
   584→    if (newPassword && newPassword.length < 6) {
   585→      toast.error('كلمة المرور الجديدة يجب أن تكون 6 أحرف على الأقل')
   586→      return
   587→    }
   588→    if (!newEmail && !newPassword) {
   589→      toast.error('لا توجد تغييرات للحفظ')
   590→      return
   591→    }
   592→    setLoading(true)
   593→    try {
   594→      const res = await fetch('/api/auth/change-credentials', {
   595→        method: 'POST',
   596→        headers: { 'Content-Type': 'application/json' },
   597→        body: JSON.stringify({
   598→          userId: user.id,
   599→          currentEmail: user.email,
   600→          currentPassword,
   601→          newEmail: newEmail || undefined,
   602→          newPassword: newPassword || undefined,
   603→          role: 'trainer',
   604→        }),
   605→      })
   606→      const data = await res.json()
   607→      if (!res.ok) {
   608→        toast.error(data.error || 'فشل التحديث')
   609→        return
   610→      }
   611→      toast.success('تم تحديث بيانات الدخول بنجاح')
   612→      updateUser({
   613→        email: newEmail || user.email,
   614→        password: newPassword || user.password,
   615→      })
   616→      setCurrentPassword('')
   617→      setNewEmail('')
   618→      setNewPassword('')
   619→      setConfirmPassword('')
   620→      if (newEmail) {
   621→        setTimeout(() => {
   622→          logout()
   623→          toast.info('تم تحديث البريد. يرجى تسجيل الدخول بالبيانات الجديدة')
   624→        }, 2000)
   625→      }
   626→    } catch {
   627→      toast.error('خطأ في الاتصال')
   628→    } finally {
   629→      setLoading(false)
   630→    }
   631→  }
   632→
   633→  return (
   634→    <SectionCard title="الحساب والأمان" description="تغيير البريد الإلكتروني وكلمة المرور">
   635→      <div className="space-y-5 max-w-2xl">
   636→        {/* Current account info */}
   637→        <div className="bg-muted/30 p-4 rounded-lg border border-border">
   638→          <div className="text-xs text-muted-foreground mb-1">الحساب الحالي:</div>
   639→          <div className="flex items-center gap-3">
   640→            <div className="w-12 h-12 rounded-full bg-navy flex items-center justify-center text-gold font-bold text-xl border-2 border-gold/30">
   641→              {user?.name?.charAt(0)}
   642→            </div>
   643→            <div>
   644→              <div className="font-bold text-navy">{user?.name}</div>
   645→              <div className="text-xs text-muted-foreground" dir="ltr">{user?.email}</div>
   646→            </div>
   647→          </div>
   648→        </div>
   649→
   650→        {/* Security verification */}
   651→        <div>
   652→          <Label className="text-navy font-semibold">كلمة المرور الحالية *</Label>
   653→          <div className="relative mt-1">
   654→            <Input
   655→              type={showCurrent ? 'text' : 'password'}
   656→              value={currentPassword}
   657→              onChange={(e) => setCurrentPassword(e.target.value)}
   658→              placeholder="••••••••"
   659→              dir="ltr"
   660→            />
   661→            <button
   662→              type="button"
   663→              onClick={() => setShowCurrent(!showCurrent)}
   664→              className="absolute left-2 top-1/2 -translate-y-1/2 text-xs text-muted-foreground hover:text-navy"
   665→            >
   666→              {showCurrent ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   667→            </button>
   668→          </div>
   669→          <p className="text-xs text-muted-foreground mt-1">مطلوبة للتأكد من صلاحية التغيير</p>
   670→        </div>
   671→
   672→        {/* New email */}
   673→        <div className="border-t border-border pt-4">
   674→          <h3 className="font-bold text-navy text-sm mb-3 flex items-center gap-2">
   675→            <Mail className="w-4 h-4 text-gold" /> تغيير البريد الإلكتروني

   678→            <Label>البريد الجديد</Label>

   685→              className="mt-1"

   705→                  dir="ltr"
