    51→  const [lessonsByModule, setLessonsByModule] = useState<Record<string, Lesson[]>>({})
    52→  const [loading, setLoading] = useState(true)
    53→  const [needsSetup, setNeedsSetup] = useState(false)
    54→  const [setupSqlCopied, setSetupSqlCopied] = useState(false)
    55→  const [expanded, setExpanded] = useState<Set<string>>(new Set())
    56→

   136→
   137→  const copySql = () => {
   138→    navigator.clipboard.writeText(SETUP_SQL).then(() => {
   139→      setSetupSqlCopied(true)
   140→      toast.success('تم نسخ SQL! الصقه في Supabase SQL Editor واضغط Run')
   141→      setTimeout(() => setSetupSqlCopied(false), 4000)
   142→    })
   143→  }
   144→
   145→  // Dialogs
   146→  const [moduleDialog, setModuleDialog] = useState<{ open: boolean; editing: Module | null }>({ open: false, editing: null })
   147→  const [lessonDialog, setLessonDialog] = useState<{ open: boolean; editing: Lesson | null; moduleId: string } | null>(null)
   148→  const [previewLesson, setPreviewLesson] = useState<Lesson | null>(null)
   149→
   150→  const loadModules = async () => {
   151→    setLoading(true)
   152→    setNeedsSetup(false)
   153→    try {
   154→      const res = await fetch(`/api/modules?course_id=${courseId}`)
   155→      const data = await res.json()
   156→      if (data.needs_setup) {
   157→        setNeedsSetup(true)
   158→        setModules([])
   159→        setLoading(false)
   160→        return
   161→      }
   162→      const mods = Array.isArray(data) ? data : []
   163→      setModules(mods)
   164→      if (mods.length > 0 && expanded.size === 0) {
   165→        setExpanded(new Set([mods[0].id]))
   166→      }
   167→      const lessonRes = await fetch(`/api/lessons?course_id=${courseId}`)
   168→      const allLessons = await lessonRes.json()
   169→      const byModule: Record<string, Lesson[]> = {}
   170→      ;(allLessons || []).forEach((l: Lesson) => {
   171→        if (!byModule[l.module_id]) byModule[l.module_id] = []
   172→        byModule[l.module_id].push(l)
   173→      })
   174→      setLessonsByModule(byModule)
   175→    } catch {
   176→      // ignore
   177→    } finally {
   178→      setLoading(false)
   179→    }
   180→  }
   181→
   182→  useEffect(() => {
   183→    loadModules()
   184→  }, [courseId])
   185→
   186→  const toggleModule = (id: string) => {
   187→    setExpanded((prev) => {
   188→      const next = new Set(prev)
   189→      if (next.has(id)) next.delete(id)
   190→      else next.add(id)
   191→      return next
   192→    })
   193→  }
   194→
   195→  const saveModule = async (data: any) => {
   196→    try {
   197→      const res = await fetch('/api/modules', {
   198→        method: moduleDialog.editing ? 'PUT' : 'POST',
   199→        headers: { 'Content-Type': 'application/json' },
   200→        body: JSON.stringify(moduleDialog.editing ? { id: moduleDialog.editing.id, ...data } : { course_id: courseId, ...data }),
   201→      })
   202→      const result = await res.json()
   203→      if (res.ok) {
   204→        toast.success(moduleDialog.editing ? 'تم تحديث الوحدة' : 'تمت إضافة الوحدة')
   205→        setModuleDialog({ open: false, editing: null })
   206→        loadModules()
   207→      } else if (result.needs_setup) {
   208→        setNeedsSetup(true)
   209→        setModuleDialog({ open: false, editing: null })
   210→        toast.error('يجب تنفيذ SQL أولاً')
   211→      } else {
   212→        toast.error(result.error || 'فشل الحفظ')
   213→      }
   214→    } catch {
   215→      toast.error('خطأ في الاتصال بالخادم')
   216→    }
   217→  }
   218→
   219→  const deleteModule = async (id: string) => {
   220→    if (!confirm('حذف الوحدة سيحذف جميع دروسها. متابعة؟')) return
   221→    await fetch('/api/modules', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id }) })
   222→    toast.success('تم الحذف')
   223→    loadModules()
   224→  }
   225→
   226→  const saveLesson = async (data: any, moduleId: string, editing: Lesson | null) => {
   227→    try {
   228→      const res = await fetch('/api/lessons', {
   229→        method: editing ? 'PUT' : 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify(editing ? { id: editing.id, ...data } : { module_id: moduleId, ...data }),
   232→      })
   233→      if (res.ok) {
   234→        toast.success(editing ? 'تم تحديث الدرس' : 'تمت إضافة الدرس')
   235→        setLessonDialog(null)
   236→        loadModules()
   237→      } else {
   238→        const result = await res.json()
   239→        toast.error(result.error || 'فشل الحفظ')
   240→      }
   241→    } catch {
   242→      toast.error('خطأ في الاتصال')
   243→    }
   244→  }
   245→
   246→  const deleteLesson = async (id: string) => {
   247→    if (!confirm('حذف الدرس؟')) return
   248→    await fetch('/api/lessons', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id }) })
   249→    toast.success('تم الحذف')
   250→    loadModules()
   251→  }
   252→
   253→  if (loading) {
   254→    return (
   255→      <div className="text-center py-12">
   256→        <Loader2 className="w-8 h-8 animate-spin mx-auto text-navy mb-3" />
   257→        <p className="text-sm text-muted-foreground">جاري تحميل محتوى الدورة...</p>
   258→      </div>
   259→    )
   260→  }
   261→
   262→  // ===== Database setup needed =====
   263→  if (needsSetup) {
   264→    return (
   265→      <div className="space-y-4">
   266→        <Card className="p-8 text-center bg-amber-50 border-2 border-amber-300">
   267→          <div className="w-16 h-16 mx-auto rounded-2xl bg-amber-500 flex items-center justify-center mb-4">
   268→            <Database className="w-8 h-8 text-white" />
   269→          </div>
   270→          <h3 className="font-bold text-navy text-lg mb-2">تهيئة قاعدة البيانات مطلوبة</h3>
   271→          <p className="text-sm text-muted-foreground mb-5 max-w-md mx-auto">
   272→            لتفعيل نظام الوحدات والدروس، يجب إنشاء جداول في Supabase. اضغط زر النسخ ثم الصق في Supabase SQL Editor واضغط Run.
   273→          </p>
   274→          <div className="flex gap-2 justify-center flex-wrap">
   275→            <Button onClick={copySql} className="bg-navy hover:bg-navy-light">
   276→              {setupSqlCopied ? <CheckCircle2 className="w-4 h-4 ml-1" /> : <Copy className="w-4 h-4 ml-1" />}
   277→              {setupSqlCopied ? 'تم النسخ!' : 'نسخ SQL'}
   278→            </Button>
   279→            <a href="https://supabase.com/dashboard/project/tzquudmfazysjogorrxz/sql/new" target="_blank" rel="noopener noreferrer">
   280→              <Button variant="outline" className="border-navy/30 text-navy hover:bg-navy/5">
   281→                <ExternalLink className="w-4 h-4 ml-1" /> فتح SQL Editor
   282→              </Button>
   283→            </a>
   284→            <Button variant="outline" onClick={loadModules} className="border-navy/30 text-navy hover:bg-navy/5">
   285→              <RefreshCw className="w-4 h-4 ml-1" /> إعادة المحاولة
   286→            </Button>
   287→          </div>
   288→          <details className="mt-5 text-right">
   289→            <summary className="text-xs text-muted-foreground cursor-pointer hover:text-navy">عرض SQL الكامل</summary>
   290→            <pre className="mt-3 text-xs bg-navy text-white p-4 rounded-lg overflow-x-auto max-h-72 overflow-y-auto font-mono" dir="ltr">{SETUP_SQL}</pre>
   291→          </details>
   292→        </Card>
   293→      </div>
   294→    )
   295→  }
   296→
   297→  return (
   298→    <div className="space-y-4">
   299→      <div className="flex items-center justify-between">
   300→        <div>
   301→          <h3 className="font-bold text-navy text-lg">محتوى الدورة</h3>
   302→          <p className="text-sm text-muted-foreground">{modules.length} وحدة، {Object.values(lessonsByModule).flat().length} درس</p>
   303→        </div>
   304→        <Button
   305→          className="bg-navy hover:bg-navy-light"
   306→          onClick={() => setModuleDialog({ open: true, editing: null })}
   307→        >
   308→          <Plus className="w-4 h-4 ml-1" /> وحدة جديدة
   309→        </Button>
   310→      </div>
   311→
   312→      {modules.length === 0 ? (
   313→        <Card className="p-12 text-center">
   314→          <FileText className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   315→          <p className="text-muted-foreground mb-4">لا توجد وحدات في هذه الدورة بعد</p>
   316→          <Button className="bg-navy hover:bg-navy-light" onClick={() => setModuleDialog({ open: true, editing: null })}>
   317→            <Plus className="w-4 h-4 ml-1" /> أضف أول وحدة
   318→          </Button>
   319→        </Card>
   320→      ) : (
   321→        modules.map((m, idx) => {
   322→          const isExpanded = expanded.has(m.id)
   323→          const lessons = lessonsByModule[m.id] || []
   324→          return (
   325→            <Card key={m.id} className="overflow-hidden">
   326→              {/* Module header */}
   327→              <div className="flex items-center justify-between p-4 bg-muted/30 border-b border-border">
   328→                <button onClick={() => toggleModule(m.id)} className="flex items-center gap-3 flex-1 text-right">
   329→                  <GripVertical className="w-5 h-5 text-muted-foreground" />
   330→                  <div className="w-8 h-8 rounded-lg bg-navy text-gold flex items-center justify-center text-sm font-bold flex-shrink-0">
   331→                    {idx + 1}
   332→                  </div>
   333→                  <div className="flex-1">
   334→                    <div className="font-bold text-navy">{m.title}</div>
   335→                    <div className="text-xs text-muted-foreground">{m.description || `${lessons.length} درس`}</div>
   336→                  </div>
   337→                  {isExpanded ? <ChevronUp className="w-5 h-5" /> : <ChevronDown className="w-5 h-5" />}
   338→                </button>
   339→                <div className="flex gap-1">
   340→                  <Button size="icon" variant="ghost" onClick={() => setModuleDialog({ open: true, editing: m })}>
   341→                    <Edit className="w-4 h-4 text-navy" />
   342→                  </Button>
   343→                  <Button size="icon" variant="ghost" onClick={() => deleteModule(m.id)}>
   344→                    <Trash2 className="w-4 h-4 text-red-500" />
   345→                  </Button>
   346→                </div>
   347→              </div>
   348→
   349→              {/* Module lessons */}
   350→              {isExpanded && (
   351→                <div className="p-4 space-y-2">
   352→                  {lessons.length === 0 ? (
   353→                    <div className="text-center py-6 text-muted-foreground text-sm">
   354→                      لا توجد دروس في هذه الوحدة
   355→                    </div>
   356→                  ) : (
   357→                    lessons.map((l, li) => {
   358→                      const type = LESSON_TYPES.find((t) => t.value === l.lesson_type) || LESSON_TYPES[0]
   359→                      const Icon = type.icon
   360→                      return (
   361→                        <div key={l.id} className="flex items-center justify-between p-3 bg-white border border-border rounded-lg hover:shadow-sm transition-shadow">
   362→                          <div className="flex items-center gap-3 flex-1">
   363→                            <GripVertical className="w-4 h-4 text-muted-foreground/50" />
   364→                            <Badge className={type.color}><Icon className="w-3 h-3 ml-1" /> {type.label}</Badge>
   365→                            <div className="flex-1">
   366→                              <div className="font-medium text-navy text-sm flex items-center gap-2">
   367→                                {l.title}
   368→                                {l.is_preview && <Badge variant="outline" className="text-xs">معاينة مجانية</Badge>}
   369→                              </div>
   370→                              {l.duration_minutes ? (
   371→                                <div className="text-xs text-muted-foreground">{l.duration_minutes} دقيقة</div>
   372→                              ) : null}
   373→                            </div>
   374→                          </div>
   375→                          <div className="flex gap-1">
   376→                            <Button size="icon" variant="ghost" onClick={() => setPreviewLesson(l)} title="معاينة الدرس">
   377→                              <Eye className="w-4 h-4 text-gold" />
   378→                            </Button>
   379→                            <Button size="icon" variant="ghost" onClick={() => setLessonDialog({ open: true, editing: l, moduleId: m.id })}>
   380→                              <Edit className="w-4 h-4 text-navy" />
   381→                            </Button>
   382→                            <Button size="icon" variant="ghost" onClick={() => deleteLesson(l.id)}>
   383→                              <Trash2 className="w-4 h-4 text-red-500" />
   384→                            </Button>
   385→                          </div>
   386→                        </div>
   387→                      )
   388→                    })
   389→                  )}
   390→                  <Button
   391→                    variant="outline"
   392→                    size="sm"
   393→                    className="w-full mt-2 border-dashed"
   394→                    onClick={() => setLessonDialog({ open: true, editing: null, moduleId: m.id })}
   395→                  >
   396→                    <Plus className="w-4 h-4 ml-1" /> إضافة درس
   397→                  </Button>
   398→                </div>
   399→              )}
   400→            </Card>
   401→          )
   402→        })
   403→      )}
   404→
   405→      {/* Module dialog */}
   406→      {moduleDialog.open && (
   407→        <ModuleDialog
   408→          open={moduleDialog.open}
   409→          onOpenChange={(o) => setModuleDialog({ open: o, editing: moduleDialog.editing })}
   410→          editing={moduleDialog.editing}
   411→          onSave={saveModule}
   412→        />
   413→      )}
   414→
   415→      {/* Lesson dialog */}
   416→      {lessonDialog?.open && (
   417→        <LessonDialog
   418→          open={lessonDialog.open}
   419→          onOpenChange={(o) => setLessonDialog(o ? { ...lessonDialog, open: o } : null)}
   420→          editing={lessonDialog.editing}
   421→          moduleId={lessonDialog.moduleId}
   422→          onSave={(data: any) => saveLesson(data, lessonDialog.moduleId, lessonDialog.editing)}
   423→        />
   424→      )}
   425→
   426→      {/* Lesson preview modal */}
   427→      <LessonPreviewModal
   428→        open={!!previewLesson}
   429→        onOpenChange={(o) => { if (!o) setPreviewLesson(null) }}
   430→        lesson={previewLesson}
   431→        modules={modules}
   432→        onNavigate={(l) => setPreviewLesson(l)}
   433→      />
   434→    </div>
   435→  )
   436→}

   438→// ============== Module Dialog ==============
   439→function ModuleDialog({ open, onOpenChange, editing, onSave }: any) {
   440→  const [title, setTitle] = useState(editing?.title || '')
   441→  const [description, setDescription] = useState(editing?.description || '')
   442→  const [position, setPosition] = useState(editing?.position || 0)
   443→
   444→  useEffect(() => {
   445→    setTitle(editing?.title || '')
   446→    setDescription(editing?.description || '')
   447→    setPosition(editing?.position || 0)
   448→  }, [editing])
   449→
   450→  return (
   451→    <Dialog open={open} onOpenChange={onOpenChange}>
   452→      <DialogContent className="sm:max-w-md">
   453→        <DialogHeader>
   454→          <DialogTitle>{editing ? 'تعديل الوحدة' : 'وحدة جديدة'}</DialogTitle>
   455→        </DialogHeader>
   456→        <div className="space-y-4 py-4">
   457→          <div>
   458→            <Label>عنوان الوحدة *</Label>
   459→            <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="مثال: الوحدة الأولى - الأساسيات" className="mt-1" />
   460→          </div>
   461→          <div>
   462→            <Label>وصف مختصر</Label>
   463→            <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={2} className="mt-1" />
   464→          </div>
   465→          <div>
   466→            <Label>الترتيب</Label>
   467→            <Input type="number" value={position} onChange={(e) => setPosition(parseInt(e.target.value) || 0)} dir="ltr" className="mt-1" />
   468→          </div>
   469→        </div>
   470→        <DialogFooter>
   471→          <Button variant="outline" onClick={() => onOpenChange(false)}>إلغاء</Button>
   472→          <Button
   473→            className="bg-navy hover:bg-navy-light"
   474→            onClick={() => { if (!title) { toast.error('أدخل العنوان'); return } onSave({ title, description, position }); }}
   475→          >
   476→            <Save className="w-4 h-4 ml-1" /> حفظ
   477→          </Button>

   526→  const [uploadingVideo, setUploadingVideo] = useState(false)

   529→    setUploadingVideo(true)

   551→        const fileId = tgData.result.video.file_id
   552→        // 3. Get the actual downloadable file URL from Telegram

   557→          setVideoUrl(directUrl)
   558→          toast.success('تم رفع الفيديو بنجاح')

   560→          // Fallback: use file_id directly as URL (won't play but stored for reference)
   561→          setVideoUrl(`tg://${fileId}`)
   562→          toast.success('تم رفع الفيديو إلى تيليجرام (الرابط المباشر غير متاح)')

   565→        const fileId = tgData.result.document.file_id

   567→        const fileData = await fileRes.json()
   568→        if (fileData.ok && fileData.result?.file_path) {
   569→          const directUrl = `https://api.telegram.org/file/bot${info.botToken}/${fileData.result.file_path}`

   583→    }
   584→  }

   746→// ============== Visual Exam Builder (no JSON editing) ==============
   747→function ExamBuilder({ value, onChange }: { value?: string | null; onChange: (v: string) => void }) {
   748→  const [questions, setQuestions] = useState<any[]>(() => {
   749→    if (!value) return []
   750→    try {
   751→      const parsed = JSON.parse(value)
   752→      return Array.isArray(parsed) ? parsed : []
   753→    } catch {
   754→      return []
   755→    }
   756→  })
   757→
   758→  const updateJson = (qs: any[]) => {
   759→    setQuestions(qs)
   760→    onChange(JSON.stringify(qs))
   761→  }
   762→
   763→  const addQuestion = () => {
   764→    updateJson([...questions, { q: '', options: ['', '', '', ''], answer: 0 }])
   765→  }
   766→
   767→  const updateQuestion = (i: number, field: string, val: any) => {
   768→    const qs = [...questions]
   769→    qs[i] = { ...qs[i], [field]: val }
   770→    updateJson(qs)
   771→  }
   772→
   773→  const removeQuestion = (i: number) => {
   774→    updateJson(questions.filter((_, idx) => idx !== i))
   775→  }
   776→
   777→  return (
   778→    <div className="space-y-3">
   779→      <div className="flex items-center justify-between">
   780→        <Label className="flex items-center gap-1">
   781→          <HelpCircle className="w-4 h-4 text-purple-600" /> أسئلة الاختبار
   782→        </Label>
   783→        <Button type="button" size="sm" variant="outline" onClick={addQuestion}>
   784→          <Plus className="w-3.5 h-3.5 ml-1" /> سؤال جديد
   785→        </Button>
   786→      </div>
   787→
   788→      {questions.length === 0 ? (
   789→        <div className="text-center py-8 bg-purple-500/5 border border-purple-500/30 rounded-lg">
   790→          <HelpCircle className="w-10 h-10 text-purple-600/30 mx-auto mb-2" />
   791→          <p className="text-sm text-muted-foreground">لا توجد أسئلة بعد</p>
   792→          <p className="text-xs text-muted-foreground mt-1">اضغط "سؤال جديد" لإضافة أول سؤال</p>
   793→        </div>
   794→      ) : (
   795→        <div className="space-y-3 max-h-96 overflow-y-auto fia-scroll pr-1">
   796→          {questions.map((q, qi) => (
   797→            <Card key={qi} className="p-3 border-purple-500/20">
   798→              <div className="flex items-start gap-2 mb-2">
   799→                <Badge variant="outline" className="text-xs bg-purple-500/5 border-purple-500/30 text-purple-600 flex-shrink-0">
   800→                  سؤال {qi + 1}
   801→                </Badge>
   802→                <Input
   803→                  value={q.q || ''}
   804→                  onChange={(e) => updateQuestion(qi, 'q', e.target.value)}
   805→                  placeholder="اكتب السؤال هنا..."
   806→                  className="flex-1 text-sm"
   807→                />
   808→                <Button type="button" size="icon" variant="ghost" onClick={() => removeQuestion(qi)} className="flex-shrink-0">
   809→                  <Trash2 className="w-4 h-4 text-red-500" />
   810→                </Button>
   811→              </div>
   812→
   813→              <div className="space-y-1.5">
   814→                <div className="text-xs text-muted-foreground">الإجابات (اضغط الدائرة لتحديد الإجابة الصحيحة):</div>
   815→                {(q.options || []).map((opt: string, oi: number) => (
   816→                  <div key={oi} className="flex items-center gap-2">
   817→                    <button
   818→                      type="button"
   819→                      onClick={() => updateQuestion(qi, 'answer', oi)}
   820→                      className={`w-5 h-5 rounded-full border-2 flex-shrink-0 flex items-center justify-center transition-colors ${
   821→                        q.answer === oi ? 'bg-green-500 border-green-500' : 'border-muted-foreground/40 hover:border-purple-400'
   822→                      }`}
   823→                      title="تحديد كإجابة صحيحة"
   824→                    >
   825→                      {q.answer === oi && <CheckCircle2 className="w-3 h-3 text-white" />}
   826→                    </button>
   827→                    <Input
   828→                      value={opt}
   829→                      onChange={(e) => {
   830→                        const newOptions = [...(q.options || [])]
   831→                        newOptions[oi] = e.target.value
   832→                        updateQuestion(qi, 'options', newOptions)
   833→                      }}
   834→                      placeholder={`الخيار ${oi + 1}`}
   835→                      className={`text-sm flex-1 ${q.answer === oi ? 'border-green-500/40 bg-green-500/5' : ''}`}
   836→                      dir="rtl"
   837→                    />
   838→                  </div>
   839→                ))}
   840→              </div>
   841→            </Card>
   842→          ))}
   843→        </div>
   844→      )}
   845→    </div>
