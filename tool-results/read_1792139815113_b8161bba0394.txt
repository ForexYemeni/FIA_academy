    39→export function StudentDashboard() {
    40→  const [active, setActive] = useState('overview')

    42→  const [viewingEnrollment, setViewingEnrollment] = useState<any | null>(null)

    61→  }, [user?.id])

    63→  const onEnrolled = () => {

    65→  }
    66→
    67→  return (
    68→    <DashboardShell

    70→      active={active}

   271→function BrowseCourses({ courses, enrollments, userId, onEnrolled }: any) {
   272→  const [paymentMethods, setPaymentMethods] = useState<any[]>([])
   273→  const [payDialog, setPayDialog] = useState<{ course: any; open: boolean } | null>(null)
   274→  const [enrolling, setEnrolling] = useState<string | null>(null)
   275→
   276→  useEffect(() => {
   277→    fetch('/api/payment-methods').then((r) => r.json()).then((d) => setPaymentMethods(Array.isArray(d) ? d.filter((p:any)=>p.enabled) : [])).catch(() => {})
   278→  }, [])
   279→
   280→  const isEnrolled = (courseId: string) => enrollments?.some((e: any) => e.course_id === courseId)
   281→
   282→  const enrollFree = async (course: any) => {
   283→    // Double-check client-side before calling API
   284→    if (isEnrolled(course.id)) {
   285→      toast.info('أنت مسجّل في هذه الدورة')
   286→      return
   287→    }
   288→    setEnrolling(course.id)
   289→    try {
   290→      const res = await fetch('/api/enrollments', {
   291→        method: 'POST',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({ user_id: userId, course_id: course.id, payment_status: 'free' }),
   294→      })
   295→      const data = await res.json()
   296→      if (data.already_enrolled) {
   297→        toast.info('أنت مسجّل في هذه الدورة')
   298→      } else if (res.ok) {
   299→        toast.success(`تم التسجيل في دورة: ${course.title}`)
   300→        onEnrolled()
   301→      } else toast.error('فشل التسجيل')
   302→    } catch {
   303→      toast.error('خطأ')
   304→    } finally {
   305→      setEnrolling(null)
   306→    }
   307→  }
   308→
   309→  const enrollPaid = async (course: any, method: any, reference: string, proofUrl?: string | null) => {
   310→    // Double-check client-side before calling API
   311→    if (isEnrolled(course.id)) {
   312→      toast.info('أنت مسجّل في هذه الدورة')
   313→      return
   314→    }
   315→    setEnrolling(course.id)
   316→    try {
   317→      // Combine reference with proof URL
   318→      const fullReference = proofUrl ? `${reference} | PROOF: ${proofUrl}` : reference
   319→
   320→      // 1. Create enrollment with pending payment
   321→      const res = await fetch('/api/enrollments', {
   322→        method: 'POST',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify({
   325→          user_id: userId,
   326→          course_id: course.id,
   327→          payment_status: 'pending',
   328→          amount_paid: course.price,
   329→          payment_method: method.name,
   330→          payment_reference: fullReference,
   331→        }),
   332→      })
   333→      const data = await res.json()
   334→      if (data.already_enrolled) {
   335→        toast.info('أنت مسجّل في هذه الدورة')
   336→      } else if (res.ok) {
   337→        // 2. Create payment record with proof URL
   338→        await fetch('/api/payments', {
   339→          method: 'POST',
   340→          headers: { 'Content-Type': 'application/json' },
   341→          body: JSON.stringify({
   342→            amount: course.price,
   343→            currency: course.currency || 'ر.ي',
   344→            method: method.name,
   345→            status: 'قيد الانتظار',
   346→            reference: fullReference,
   347→            user_id: userId,
   348→            course_id: course.id,
   349→          }),
   350→        })
   351→        toast.success('تم تسجيل طلبك! سيتم تفعيل الدورة بعد تأكيد الدفع')
   352→        onEnrolled()
   353→        setPayDialog(null)
   354→      } else toast.error('فشل التسجيل')
   355→    } catch {
   356→      toast.error('خطأ')
   357→    } finally {
   358→      setEnrolling(null)
   359→    }
   360→  }
   361→
   362→  return (
   363→    <SectionCard title="استكشاف الدورات" description="تصفّح جميع الدورات المتاحة">
   364→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   365→        {courses.map((c: any) => {
   366→          const enrolled = isEnrolled(c.id)
   367→          const isFree = c.price_type === 'مجانية' || c.price === 0
   368→          return (
   369→            <Card key={c.id} className="overflow-hidden card-hover">
   370→              <div className="h-32 bg-navy relative">
   371→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   372→                <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   373→                {isFree ? (
   374→                  <Badge className="absolute top-2 left-2 bg-green-500 text-white">مجانية</Badge>
   375→                ) : (
   376→                  <Badge className="absolute top-2 left-2 bg-gold text-navy">مدفوعة</Badge>
   377→                )}
   378→              </div>
   379→              <div className="p-4">
   380→                <h3 className="font-bold text-navy text-sm mb-1 line-clamp-2 min-h-[2.5rem]">{c.title}</h3>
   381→                <p className="text-xs text-muted-foreground mb-3 line-clamp-2">{c.short_description}</p>
   382→                <div className="flex items-center justify-between text-xs text-muted-foreground mb-3">
   383→                  <span className="flex items-center gap-1">
   384→                    <Clock className="w-3 h-3" /> {c.duration_hours || 0} ساعة
   385→                  </span>
   386→                  <span className="font-bold text-navy">
   387→                    {isFree ? <span className="text-green-600">مجانية</span> : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}
   388→                  </span>
   389→                </div>
   390→                {enrolled ? (
   391→                  <Button size="sm" variant="outline" className="w-full border-green-500/30 text-green-600 bg-green-500/5">
   392→                    <CheckCircle2 className="w-4 h-4 ml-1" /> مسجّل
   393→                  </Button>
   394→                ) : isFree ? (
   395→                  <Button
   396→                    size="sm"
   397→                    className="w-full bg-green-600 hover:bg-green-700"
   398→                    onClick={() => enrollFree(c)}
   399→                    disabled={enrolling === c.id}
   400→                  >
   401→                    {enrolling === c.id ? 'جاري التسجيل...' : 'سجّل الآن'}
   402→                  </Button>
   403→                ) : (
   404→                  <Button
   405→                    size="sm"
   406→                    className="w-full bg-gold hover:bg-gold-light text-navy"
   407→                    onClick={() => setPayDialog({ course: c, open: true })}
   408→                  >
   409→                    <Lock className="w-3.5 h-3.5 ml-1" /> اشترك الآن
   410→                  </Button>
   411→                )}
   412→              </div>
   413→            </Card>
   414→          )
   415→        })}
   416→      </div>
   417→
   418→      {payDialog?.open && (
   419→        <PaymentDialog
   420→          course={payDialog.course}
   421→          methods={paymentMethods}
   422→          open={payDialog.open}
   423→          onOpenChange={(o) => setPayDialog({ ...payDialog, open: o })}
   424→          onPay={(method: any, reference: string, proofUrl?: string | null) => enrollPaid(payDialog.course, method, reference, proofUrl)}
   425→          loading={enrolling === payDialog.course.id}
   426→        />
   427→      )}
   428→    </SectionCard>
   429→  )
   430→}

   432→function PaymentDialog({ course, methods, open, onOpenChange, onPay, loading }: any) {
   433→  const [selectedMethod, setSelectedMethod] = useState<any>(methods[0] || null)
   434→  const [reference, setReference] = useState('')
   435→  const [proofUrl, setProofUrl] = useState<string | null>(null)
   436→  const [uploadingProof, setUploadingProof] = useState(false)
   437→
   438→  const handleProofUpload = async (file: File) => {
   439→    setUploadingProof(true)
   440→    try {
   441→      // Compress image on client side
   442→      const reader = new FileReader()
   443→      reader.onload = (e) => {
   444→        const img = new Image()
