   185→   185→function TrainerOverview({ user, courses, students }: any) {
   186→   186→  // Compute the real average rating from courses that actually carry a
   187→   187→  // numeric rating. If no ratings exist yet, fall back to a clear message
   188→   188→  // instead of a hardcoded fake number.
   189→   189→  const ratedCourses = (courses || []).filter(
   190→   190→    (c: any) => typeof c.rating === 'number' && c.rating > 0
   191→   191→  )
   192→   192→  const avgRating =
   193→   193→    ratedCourses.length > 0
   194→   194→      ? (ratedCourses.reduce((acc, c) => acc + c.rating, 0) / ratedCourses.length).toFixed(1)
   195→   195→      : 'لا توجد تقييمات بعد'
   196→   196→
   197→   197→  // Real "active exams" count: lessons with lesson_type === 'exam' across
   198→   198→  // the trainer's own courses (same data source as TrainerExams).
   199→   199→  const [examsCount, setExamsCount] = useState<number>(0)
   200→   200→  const [examsLoading, setExamsLoading] = useState(false)
   201→   201→
   202→   202→  useEffect(() => {
   203→   203→    let cancelled = false
   204→   204→    const load = async () => {
   205→   205→      if (!courses || courses.length === 0) {
   206→   206→        setExamsCount(0)
   207→   207→        return
   208→   208→      }
   209→   209→      setExamsLoading(true)
   210→   210→      try {
   211→   211→        const results = await Promise.all(
   212→   212→          courses.map(async (c: any) => {
   213→   213→            try {
   214→   214→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   215→   215→              const data = await res.json()
   216→   216→              const lessons = Array.isArray(data) ? data : []
   217→   217→              return lessons.filter((l: any) => l.lesson_type === 'exam').length
   218→   218→            } catch {
   219→   219→              return 0
   220→   220→            }
   221→   221→          })
   222→   222→        )
   223→   223→        if (cancelled) return
   224→   224→        setExamsCount(results.reduce((a, b) => a + b, 0))
   225→   225→      } catch {
   226→   226→        if (!cancelled) setExamsCount(0)
   227→   227→      } finally {
   228→   228→        if (!cancelled) setExamsLoading(false)
   229→   229→      }
   230→   230→    }
   231→   231→    load()
   232→   232→    return () => { cancelled = true }
   233→   233→  }, [courses])
   234→   234→
   235→   235→  return (
   236→   236→    <div className="space-y-6">
   237→   237→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   238→   238→        <h2 className="text-2xl font-bold mb-1">مرحباً، {user?.name} 👋</h2>
   239→   239→        <p className="text-white/80">نظرة عامة على نشاطك التدريبي</p>
   240→   240→      </div>
   241→   241→
   242→   242→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   243→   243→        <StatCard icon={BookOpen} label="الدورات التدريبية" value={courses.length} color="navy" />
   244→   244→        <StatCard icon={Users} label="إجمالي المتدربين" value={String(students.length)} color="gold" />
   245→   245→        <StatCard icon={Star} label="متوسط التقييم" value={avgRating} color="green" />
   246→   246→        <StatCard
   247→   247→          icon={FileText}
   248→   248→          label="الاختبارات النشطة"
   249→   249→          value={examsLoading ? '—' : examsCount}
   250→   250→          color="blue"
   251→   251→        />
   252→   252→      </div>
   253→   253→
   254→   254→      <SectionCard title="أحدث الدورات" description="الدورات التي تدرّسها">
   255→   255→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   256→   256→          {courses.slice(0, 3).map((c: any) => (
   257→   257→            <Card key={c.id} className="overflow-hidden card-hover">
   258→   258→              <div className="h-32 bg-navy">
   259→   259→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   260→   260→              </div>
   261→   261→              <div className="p-4">
   262→   262→                <h3 className="font-bold text-navy text-sm mb-1 line-clamp-1">{c.title}</h3>
   263→   263→                <div className="text-xs text-muted-foreground">{c.enrolled || 0} متدرب • {c.rating || 0} ⭐</div>
   264→   264→              </div>
   265→   265→            </Card>
   266→   266→          ))}
   267→   267→        </div>
   268→   268→      </SectionCard>
   269→   269→    </div>
   270→   270→  )
   271→   271→}

   273→   273→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   274→   274→  const [deleting, setDeleting] = useState<string | null>(null)
   275→   275→  // Course currently awaiting delete confirmation. Null means the dialog is closed.
   276→   276→  const [deleteTarget, setDeleteTarget] = useState<any | null>(null)
   277→   277→
   278→   278→  const handleDelete = async () => {
   279→   279→    if (!deleteTarget) return
   280→   280→    const id = deleteTarget.id
   281→   281→    setDeleting(id)
   282→   282→    try {
   283→   283→      const res = await fetch('/api/courses', {
   284→   284→        method: 'DELETE',
   285→   285→        headers: { 'Content-Type': 'application/json' },
   286→   286→        body: JSON.stringify({ id }),
   287→   287→      })
   288→   288→      if (res.ok) {
   289→   289→        toast.success('تم حذف الدورة')
   290→   290→        setDeleteTarget(null)
   291→   291→        onRefresh()
   292→   292→      } else {
   293→   293→        const err = await res.json().catch(() => ({}))
   294→   294→        toast.error(err.error || 'فشل الحذف')
   295→   295→      }
   296→   296→    } catch {
   297→   297→      toast.error('خطأ في الاتصال')
   298→   298→    } finally {
   299→   299→      setDeleting(null)
   300→   300→    }
   301→   301→  }
   302→   302→
   303→   303→  return (
   304→   304→    <SectionCard
   305→   305→      title="دوراتي التدريبية"
   306→   306→      description="إدارة الدورات التي تدرّسها"
   307→   307→      action={
   308→   308→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   309→   309→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   310→   310→        </Button>
   311→   311→      }
   312→   312→    >
   313→   313→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   314→   314→        {courses.map((c: any) => (
   315→   315→          <Card key={c.id} className="overflow-hidden card-hover">
   316→   316→            <div className="h-32 bg-navy relative">
   317→   317→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   318→   318→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   319→   319→            </div>
   320→   320→            <div className="p-4">
   321→   321→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   322→   322→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   323→   323→                <span>{c.enrolled || 0} متدرب</span>
   324→   324→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   325→   325→              </div>
   326→   326→              <div className="space-y-2">
   327→   327→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   328→   328→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   329→   329→                </Button>
   330→   330→                <div className="flex gap-2">
   331→   331→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   332→   332→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   333→   333→                  </Button>
   334→   334→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => setDeleteTarget(c)} disabled={deleting === c.id}>
   335→   335→                    <Trash2 className="w-3.5 h-3.5" />
   336→   336→                  </Button>
   337→   337→                </div>
   338→   338→              </div>
   339→   339→            </div>
   340→   340→          </Card>
   341→   341→        ))}
   342→   342→      </div>
   343→   343→
   344→   344→      {/* Professional delete-confirmation card */}
   345→   345→      <Dialog
   346→   346→        open={!!deleteTarget}
   347→   347→        onOpenChange={(open) => { if (!open && !deleting) setDeleteTarget(null) }}
   348→   348→      >
   349→   349→        <DialogContent className="sm:max-w-md">
   350→   350→          <DialogHeader>
   351→   351→            <DialogTitle className="flex items-center gap-2 text-red-600">
   352→   352→              <AlertTriangle className="w-5 h-5" />
   353→   353→              تأكيد الحذف
   354→   354→            </DialogTitle>
   355→   355→          </DialogHeader>
   356→   356→          <div className="py-2 space-y-3">
   357→   357→            <div className="flex items-start gap-3 p-3 bg-red-500/5 border border-red-500/20 rounded-lg">
   358→   358→              <Trash2 className="w-5 h-5 text-red-500 flex-shrink-0 mt-0.5" />
   359→   359→              <div className="min-w-0">
   360→   360→                <div className="text-xs text-muted-foreground mb-1">سيتم حذف الدورة التالية:</div>
   361→   361→                <div className="font-bold text-navy line-clamp-2 break-words">{deleteTarget?.title}</div>
   362→   362→              </div>
   363→   363→            </div>
   364→   364→            <p className="text-sm text-muted-foreground text-center">
   365→   365→              هل أنت متأكد من الحذف؟ لا يمكن التراجع عن هذا الإجراء.
   366→   366→            </p>
   367→   367→          </div>
   368→   368→          <DialogFooter>
   369→   369→            <Button
   370→   370→              variant="outline"
   371→   371→              onClick={() => setDeleteTarget(null)}
   372→   372→              disabled={!!deleting}
   373→   373→            >
   374→   374→              إلغاء
   375→   375→            </Button>
   376→   376→            <Button
   377→   377→              onClick={handleDelete}
   378→   378→              disabled={!!deleting}
   379→   379→              className="bg-red-600 hover:bg-red-700 text-white"
   380→   380→            >
   381→   381→              {deleting
   382→   382→                ? <><Loader2 className="w-4 h-4 ml-1 animate-spin" /> جاري الحذف...</>
   383→   383→                : <><Trash2 className="w-4 h-4 ml-1" /> حذف الدورة</>}
   384→   384→            </Button>
   385→   385→          </DialogFooter>
   386→   386→        </DialogContent>
   387→   387→      </Dialog>
   388→   388→    </SectionCard>
   389→   389→  )
   390→   390→}

   750→   750→function TrainerExams({ courses }: { courses: any[] }) {

   758→   758→        setExams([])

   790→   790→                  }

   806→   806→    loadExams()
   807→   807→    return () => { cancelled = true }

   809→   809→
   810→   810→  const totalQuestions = exams.reduce((acc, e) => acc + (e.questions_count || 0), 0)
   811→   811→  const totalCoursesWithExams = new Set(exams.map((e) => e.course_id)).size
   812→   812→
   813→   813→  return (
