    51→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    52→// Show only the human-readable text in the UI, but preserve the extra data on save.
    53→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    54→const splitBio = (raw: string) => {
    55→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    56→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    57→}
    58→const joinBio = (display: string, original?: string) => {
    59→  if (!original) return display
    60→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    61→  if (i === -1) return display
    62→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    63→}

    65→// Returns a normalized status info object for an enrollment's payment_status.
    66→// payment_status can be one of: 'pending' | 'confirmed' | 'completed' | 'free'.
    67→// - 'pending'    -> student paid, waiting for admin approval (yellow)
    68→// - 'confirmed'  -> admin approved the payment, course in progress (green)
    69→// - 'completed'  -> payment confirmed AND course finished (green)
    70→// - 'free'       -> free enrollment (blue)
    71→export function getPaymentStatusInfo(paymentStatus: string | undefined | null): {
    72→  label: string
    73→  badgeClass: string
    74→} {
    75→  switch (paymentStatus) {
    76→    case 'completed':
    77→      return { label: 'مكتمل', badgeClass: 'bg-green-500 text-white' }
    78→    case 'confirmed':
    79→      return { label: 'تم تأكيد الدفع', badgeClass: 'bg-green-500 text-white' }
    80→    case 'free':
    81→      return { label: 'مجانية', badgeClass: 'bg-blue-500 text-white' }
    82→    case 'pending':
    83→    default:
    84→      return { label: 'بانتظار تأكيد الدفع', badgeClass: 'bg-yellow-500 text-white' }
    85→  }
    86→}
    87→
    88→// Returns a status info object for a raw payment record's `status` field
    89→// (used in the Payments section). Raw status can be: 'مكتمل' | 'completed' |
    90→// 'قيد الانتظار' | 'pending' | 'بانتظار' | 'ملغي' | 'failed' | 'فشل' | etc.
    91→export function getPaymentRecordStatusInfo(status: string | undefined | null): {
    92→  label: string
    93→  badgeClass: string
    94→} {
    95→  const s = (status || '').trim()
    96→  const norm = s.toLowerCase()
    97→  if (s === 'مكتمل' || norm === 'completed' || norm === 'paid' || norm === 'confirmed') {
    98→    return { label: s || 'مكتمل', badgeClass: 'bg-green-500/15 text-green-600 border-green-500/30' }
    99→  }
   100→  if (s === 'ملغي' || norm === 'failed' || s === 'فشل' || norm === 'cancelled' || norm === 'canceled') {
   101→    return { label: s || 'فشل', badgeClass: 'bg-red-500/15 text-red-600 border-red-500/30' }
   102→  }
   103→  if (s === 'قيد الانتظار' || s === 'بانتظار' || norm === 'pending' || s === '') {
   104→    return { label: s || 'بانتظار', badgeClass: 'bg-yellow-500/15 text-yellow-600 border-yellow-500/30' }
   105→  }
   106→  // Unknown status — show as neutral
   107→  return { label: s, badgeClass: 'bg-muted text-muted-foreground border-border' }
   108→}

   110→export function StudentDashboard() {
   111→  const [active, setActive] = useState('overview')
   112→  useHashTab(NAV.map((n) => n.id), setActive)
   113→  const [viewingCourse, setViewingCourse] = useState<any | null>(null)
   114→  const [viewingEnrollment, setViewingEnrollment] = useState<any | null>(null)
   115→  const user = useAuth((s) => s.user)
   116→  const updateUser = useAuth((s) => s.updateUser)
   117→  const [courses, setCourses] = useState<any[]>([])
   118→  const [certificates, setCertificates] = useState<any[]>([])
   119→  const [payments, setPayments] = useState<any[]>([])
   120→  const [enrollments, setEnrollments] = useState<any[]>([])
   121→  const [profile, setProfile] = useState({ name: user?.name || '', phone: user?.phone || '', title: user?.title || '', bio: splitBio(user?.bio || '') })
   122→
   123→  const loadEnrollments = () => {
   124→    if (!user?.id) return
   125→    fetch(`/api/enrollments?user_id=${user.id}`).then((r) => r.json()).then((d) => setEnrollments(Array.isArray(d) ? d : [])).catch(() => {})
   126→  }
   127→
   128→  useEffect(() => {
   129→    fetch('/api/courses').then((r) => r.json()).then((d) => setCourses(Array.isArray(d) ? d : [])).catch(() => {})
   130→    fetch('/api/certificates').then((r) => r.json()).then((d) => setCertificates(Array.isArray(d) ? d : [])).catch(() => {})
   131→    fetch('/api/payments').then((r) => r.json()).then((d) => setPayments(Array.isArray(d) ? d : [])).catch(() => {})
   132→    loadEnrollments()
   133→  }, [user?.id])
   134→
   135→  // Real-time: when admin updates a course (e.g. changes price_type
   136→  // from مدفوعة to مجانية or vice versa), refresh the courses list
   137→  // immediately so the student sees the updated price/status.
   138→  useEffect(() => {
   139→    let channel: any
   140→    try {
   141→      channel = supabasePublic
   142→        .channel('courses-realtime')
   143→        .on('postgres_changes',
   144→          { event: 'UPDATE', schema: 'public', table: 'courses' },
   145→          () => {
   146→            fetch('/api/courses').then((r) => r.json()).then((d) => setCourses(Array.isArray(d) ? d : [])).catch(() => {})
   147→          }
   148→        )
   149→        .subscribe()
   150→    } catch {}
   151→    return () => { try { supabasePublic.removeChannel(channel) } catch {} }
   152→  }, [])
   153→
   154→  const onEnrolled = () => {
   155→    loadEnrollments()
   156→  }
   157→
   158→  return (
   159→    <DashboardShell
   160→      nav={NAV}
   161→      active={active}
   162→      onNavigate={(id) => { setActive(id); if (id !== 'course-view') setViewingCourse(null) }}
   163→      roleLabel="متدرب"
   164→      roleColor="bg-blue-500/15 text-blue-600 border-blue-500/30"
   165→    >
   166→      {active === 'course-view' && viewingCourse ? (
   167→        <StudentCourseViewer
   168→          course={viewingCourse}
   169→          enrollment={viewingEnrollment}
   170→          onBack={() => { setViewingCourse(null); setActive('courses'); loadEnrollments() }}
   171→        />
   172→      ) : null}
   173→      {active !== 'course-view' && (
   174→        <>
   175→          {active === 'overview' && <Overview user={user} courses={courses} certificates={certificates} enrollments={enrollments} onViewCourse={(c) => {

   178→            const isFree = c.price_type === 'مجانية' || c.price === 0

   180→              // Not enrolled at all — route to BrowseCourses

   198→            }

   222→function Overview({ user, courses, certificates, enrollments, onViewCourse }: any) {
   223→  // Helper: determine if a course can be opened by the student.
   224→  // Returns { canOpen, reason, enrollment } so the UI can decide what
   225→  // to do when the user clicks "عرض الدورة".
   226→  // - Free + not enrolled  -> should auto-enroll then open
   227→  // - Paid + not enrolled  -> must open payment dialog
   228→  // - Paid + enrolled pending -> cannot open (waiting admin approval)
   229→  // - Enrolled (free/confirmed/completed) -> can open
   230→  const getCourseAccess = (c: any) => {
   231→    const enr = enrollments?.find((e: any) => e.course_id === c.id)
   232→    const isFree = c.price_type === 'مجانية' || c.price === 0
   233→    if (enr) {
   234→      if (enr.payment_status === 'pending') {
   235→        return { canOpen: false, reason: 'بانتظار تأكيد الدفع', enrollment: enr }
   236→      }
   237→      // free | confirmed | completed -> can open
   238→      return { canOpen: true, reason: '', enrollment: enr }
   239→    }
   240→    if (isFree) {
   241→      return { canOpen: false, reason: 'free-not-enrolled', enrollment: null }
   242→    }
   243→    return { canOpen: false, reason: 'paid-not-enrolled', enrollment: null }
   244→  }
   245→
   246→  return (
   247→    <div className="space-y-6">
   248→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   249→        <h2 className="text-2xl font-bold mb-1">أهلاً، {user?.name} 👋</h2>
   250→        <p className="text-white/80">إليك ملخّص نشاطك في أكاديمية المستقبل الدولية</p>
   251→      </div>
   252→
   253→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   254→        <StatCard icon={BookOpen} label="الدورات المسجّلة" value={enrollments?.length || 0} color="navy" />
   255→        <StatCard icon={CheckCircle2} label="دورات مكتملة" value={enrollments?.filter((e:any)=>e.status==='completed').length || 0} color="green" />
   256→        <StatCard icon={Clock} label="ساعات التعلّم" value={24} color="gold" />
   257→        <StatCard icon={Award} label="الشهادات المكتسبة" value={certificates.length} color="blue" />
   258→      </div>
   259→
   260→      <SectionCard title="تقدّم الدورات" description="متابعة آخر التحديثات في دوراتك">
   261→        <div className="space-y-4">
   262→          {enrollments && enrollments.length > 0 ? (
   263→            enrollments.slice(0, 3).map((e: any) => {
   264→              const c = courses.find((x:any) => x.id === e.course_id)
   265→              if (!c) return null
   266→              return (
   267→                <div key={e.id} className="space-y-2">
   268→                  <div className="flex items-center justify-between">
   269→                    <div className="font-medium text-navy text-sm">{c.title}</div>
   270→                    <span className="text-xs text-muted-foreground">{e.progress || 0}%</span>
   271→                  </div>
   272→                  <Progress value={e.progress || 0} className="h-2" />
   273→                </div>
   274→              )
   275→            })
   276→          ) : (
   277→            <div className="text-center py-6 text-muted-foreground text-sm">لا توجد دورات مسجّلة بعد — تصفّح الدورات المتاحة</div>
   278→          )}
   279→        </div>
   280→      </SectionCard>
   281→
   282→      <SectionCard title="الدورات الموصى بها" description="استكشف المزيد من الدورات — اضغط للتسجيل أو فتح الدورة">
   283→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   284→          {courses.slice(0, 3).map((c: any) => {
   285→            const access = getCourseAccess(c)
   286→            const isFree = c.price_type === 'مجانية' || c.price === 0
   287→            // Decide button label + action based on access state
   288→            let btnLabel = 'عرض الدورة'
   289→            let btnAction: () => void = () => onViewCourse?.(c)
   290→            let btnClass = 'w-full bg-navy hover:bg-navy-light'
   291→            if (access.enrollment?.payment_status === 'pending') {
   292→              btnLabel = 'بانتظار تأكيد الدفع'
   293→              btnClass = 'w-full bg-yellow-500/20 text-yellow-700 border border-yellow-500/30 cursor-not-allowed'
   294→              btnAction = () => toast.info('سيتم تفعيل الدورة بعد تأكيد الإدارة للدفع')
   295→            } else if (access.reason === 'free-not-enrolled') {
   296→              btnLabel = 'سجّل الآن — مجانية'
   297→              btnClass = 'w-full bg-green-600 hover:bg-green-700 text-white'
   298→              // Will be handled by parent (BrowseCourses logic) — for now, just route to "browse"
   299→              btnAction = () => { toast.info('افتح قسم "استكشاف الدورات" للتسجيل') }
   300→            } else if (access.reason === 'paid-not-enrolled') {
   301→              btnLabel = 'اشترك الآن'
   302→              btnClass = 'w-full bg-gold hover:bg-gold-light text-navy'
   303→              btnAction = () => { toast.info('افتح قسم "استكشاف الدورات" لإتمام الدفع') }
   304→            } else if (access.canOpen) {
   305→              btnLabel = 'عرض الدورة'
   306→              btnClass = 'w-full bg-navy hover:bg-navy-light'
   307→              btnAction = () => onViewCourse?.(c)
   308→            }
   309→            return (
   310→              <Card key={c.id} className="p-4 card-hover">
   311→                <div className="h-24 bg-navy rounded-lg mb-3 overflow-hidden relative">
   312→                  {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   313→                  <Badge className={`absolute top-2 left-2 ${isFree ? 'bg-green-500 text-white' : 'bg-gold text-navy'}`}>
   314→                    {isFree ? 'مجانية' : `${c.price?.toLocaleString()} ${c.currency || 'ر.ي'}`}
   315→                  </Badge>
   316→                </div>
   317→                <div className="font-bold text-navy text-sm mb-1 line-clamp-1">{c.title}</div>
   318→                <div className="text-xs text-muted-foreground mb-2">{c.duration_hours} ساعة</div>
   319→                <Button size="sm" className={btnClass} onClick={btnAction} disabled={access.enrollment?.payment_status === 'pending'}>
   320→                  {btnLabel}
   321→                </Button>
   322→              </Card>
   323→            )
   324→          })}
   325→        </div>
   326→      </SectionCard>
   327→    </div>
   328→  )
   329→}

   331→function MyCourses({ courses, enrollments, onContinue, studentName }: any) {
   332→  const [requesting, setRequesting] = useState<string | null>(null)
   333→
   334→  const requestCertificate = async (course: any) => {
   335→    setRequesting(course.id)
   336→    try {
   337→      const res = await fetch('/api/certificate-requests', {
   338→        method: 'POST',
   339→        headers: { 'Content-Type': 'application/json' },
   340→        body: JSON.stringify({
   341→          user_id: course.enrollment.user_id,
   342→          course_id: course.id,
   343→          student_name: studentName || course.enrollment.user_id || 'متدرب',
   344→          course_title: course.title,
   345→          fee: 0,
   346→        }),
   347→      })
   348→      const data = await res.json()
   349→      if (res.ok) {
   350→        toast.success('تم إرسال طلب الشهادة! سيتم مراجعته من قبل المدرب والإدارة')
   351→      } else {
   352→        toast.error(data.error || 'فشل الطلب')
   353→      }
   354→    } catch {
   355→      toast.error('خطأ في الاتصال')
   356→    } finally {
   357→      setRequesting(null)
   358→    }
   359→  }
   360→
   361→  const enrolled = (enrollments || [])
   362→    .map((e: any) => {
   363→      const course = courses.find((c: any) => c.id === e.course_id)
   364→      return course ? { ...course, enrollment: e } : null
   365→    })
   366→    .filter(Boolean)
   367→
   368→  return (
   369→    <SectionCard title="دوراتي المسجّلة" description={`${enrolled.length} دورة`}>
   370→      {enrolled.length === 0 ? (
   371→        <div className="text-center py-12">
   372→          <BookOpen className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   373→          <p className="text-muted-foreground mb-4">لم تسجّل في أي دورة بعد</p>
   374→          <p className="text-xs text-muted-foreground mb-4">تصفّح الدورات المتاحة في قسم "استكشاف الدورات"</p>
   375→        </div>
   376→      ) : (
   377→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   378→          {enrolled.map((c: any) => (
   379→            <Card key={c.id} className="overflow-hidden card-hover">
   380→              <div className="h-32 bg-navy relative">
   381→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   382→                <div className="absolute top-2 right-2">
   383→                  <Badge className={getPaymentStatusInfo(c.enrollment.payment_status).badgeClass}>
   384→                    {getPaymentStatusInfo(c.enrollment.payment_status).label}
   385→                  </Badge>
   386→                </div>
   387→              </div>
   388→              <div className="p-4">
   389→                <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   390→                <Progress value={c.enrollment.progress || 0} className="h-2 mb-2" />
   391→                <div className="flex justify-between text-xs text-muted-foreground mb-3">
   392→                  <span>{c.enrollment.progress || 0}% مكتمل</span>
   393→                  <span>{c.duration_hours} ساعة</span>
   394→                </div>
   395→                {c.enrollment.payment_status === 'pending' ? (
   396→                  <Button size="sm" variant="outline" disabled className="w-full border-yellow-500/30 text-yellow-600">
   397→                    بانتظار تأكيد الدفع
   398→                  </Button>

   442→function BrowseCourses({ courses, enrollments, userId, onEnrolled, onViewCourse }: any) {

   451→  const isEnrolled = (courseId: string) => enrollments?.some((e: any) => e.course_id === courseId)

   456→  // confirms the payment (payment_status === 'confirmed' or 'completed').
   457→  // - Not enrolled + free  → can enroll then open (caller handles)

   465→      return { state: isFree ? 'free-not-enrolled' : 'paid-not-enrolled', enrollment: null }

   468→      return { state: 'pending', enrollment: enr }

   484→        method: 'POST',

   498→        // (the user expects: click free course → it opens)
   499→        onViewCourse?.(course)

   501→    } catch {
   502→      toast.error('خطأ')

   504→      setEnrolling(null)

   508→  // Clicking the entire card triggers the appropriate action:
   509→  // - If enrolled + payment confirmed → open the course viewer immediately
   510→  // - If enrolled + payment PENDING   → show toast (waiting admin approval)
   511→  // - If free + not enrolled          → auto-enroll then open the course viewer
   512→  // - If paid + not enrolled          → open the payment dialog (course opens AFTER admin approval)

   521→      }

   523→      onViewCourse?.(c)

   525→      enrollFree(c)

   527→      setPayDialog({ course: c, open: true })

   531→  const enrollPaid = async (course: any, method: any, reference: string, proofUrl?: string | null, couponCode?: string) => {

   693→          </div>

   697→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">

   876→

   878→  // === PRICING LOGIC FOR PAYMENT ===
   879→  // - `course.price` = the regular course price (shown when no offer)
   880→  // - `course.original_price` = the "was" price (struck through)
   881→  // - `course.offer_price` = the special promotional offer price (if
   882→  //   the course is currently in an offer period — limited_time or
   883→  //   featured). When set and > 0, this is what the student pays.
   884→  // - `course.auto_coupon_code` = coupon that's auto-applied at
   885→  //   checkout (if the student hasn't already used it on another
   886→  //   course)
   887→  // The student is charged `finalPrice` below, which is the offer
   888→  // price (or regular price if no offer) minus any coupon discount.
   889→  const regularPrice = Number(course.price) || 0
   890→  const offerPrice = Number(course.offer_price) || 0
   891→  const officialPrice = Number(course.original_price) || 0
   892→  // The "base price" is what the student pays BEFORE any coupon.
   893→  // If there's an offer_price, use it; otherwise fall back to the
   894→  // regular course.price.
   895→  const basePrice = offerPrice > 0 ? offerPrice : regularPrice
   896→  // The struck-through "was" price — pick the larger of official_price
   897→  // and regular_price so the student sees the biggest discount.

   900→  // For backward compat, alias to old variable names used elsewhere.
   901→  const originalPrice = basePrice
   902→
   903→  const discount = appliedCoupon
   904→    ? appliedCoupon.discount_type === 'percentage'

   935→          // don't apply it, just clear the code silently.
   936→          setCouponCode('')
   937→        } else {
   938→          // Invalid coupon — don't show an error (the user didn't
   939→          // type it manually). Just clear the code.
   940→          setCouponCode('')
   941→        }
   942→      } catch {
   943→        setCouponCode('')
   944→      } finally {
   945→        setValidatingCoupon(false)
   946→      }
   947→    })()
   948→  }, [open, course?.auto_coupon_code, userId])

  1176→function Exams({ userId, courses }: any) {

  1178→  const [loading, setLoading] = useState(true)

  1210→                courses?.find((c: any) => c.id === e.course_id)?.title || null

  1414→function Payments({ payments, userId, courses }: any) {
  1415→  const [viewingReceipt, setViewingReceipt] = useState<string | null>(null)
  1416→  // Filter by the current user's id so we never display someone else's payments.
  1417→  const myPayments = (payments || []).filter((p: any) => p.user_id === userId)
  1418→  // Sort by date DESC (most recent first) for display
  1419→  const sortedPayments = [...myPayments].sort((a: any, b: any) => {
  1420→    const da = a.date ? new Date(a.date).getTime() : 0
  1421→    const db = b.date ? new Date(b.date).getTime() : 0
  1422→    return db - da
  1423→  })
  1424→
  1425→  const getCourseTitle = (courseId: string | null | undefined) => {
  1426→    if (!courseId) return null
  1427→    return courses?.find((c: any) => c.id === courseId)?.title || null
  1428→  }
  1429→
  1430→  // Determine the currency to display for a payment. We prefer the
  1431→  // course's currency (source of truth) over the payment's stored
  1432→  // currency — older payment rows may have defaulted to 'ر.ي' even
  1433→  // when the course is in USD.
  1434→  const getCurrency = (p: any) => {
  1435→    if (p.course_id) {
  1436→      const course = courses?.find((c: any) => c.id === p.course_id)
  1437→      if (course?.currency) return course.currency
  1438→    }
  1439→    return p.currency || 'ر.ي'
  1440→  }
  1441→
  1442→  // Extract coupon code from the reference field — stored as
  1443→  // "| COUPON:CODE-HERE". Returns null if no coupon was applied.
  1444→  const getCouponCode = (p: any): string | null => {
  1445→    const ref = p?.reference
  1446→    if (!ref) return null
  1447→    const match = String(ref).match(/COUPON:\s*([A-Za-z0-9_-]+)/i)
  1448→    return match ? match[1].toUpperCase() : null
  1449→  }
  1450→
  1451→  // Format currency code for nicer display (e.g. 'ر.ي' stays as is,
  1452→  // '$' gets a small "USD" tooltip-style wrapper).
  1453→  const fmtCurrency = (code: string) => code || 'ر.ي'
  1454→
  1455→  const fmtAmount = (p: any) => {
  1456→    const amt = typeof p.amount === 'number' ? p.amount : Number(p.amount) || 0
  1457→    return amt.toLocaleString('en-US')
  1458→  }
  1459→
  1460→  const fmtDate = (d: string | null | undefined) => {
  1461→    if (!d) return null
  1462→    try {
  1463→      return new Date(d).toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric' })
  1464→    } catch { return null }
  1465→  }
  1466→
  1467→  const cleanReference = (ref: string | null | undefined) => {
  1468→    if (!ref) return null
  1469→    // Strip both PROOF: and COUPON: markers so the user sees only the
  1470→    // transaction reference they typed, not the internal storage markers.
  1471→    return ref
  1472→      .replace(/\s*\|\s*PROOF:.*$/i, '')
  1473→      .replace(/\s*\|\s*COUPON:.*$/i, '')
  1474→      .trim() || null
  1475→  }
  1476→
  1477→  const getProofUrl = (p: any): string | null => {
  1478→    const ref = p?.reference
  1479→    if (!ref) return null
  1480→    const match = String(ref).match(/PROOF:\s*(\S+)/i)
  1481→    return match ? match[1] : null
  1482→  }
  1483→
  1484→  return (
  1485→    <>
  1486→    <SectionCard title="مدفوعاتي" description={`${sortedPayments.length} معاملة`}>
  1487→      {sortedPayments.length === 0 ? (
  1488→        <div className="text-center py-16">
  1489→          <div className="w-20 h-20 mx-auto mb-4 rounded-full bg-muted/40 flex items-center justify-center">
  1490→            <CreditCard className="w-10 h-10 text-muted-foreground/50" />
  1491→          </div>
  1492→          <p className="text-base font-semibold text-navy mb-1">لا توجد مدفوعات بعد</p>
  1493→          <p className="text-xs text-muted-foreground">عند تسجيلك في دورة مدفوعة ورفع إيصال الدفع، ستظهر تفاصيل المعاملة هنا.</p>
  1494→        </div>
  1495→      ) : (
  1496→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
  1497→          {sortedPayments.map((p: any) => {
  1498→            const statusInfo = getPaymentRecordStatusInfo(p.status)
  1499→            const courseTitle = getCourseTitle(p.course_id)
  1500→            const ref = cleanReference(p.reference)
  1501→            const dateStr = fmtDate(p.date)
  1502→            const proofUrl = getProofUrl(p)
  1503→            const couponCode = getCouponCode(p)
  1504→            const currency = getCurrency(p)
  1505→            const hasCoupon = !!couponCode
  1506→            return (
  1507→              <Card key={p.id} className={`p-0 overflow-hidden card-hover ${hasCoupon ? 'border-l-4 border-l-red-500 ring-1 ring-red-500/20' : 'border-border'}`}>
  1508→                {/* Amount header — has red accent when coupon applied */}
  1509→                <div className={`p-4 text-white ${hasCoupon ? 'bg-gradient-to-br from-red-700 to-red-900' : 'bg-gradient-to-br from-navy to-navy-light'}`}>
  1510→                  <div className="flex items-start justify-between gap-2">
  1511→                    <div className="min-w-0">

  1513→                        {hasCoupon ? '⚠ المبلغ بعد الخصم' : 'المبلغ'}
