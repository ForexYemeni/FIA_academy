    33→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    34→// Show only the human-readable text in the UI, but preserve the extra data on save.
    35→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    36→const splitBio = (raw: string) => {
    37→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    38→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    39→}
    40→const joinBio = (display: string, original?: string) => {
    41→  if (!original) return display
    42→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    43→  if (i === -1) return display
    44→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    45→}

    47→export function TrainerDashboard() {
    48→  const [active, setActive] = useState('overview')
    49→  const [contentCourseId, setContentCourseId] = useState<string | null>(null)
    50→  const user = useAuth((s) => s.user)
    51→  const updateUser = useAuth((s) => s.updateUser)
    52→  const logout = useAuth((s) => s.logout)
    53→  const [courses, setCourses] = useState<any[]>([])
    54→  const [students, setStudents] = useState<any[]>([])

   101→      {active === 'profile' && <TrainerProfile profile={profile} setProfile={setProfile} user={user} updateUser={updateUser} />}
   102→      {active === 'notifications' && <TrainerNotifs courses={courses} />}

   104→
   105→      {isDialogOpen && (
   106→        <CourseDialog
   107→          open={isDialogOpen}
   108→          onOpenChange={setIsDialogOpen}
   109→          course={editing}
   110→          trainerId={user?.id}
   111→          onSaved={() => {
   112→            setIsDialogOpen(false)
   113→            loadCourses()

   155→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   156→  const [deleting, setDeleting] = useState<string | null>(null)
   157→
   158→  const handleDelete = async (id: string) => {
   159→    if (!confirm('هل أنت متأكد من حذف هذه الدورة؟')) return
   160→    setDeleting(id)
   161→    try {
   162→      const res = await fetch('/api/courses', {
   163→        method: 'DELETE',
   164→        headers: { 'Content-Type': 'application/json' },
   165→        body: JSON.stringify({ id }),
   166→      })
   167→      if (res.ok) {
   168→        toast.success('تم حذف الدورة')
   169→        onRefresh()
   170→      } else {
   171→        toast.error('فشل الحذف')
   172→      }
   173→    } catch {
   174→      toast.error('خطأ في الاتصال')
   175→    } finally {
   176→      setDeleting(null)
   177→    }
   178→  }
   179→
   180→  return (
   181→    <SectionCard
   182→      title="دوراتي التدريبية"
   183→      description="إدارة الدورات التي تدرّسها"
   184→      action={
   185→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   186→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   187→        </Button>
   188→      }
   189→    >
   190→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   191→        {courses.map((c: any) => (
   192→          <Card key={c.id} className="overflow-hidden card-hover">
   193→            <div className="h-32 bg-navy relative">
   194→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   195→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   196→            </div>
   197→            <div className="p-4">
   198→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   199→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   200→                <span>{c.enrolled || 0} متدرب</span>
   201→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   202→              </div>
   203→              <div className="space-y-2">
   204→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   205→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   206→                </Button>
   207→                <div className="flex gap-2">
   208→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   209→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   210→                  </Button>
   211→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => handleDelete(c.id)} disabled={deleting === c.id}>
   212→                    <Trash2 className="w-3.5 h-3.5" />
   213→                  </Button>
   214→                </div>
   215→              </div>
   216→            </div>
   217→          </Card>
   218→        ))}
   219→      </div>
   220→    </SectionCard>
   221→  )
   222→}

   224→function CourseDialog({ open, onOpenChange, course, trainerId, onSaved }: any) {
   225→  const [form, setForm] = useState<any>(
   226→    course || {
   227→      title: '',
   228→      short_description: '',
   229→      description: '',
   230→      level: 'متوسط',
   231→      price_type: 'مجانية',
   232→      price: 0,
   233→      currency: 'ر.ي',
   234→      duration_hours: 0,
   235→      pass_score: 60,
   236→      cover: '',
   237→      status: 'منشورة',
   238→    }
   239→  )
   240→  const [saving, setSaving] = useState(false)
   241→  const [uploadingCover, setUploadingCover] = useState(false)
   242→
   243→  // Keep the form in sync with the `course` prop whenever the dialog opens or
   244→  // the course being edited changes. Without this, a dialog that stays mounted
   245→  // across course switches would show stale state (and could overwrite the
   246→  // wrong course's `cover` field).
   247→  useEffect(() => {
   248→    if (open) {
   249→      setForm(
   250→        course || {
   251→          title: '',
   252→          short_description: '',
   253→          description: '',
   254→          level: 'متوسط',
   255→          price_type: 'مجانية',
   256→          price: 0,
   257→          currency: 'ر.ي',
   258→          duration_hours: 0,
   259→          pass_score: 60,
   260→          cover: '',
   261→          status: 'منشورة',
   262→        }
   263→      )
   264→    }
   265→  }, [open, course])
   266→
   267→  const handleUploadCover = async (file: File) => {
   268→    setUploadingCover(true)
   269→    try {
   270→      const fd = new FormData()
   271→      fd.append('file', file)
   272→      fd.append('folder', 'courses')
   273→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   274→      const data = await res.json().catch(() => ({}))
   275→      if (data.url) {
   276→        // Store the Telegram-hosted URL on form.cover — it is later persisted
   277→        // to the courses.cover column by the save() call below.
   278→        setForm((prev: any) => ({ ...prev, cover: data.url }))
   279→        toast.success('تم رفع صورة الغلاف')
   280→      } else {
   281→        toast.error('فشل رفع الصورة: ' + (data.error || ''))
   282→      }
   283→    } catch {
   284→      toast.error('خطأ في الاتصال')
   285→    } finally {
   286→      setUploadingCover(false)
   287→    }
   288→  }
   289→
   290→  const save = async () => {
   291→    if (!form.title) {
   292→      toast.error('أدخل عنوان الدورة')
   293→      return
   294→    }
   295→    setSaving(true)
   296→    try {
   297→      const slug = form.title.toLowerCase().replace(/\s+/g, '-').slice(0, 80)
   298→      // Build the payload. Always include `cover` (string URL or null) so the
   299→      // courses.cover column is updated on every save — even when the trainer
   300→      // cleared the field.
   301→      const body: any = {
   302→        title: form.title,
   303→        short_description: form.short_description ?? '',
   304→        description: form.description ?? '',
   305→        level: form.level,
   306→        price_type: form.price_type,
   307→        price: Number(form.price) || 0,
   308→        currency: form.currency || 'ر.ي',
   309→        duration_hours: Number(form.duration_hours) || 0,
   310→        pass_score: Number(form.pass_score) || 60,
   311→        status: form.status || 'منشورة',
   312→        cover: form.cover ?? null,
   313→        slug,
   314→        trainer_id: trainerId,
   315→        modules_count: form.modules_count || 0,
   316→        lessons_count: form.lessons_count || 0,
   317→      }
   318→      const res = await fetch('/api/courses', {
   319→        method: course ? 'PUT' : 'POST',
   320→        headers: { 'Content-Type': 'application/json' },
   321→        body: JSON.stringify(course ? { id: course.id, ...body } : body),
   322→      })
   323→      if (res.ok) {
   324→        toast.success(course ? 'تم تحديث الدورة' : 'تم إنشاء الدورة')
   325→        onSaved()
   326→      } else {
   327→        const err = await res.json().catch(() => ({}))
   328→        toast.error(err.error || 'فشل الحفظ')
   329→      }
   330→    } catch {
   331→      toast.error('خطأ')
   332→    } finally {
   333→      setSaving(false)
   334→    }
   335→  }
   336→
   337→  return (
   338→    <Dialog open={open} onOpenChange={onOpenChange}>
   339→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   340→        <DialogHeader>
   341→          <DialogTitle>{course ? 'تعديل الدورة' : 'دورة جديدة'}</DialogTitle>
   342→        </DialogHeader>
   343→        <div className="space-y-4 py-4">
   344→          <div>
   345→            <Label>عنوان الدورة</Label>
   346→            <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} className="mt-1" />
   347→          </div>
   348→          <div>
   349→            <Label>وصف مختصر</Label>
   350→            <Input value={form.short_description || ''} onChange={(e) => setForm({ ...form, short_description: e.target.value })} className="mt-1" />
   351→          </div>
   352→          <div>
   353→            <Label>الوصف الكامل</Label>
   354→            <Textarea value={form.description || ''} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} className="mt-1" />
   355→          </div>
   356→          <div className="grid grid-cols-2 gap-3">
   357→            <div>
   358→              <Label>المستوى</Label>
   359→              <select
   360→                value={form.level}
   361→                onChange={(e) => setForm({ ...form, level: e.target.value })}
   362→                className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background"
   363→              >
   364→                <option value="مبتدئ">مبتدئ</option>
   365→                <option value="متوسط">متوسط</option>
   366→                <option value="متقدم">متقدم</option>
   367→              </select>
   368→            </div>
   369→            <div>
   370→              <Label>السعر</Label>
   371→              <select
   372→                value={form.price_type}
   373→                onChange={(e) => setForm({ ...form, price_type: e.target.value })}
   374→                className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background"
   375→              >
   376→                <option value="مجانية">مجانية</option>
   377→                <option value="مدفوعة">مدفوعة</option>
   378→              </select>
   379→            </div>
   380→          </div>
   381→          {form.price_type === 'مدفوعة' && (
   382→            <div>
   383→              <Label>المبلغ</Label>
   384→              <Input type="number" value={form.price || 0} onChange={(e) => setForm({ ...form, price: parseInt(e.target.value) || 0 })} className="mt-1" dir="ltr" />
   385→            </div>
   386→          )}
   387→          <div className="grid grid-cols-2 gap-3">
   388→            <div>
   389→              <Label>عدد الساعات</Label>
   390→              <Input type="number" value={form.duration_hours || 0} onChange={(e) => setForm({ ...form, duration_hours: parseInt(e.target.value) || 0 })} className="mt-1" dir="ltr" />
   391→            </div>
   392→            <div>
   393→              <Label>درجة النجاح (%)</Label>
   394→              <Input type="number" value={form.pass_score || 60} onChange={(e) => setForm({ ...form, pass_score: parseInt(e.target.value) || 60 })} className="mt-1" dir="ltr" />
   395→            </div>
   396→          </div>
   397→          <div>
   398→            <Label>صورة الغلاف</Label>
   399→            <div className="flex items-center gap-3 mt-1 flex-wrap">
   400→              {form.cover ? (
   401→                <div className="relative w-16 h-16 rounded-md overflow-hidden border border-border">
   402→                  <img src={form.cover} alt="cover" className="w-full h-full object-cover" />
   403→                  <button
   404→                    type="button"
   405→                    onClick={() => setForm({ ...form, cover: '' })}
   406→                    className="absolute top-0 right-0 bg-red-500 text-white rounded-bl-md p-0.5"
   407→                    title="إزالة الصورة"
   408→                  >
   409→                    <X className="w-3 h-3" />
   410→                  </button>
   411→                </div>
   412→              ) : (
   413→                <div className="w-16 h-16 rounded-md border border-dashed border-border bg-muted/30 flex items-center justify-center text-muted-foreground/40">
   414→                  <FileUp className="w-5 h-5" />
   415→                </div>
   416→              )}
   417→              <label className="cursor-pointer">
   418→                <Button
   419→                  variant="outline"
   420→                  type="button"
   421→                  disabled={uploadingCover}
   422→                  onClick={() => document.getElementById('cover-upload')?.click()}
   423→                >
   424→                  {uploadingCover ? <Loader2 className="w-4 h-4 ml-1 animate-spin" /> : <FileUp className="w-4 h-4 ml-1" />}
   425→                  {uploadingCover ? 'جاري الرفع...' : form.cover ? 'تغيير الصورة' : 'رفع صورة'}
   426→                </Button>
   427→                <input
   428→                  id="cover-upload"
   429→                  type="file"
   430→                  accept="image/*"
   431→                  className="hidden"
   432→                  onChange={(e) => { const f = e.target.files?.[0]; if (f) handleUploadCover(f); e.target.value = '' }}
   433→                />
   434→              </label>
   435→            </div>
   436→          </div>
   437→        </div>
   438→        <DialogFooter>
   439→          <Button variant="outline" onClick={() => onOpenChange(false)}>إلغاء</Button>
   440→          <Button onClick={save} disabled={saving} className="bg-navy hover:bg-navy-light">
   441→            {saving ? 'جاري الحفظ...' : 'حفظ'}
   442→          </Button>
   443→        </DialogFooter>
   444→      </DialogContent>
   445→    </Dialog>
   446→  )
   447→}

   449→function TrainerStudents({ students, courses }: any) {
   450→  const [enrollments, setEnrollments] = useState<any[]>([])
   451→  const [enrollmentsLoaded, setEnrollmentsLoaded] = useState(false)
   452→
   453→  useEffect(() => {
   454→    fetch('/api/enrollments').then((r) => r.json()).then((d) => {
   455→      const all = Array.isArray(d) ? d : []
   456→      // Filter to ONLY enrollments in this trainer's courses
   457→      const trainerCourseIds = courses.map((c: any) => c.id)
   458→      const mine = all.filter((e: any) => trainerCourseIds.includes(e.course_id))
   459→      setEnrollments(mine)
   460→      setEnrollmentsLoaded(true)
   461→    }).catch(() => setEnrollmentsLoaded(true))
   462→  }, [courses])
   463→
   464→  // Helper: get all enrollments for a student
   465→  const studentEnrollments = (studentId: string) => enrollments.filter((e) => e.user_id === studentId)
   466→
   467→  // Filter students to ONLY those enrolled in at least one of this trainer's courses.
   468→  // The `enrollments` state is already scoped to the trainer's course IDs, so its
   469→  // user_id values are exactly the students we want to show.
   470→  const enrolledStudentIds = new Set(enrollments.map((e: any) => e.user_id))

   567→function TrainerExams({ courses }: { courses: any[] }) {
   568→  const [loading, setLoading] = useState(true)
   569→  const [exams, setExams] = useState<any[]>([])
   570→
   571→  useEffect(() => {
   572→    let cancelled = false
   573→    const loadExams = async () => {
   574→      if (!courses || courses.length === 0) {
   575→        setExams([])
   576→        setLoading(false)
   577→        return
   578→      }
   579→      setLoading(true)
   580→      try {
   581→        // Fetch lessons for each course in parallel
   582→        const results = await Promise.all(
   583→          courses.map(async (c) => {
   584→            try {
   585→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   586→              const data = await res.json()
   587→              const lessons = Array.isArray(data) ? data : []
   588→              // Filter exam-type lessons only
   589→              return lessons
   590→                .filter((l: any) => l.lesson_type === 'exam')
   591→                .map((l: any) => {
   592→                  // Parse question count from content JSON
   593→                  let questions: any[] = []
   594→                  if (l.content) {
   595→                    try {
   596→                      const parsed = JSON.parse(l.content)
   597→                      if (Array.isArray(parsed)) questions = parsed
   598→                    } catch {}
   599→                  }
   600→                  return {
   601→                    ...l,
   602→                    course_id: c.id,
   603→                    course_title: c.title,
   604→                    course_pass_score: c.pass_score,
   605→                    questions_count: questions.length,
   606→                    questions,
   607→                  }
   608→                })
   609→            } catch {
   610→              return []
   611→            }
   612→          })
   613→        )
   614→        if (cancelled) return
   615→        const flat = results.flat()
   616→        setExams(flat)
   617→      } catch {
   618→        if (!cancelled) setExams([])
   619→      } finally {
   620→        if (!cancelled) setLoading(false)
   621→      }
   622→    }
   623→    loadExams()
   624→    return () => { cancelled = true }
   625→  }, [courses])
   626→
   627→  const totalQuestions = exams.reduce((acc, e) => acc + (e.questions_count || 0), 0)
   628→  const totalCoursesWithExams = new Set(exams.map((e) => e.course_id)).size
   629→
   630→  return (
   631→    <div className="space-y-6">
   632→      <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
   633→        <StatCard icon={FileText} label="الاختبارات" value={exams.length} color="blue" />
   634→        <StatCard icon={HelpCircle} label="إجمالي الأسئلة" value={totalQuestions} color="navy" />
   635→        <StatCard icon={BookOpen} label="دورات بها اختبارات" value={totalCoursesWithExams} color="gold" />
   636→      </div>
   637→
   638→      <SectionCard
   639→        title="الاختبارات"
   640→        description="الاختبارات المستندة إلى محتوى دوراتك التدريبية"
   641→      >
   642→        {loading ? (
   643→          <div className="text-center py-12">
   644→            <Loader2 className="w-8 h-8 text-navy mx-auto mb-3 animate-spin" />
   645→            <p className="text-muted-foreground text-sm">جاري تحميل الاختبارات...</p>
   646→          </div>
   647→        ) : exams.length === 0 ? (
   648→          <div className="text-center py-12">
   649→            <FileText className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   650→            <p className="text-muted-foreground mb-2">لا توجد اختبارات بعد</p>
   651→            <p className="text-xs text-muted-foreground">
   652→              أنشئ دروساً من نوع "اختبار" في محتوى دوراتك لتظهر هنا
   653→            </p>
   654→          </div>
   655→        ) : (
   656→          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   657→            {exams.map((exam) => (
   658→              <Card key={exam.id} className="overflow-hidden card-hover">
   659→                <div className="p-4 border-b border-border bg-purple-500/5">
   660→                  <div className="flex items-start justify-between gap-2 mb-2">
   661→                    <Badge className="bg-purple-500/15 text-purple-600 border-purple-500/30 flex-shrink-0">
   662→                      <HelpCircle className="w-3 h-3 ml-1" /> اختبار
   663→                    </Badge>
   664→                    <Badge variant="outline" className="text-xs flex-shrink-0">
   665→                      {exam.questions_count} سؤال
   666→                    </Badge>
   667→                  </div>
   668→                  <h3 className="font-bold text-navy text-sm mb-1 line-clamp-2">{exam.title}</h3>
   669→                  <div className="text-xs text-muted-foreground flex items-center gap-1.5">
   670→                    <BookOpen className="w-3 h-3" />
   671→                    <span className="truncate">{exam.course_title}</span>

   952→function TrainerNotifs({ courses }: { courses: any[] }) {
   953→  const [loading, setLoading] = useState(true)
   954→  const [notifs, setNotifs] = useState<any[]>([])
   955→
   956→  useEffect(() => {
   957→    let cancelled = false
   958→    const loadNotifs = async () => {
   959→      if (!courses || courses.length === 0) {
   960→        setNotifs([])
   961→        setLoading(false)
   962→        return
   963→      }
   964→      setLoading(true)
   965→      try {
   966→        const courseIds = courses.map((c) => c.id)
   967→        const courseById = new Map(courses.map((c) => [c.id, c]))
   968→
   969→        // Fetch enrollments for all courses (we'll filter on the client)
   970→        const [enrRes, stuRes] = await Promise.all([
   971→          fetch('/api/enrollments'),
   972→          fetch('/api/students?role=student'),
   973→        ])
   974→        const [enrollments, students] = await Promise.all([enrRes.json(), stuRes.json()])
   975→
   976→        if (cancelled) return
   977→
   978→        const enrList = Array.isArray(enrollments) ? enrollments : []
   979→        const stuList = Array.isArray(students) ? students : []
   980→        const studentById = new Map(stuList.map((s: any) => [s.id, s]))
   981→
   982→        // Only enrollments for this trainer's courses
   983→        const mine = enrList.filter((e: any) => courseIds.includes(e.course_id))
   984→
   985→        // Sort by enrolled_at desc
   986→        mine.sort((a: any, b: any) => {
   987→          const aT = a.enrolled_at ? new Date(a.enrolled_at).getTime() : 0
   988→          const bT = b.enrolled_at ? new Date(b.enrolled_at).getTime() : 0
   989→          return bT - aT
   990→        })
   991→
   992→        // Build notification objects — newest 25
   993→        const built = mine.slice(0, 25).map((e: any) => {
   994→          const student = studentById.get(e.user_id)
   995→          const course = courseById.get(e.course_id)
   996→          const isFree = !e.amount_paid || e.amount_paid === 0
   997→          return {
   998→            id: e.id,
   999→            type: 'enrollment',
  1000→            studentName: student?.name || 'متدرب',
  1001→            studentAvatar: student?.avatar,
  1002→            courseTitle: course?.title || 'دورة',
  1003→            enrolledAt: e.enrolled_at,
  1004→            isFree,
  1005→            amountPaid: e.amount_paid,
  1006→            currency: e.currency,
  1007→            paymentStatus: e.payment_status,
  1008→          }
  1009→        })
  1010→
  1011→        if (!cancelled) setNotifs(built)
  1012→      } catch {
  1013→        if (!cancelled) setNotifs([])
  1014→      } finally {
  1015→        if (!cancelled) setLoading(false)
  1016→      }
  1017→    }
  1018→    loadNotifs()
  1019→    return () => { cancelled = true }
  1020→  }, [courses])
  1021→
  1022→  const formatRelativeTime = (iso: string | null | undefined) => {
  1023→    if (!iso) return ''
  1024→    try {
  1025→      const date = new Date(iso)
  1026→      const diffMs = Date.now() - date.getTime()
  1027→      const diffMin = Math.floor(diffMs / 60000)
  1028→      if (diffMin < 1) return 'الآن'
  1029→      if (diffMin < 60) return `منذ ${diffMin} دقيقة`
  1030→      const diffHr = Math.floor(diffMin / 60)
  1031→      if (diffHr < 24) return `منذ ${diffHr} ساعة`
  1032→      const diffDay = Math.floor(diffHr / 24)
  1033→      if (diffDay < 30) return `منذ ${diffDay} يوم`
  1034→      return date.toLocaleDateString('ar-EG')
  1035→    } catch {
  1036→      return ''
  1037→    }
  1038→  }
  1039→
  1040→  return (
  1041→    <SectionCard
  1042→      title="الإشعارات"
  1043→      description="آخر التسجيلات في دوراتك التدريبية"
  1044→    >
  1045→      {loading ? (
  1046→        <div className="text-center py-12">
  1047→          <Loader2 className="w-8 h-8 text-navy mx-auto mb-3 animate-spin" />
  1048→          <p className="text-muted-foreground text-sm">جاري تحميل الإشعارات...</p>
  1049→        </div>
  1050→      ) : notifs.length === 0 ? (
  1051→        <div className="text-center py-12">
  1052→          <Bell className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
  1053→          <p className="text-muted-foreground">لا توجد إشعارات</p>
  1054→          <p className="text-xs text-muted-foreground mt-1">
  1055→            ستظهر هنا إشعارات التسجيل الجديدة في دوراتك
  1056→          </p>
  1057→        </div>
  1058→      ) : (
  1059→        <div className="space-y-3">
  1060→          {notifs.map((n) => (
  1061→            <div
  1062→              key={n.id}
  1063→              className="p-4 bg-gold/5 border border-gold/20 rounded-lg flex items-start gap-3"
  1064→            >
  1065→              <div className="w-10 h-10 rounded-full bg-navy flex items-center justify-center text-gold font-bold flex-shrink-0 border border-gold/30 overflow-hidden">
  1066→                {n.studentAvatar ? (
  1067→                  <img src={n.studentAvatar} alt={n.studentName} className="w-full h-full object-cover" />
  1068→                ) : (
  1069→                  <UserPlus className="w-5 h-5" />
  1070→                )}
  1071→              </div>
  1072→              <div className="flex-1 min-w-0">
  1073→                <div className="font-semibold text-navy text-sm">
  1074→                  متدرب جديد في دورتك
  1075→                </div>
  1076→                <div className="text-xs text-muted-foreground mt-1">
  1077→                  <span className="font-medium text-navy">{n.studentName}</span>
  1078→                  {' '}انضم إلى{' '}
  1079→                  <span className="font-medium text-navy">{n.courseTitle}</span>
  1080→                </div>
  1081→                <div className="flex items-center gap-2 mt-2 flex-wrap">
  1082→                  <Badge
  1083→                    className={
  1084→                      n.isFree
  1085→                        ? 'bg-green-500/15 text-green-600 border-green-500/30 text-xs'
  1086→                        : 'bg-gold/15 text-gold border-gold/30 text-xs'
  1087→                    }
  1088→                  >
  1089→                    {n.isFree ? 'مجانية' : `${(n.amountPaid || 0).toLocaleString()} ${n.currency || 'ر.ي'}`}
  1090→                  </Badge>
  1091→                  {n.paymentStatus === 'pending' && (
  1092→                    <Badge variant="outline" className="border-yellow-500/30 text-yellow-600 text-xs">
  1093→                      بانتظار الدفع
  1094→                    </Badge>
  1095→                  )}
  1096→                  {n.enrolledAt && (
  1097→                    <span className="text-xs text-muted-foreground">
  1098→                      {formatRelativeTime(n.enrolledAt)}
  1099→                    </span>
  1100→                  )}
  1101→                </div>
  1102→              </div>
  1103→            </div>
  1104→          ))}
  1105→        </div>
  1106→      )}
  1107→    </SectionCard>
  1108→  )
  1109→}
