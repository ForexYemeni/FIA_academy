    24→// ============================================================================
    25→// Reusable reaction buttons component (like / dislike)
    26→// ============================================================================
    27→function ReactionButtons({ type, targetId, userId, initialLikes, initialDislikes }: {
    28→  type: 'topic' | 'post' | 'blog'
    29→  targetId: string
    30→  userId?: string
    31→  initialLikes: number
    32→  initialDislikes: number
    33→}) {
    34→  const [likes, setLikes] = useState(initialLikes)
    35→  const [dislikes, setDislikes] = useState(initialDislikes)
    36→  const [myReaction, setMyReaction] = useState<'like' | 'dislike' | null>(null)
    37→  const [busy, setBusy] = useState(false)
    38→
    39→  const react = async (reaction: 'like' | 'dislike') => {
    40→    if (!userId || busy) return
    41→    setBusy(true)
    42→    try {
    43→      const apiPath = type === 'blog' ? '/api/blog/react' : '/api/forum/react'
    44→      const body: any = { user_id: userId, reaction }
    45→      if (type === 'topic') body.topic_id = targetId
    46→      else if (type === 'post') body.post_id = targetId
    47→      else if (type === 'blog') body.blog_post_id = targetId
    48→
    49→      const res = await fetch(apiPath, {
    50→        method: 'POST', headers: { 'Content-Type': 'application/json' },
    51→        body: JSON.stringify(body),
    52→      })
    53→      const data = await res.json()
    54→
    55→      if (data.action === 'added') {
    56→        setMyReaction(reaction)
    57→        if (reaction === 'like') setLikes(l => l + 1)
    58→        else setDislikes(d => d + 1)
    59→      } else if (data.action === 'removed') {
    60→        setMyReaction(null)
    61→        if (reaction === 'like') setLikes(l => Math.max(0, l - 1))
    62→        else setDislikes(d => Math.max(0, d - 1))
    63→      } else if (data.action === 'switched') {
    64→        setMyReaction(reaction)
    65→        if (reaction === 'like') {
    66→          setLikes(l => l + 1)
    67→          setDislikes(d => Math.max(0, d - 1))
    68→        } else {
    69→          setDislikes(d => d + 1)
    70→          setLikes(l => Math.max(0, l - 1))
    71→        }
    72→      }
    73→    } catch {} finally { setBusy(false) }
    74→  }
    75→
    76→  return (
    77→    <div className="flex items-center gap-2">
    78→      <button
    79→        onClick={() => react('like')}
    80→        disabled={busy || !userId}
    81→        className={`flex items-center gap-1 px-2.5 py-1 rounded-lg text-xs font-bold transition-all disabled:opacity-50 ${
    82→          myReaction === 'like'
    83→            ? 'bg-green-500/15 text-green-600 border border-green-500/30'
    84→            : 'bg-muted/50 text-muted-foreground hover:bg-muted border border-transparent'
    85→        }`}
    86→      >
    87→        {type === 'blog' ? <Heart className={`w-3.5 h-3.5 ${myReaction === 'like' ? 'fill-green-500 text-green-500' : ''}`} /> : <ThumbsUp className="w-3.5 h-3.5" />}
    88→        {likes > 0 && <span>{likes}</span>}
    89→      </button>
    90→      <button
    91→        onClick={() => react('dislike')}
    92→        disabled={busy || !userId}
    93→        className={`flex items-center gap-1 px-2.5 py-1 rounded-lg text-xs font-bold transition-all disabled:opacity-50 ${
    94→          myReaction === 'dislike'
    95→            ? 'bg-red-500/15 text-red-600 border border-red-500/30'
    96→            : 'bg-muted/50 text-muted-foreground hover:bg-muted border border-transparent'
    97→        }`}
    98→      >
    99→        <ThumbsDown className="w-3.5 h-3.5" />
   100→        {dislikes > 0 && <span>{dislikes}</span>}
   101→      </button>
   102→    </div>
   103→  )
   104→}

   109→function BlogCommentSection({ blogPostId, userId }: { blogPostId: string; userId?: string }) {
   110→  const [comments, setComments] = useState<any[]>([])
   111→  const [loading, setLoading] = useState(true)
   112→  const [commentText, setCommentText] = useState('')
   113→  const [submitting, setSubmitting] = useState(false)
   114→
   115→  const load = useCallback(async () => {
   116→    try {
   117→      const res = await fetch(`/api/blog/comments?blog_post_id=${blogPostId}`)
   118→      const data = await res.json()
   119→      setComments(Array.isArray(data) ? data : [])
   120→    } catch {} finally { setLoading(false) }
   121→  }, [blogPostId])
   122→
   123→  useEffect(() => { load() }, [load])
   124→
   125→  const submit = async () => {
   126→    if (!commentText.trim() || !userId) return
   127→    setSubmitting(true)
   128→    try {
   129→      const res = await fetch('/api/blog/comments', {
   130→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   131→        body: JSON.stringify({ blog_post_id: blogPostId, user_id: userId, body: commentText }),
   132→      })
   133→      const data = await res.json()
   134→      if (res.ok) {
   135→        setCommentText('')
   136→        load()
   137→        toast.success('تم نشر تعليقك')
   138→      }
   139→    } catch {} finally { setSubmitting(false) }
   140→  }
   141→
   142→  const handleDelete = async (id: string) => {
   143→    try {
   144→      await fetch('/api/blog/comments', {
   145→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   146→        body: JSON.stringify({ id }),
   147→      })
   148→      load()
   149→      toast.success('تم حذف التعليق')
   150→    } catch {}
   151→  }
   152→
   153→  return (
   154→    <div className="mt-8 pt-6 border-t border-border">
   155→      <h3 className="font-bold text-navy mb-4 flex items-center gap-2">
   156→        <MessageCircle className="w-5 h-5 text-gold" /> التعليقات ({comments.length})
   157→      </h3>
   158→
   159→      {userId && (
   160→        <div className="mb-6 bg-muted/30 rounded-lg p-3 border border-border">
   161→          <Textarea
   162→            value={commentText}
   163→            onChange={(e) => setCommentText(e.target.value)}
   164→            rows={3}
   165→            placeholder="أكتب تعليقك..."
   166→            dir="rtl"
   167→            disabled={submitting}
   168→          />
   169→          <div className="flex justify-end mt-2">
   170→            <Button onClick={submit} disabled={submitting || !commentText.trim()} className="bg-navy hover:bg-navy-light text-xs">
   171→              {submitting ? <Loader2 className="w-3.5 h-3.5 ml-1 animate-spin" /> : <Send className="w-3.5 h-3.5 ml-1" />}
   172→              {submitting ? 'جاري...' : 'نشر التعليق'}
   173→            </Button>
   174→          </div>
   175→        </div>
   176→      )}
   177→
   178→      {loading ? (
   179→        <div className="text-center py-4"><Loader2 className="w-5 h-5 mx-auto animate-spin text-navy" /></div>
   180→      ) : comments.length === 0 ? (
   181→        <p className="text-sm text-muted-foreground text-center py-4">لا توجد تعليقات بعد. كن أول من يعلّق!</p>
   182→      ) : (
   183→        <div className="space-y-3">
   184→          {comments.map((c) => (
   185→            <div key={c.id} className="bg-white border border-border rounded-lg p-3 flex gap-3 group">
   186→              <Avatar className="w-8 h-8 border border-gold/20 flex-shrink-0">
   187→                <AvatarImage src={c.author?.avatar || ''} />
   188→                <AvatarFallback className="bg-navy text-gold text-xs">{c.author?.name?.charAt(0) || '؟'}</AvatarFallback>
   189→              </Avatar>
   190→              <div className="flex-1 min-w-0">
   191→                <div className="flex items-center justify-between mb-1">
   192→                  <span className="text-xs font-bold text-navy">{c.author?.name || 'مستخدم'}</span>
   193→                  <div className="flex items-center gap-2">
   194→                    <span className="text-[10px] text-muted-foreground">
   195→                      <Clock className="w-2.5 h-2.5 inline" /> {new Date(c.created_at).toLocaleDateString('ar-EG')}
   196→                    </span>
   197→                    {c.user_id === userId && (
   198→                      <button onClick={() => handleDelete(c.id)} className="opacity-0 group-hover:opacity-100 text-red-500 hover:bg-red-500/10 p-1 rounded transition-opacity" title="حذف">
   199→                        <Trash2 className="w-3 h-3" />
   200→                      </button>
   201→                    )}
   202→                  </div>
   203→                </div>
   204→                <p className="text-sm text-gray-700 whitespace-pre-wrap leading-relaxed">{c.body}</p>
   205→              </div>
   206→            </div>
   207→          ))}
   208→        </div>
   209→      )}
   210→    </div>
   211→  )
   212→}

   217→function roleBadge(role?: string) {
   218→  if (!role) return null
   219→  const map: Record<string, { label: string; cls: string }> = {
   220→    admin: { label: '🔐 مدير', cls: 'bg-red-500/15 text-red-600 border-red-500/30' },
   221→    trainer: { label: '🎓 مدرب', cls: 'bg-green-500/15 text-green-600 border-green-500/30' },
   222→    student: { label: '📚 متدرب', cls: 'bg-blue-500/15 text-blue-600 border-blue-500/30' },
   223→    founder: { label: '👑 المؤسس', cls: 'bg-gold/15 text-gold border-gold/30' },
   224→    leader: { label: '🛡️ قائد', cls: 'bg-purple-500/15 text-purple-600 border-purple-500/30' },
   225→    team: { label: '🤝 فريق', cls: 'bg-gray-500/15 text-gray-600 border-gray-500/30' },
   226→  }
   227→  const cfg = map[role]
   228→  if (!cfg) return null
   229→  return <Badge className={`text-[10px] ${cfg.cls}`}>{cfg.label}</Badge>
   230→}

   246→}

   248→// ============================================================================
   249→// SHARED FORUM — used by student / trainer / team dashboards
   250→// ============================================================================
   251→export function SharedForum({ user }: { user: any }) {
   252→  const [categories, setCategories] = useState<any[]>([])
   253→  const [activeCategory, setActiveCategory] = useState<string>('')
   254→  const [topics, setTopics] = useState<any[]>([])
   255→  const [viewingTopic, setViewingTopic] = useState<any | null>(null)
   256→  const [posts, setPosts] = useState<any[]>([])
   257→  const [newTopicOpen, setNewTopicOpen] = useState(false)
   258→  const [banInfo, setBanInfo] = useState<any>(null)
   259→  const [loading, setLoading] = useState(true)
   260→  const [replyText, setReplyText] = useState('')
   261→  const [replying, setReplying] = useState(false)
   262→  const [banTarget, setBanTarget] = useState<any | null>(null) // {userId, name}
   263→  const [banReason, setBanReason] = useState('')
   264→  const [banning, setBanning] = useState(false)
   265→
   266→  const canBan = user && ['admin', 'trainer', 'founder', 'leader', 'team'].includes(user.role)
   267→
   268→  const loadCategories = useCallback(async () => {
   269→    const res = await fetch('/api/forum-categories')
   270→    const data = await res.json()
   271→    setCategories(Array.isArray(data) ? data : [])
   272→    if (data.length > 0 && !activeCategory) setActiveCategory(data[0].id)
   273→  }, [activeCategory])
   274→
   275→  const loadTopics = useCallback(async () => {
   276→    if (!activeCategory) return
   277→    setLoading(true)
   278→    const res = await fetch(`/api/forum-topics?category_id=${activeCategory}`)
   279→    const data = await res.json()
   280→    setTopics(Array.isArray(data) ? data : [])
   281→    setLoading(false)
   282→  }, [activeCategory])
   283→
   284→  const checkBan = useCallback(async () => {
   285→    if (!user?.id) return
   286→    const res = await fetch(`/api/forum-bans?user_id=${user.id}`)
   287→    const data = await res.json()
   288→    if (data.banned) setBanInfo(data.bans?.[0] || { reason: 'تم حظرك من المنتدى' })
   289→  }, [user?.id])
   290→
   291→  useEffect(() => { loadCategories() }, [loadCategories])
   292→  useEffect(() => { loadTopics() }, [loadTopics])
   293→  useEffect(() => { checkBan() }, [checkBan])
   294→
   295→  // Auto-open a topic when arriving via a deep-link notification.
   296→  // Notification link format: `/#forum?t=<topic-id>`. The useHashTab hook
   297→  // in the dashboard shell sets the active tab to `forum`; then this
   298→  // effect parses `?t=ID` from the hash and fetches the topic + posts.
   299→  useEffect(() => {
   300→    if (typeof window === 'undefined') return
   301→    const applyTopicDeepLink = async () => {
   302→      const hash = window.location.hash.replace(/^#/, '')
   303→      if (!hash) return
   304→      const params = new URLSearchParams(hash.split('?')[1] || hash.split('&').slice(1).join('&'))
   305→      const tId = params.get('t')
   306→      if (!tId || !topics.length) return
   307→      // Find the topic in the loaded list
   308→      const topic = topics.find((x: any) => x.id === tId)
   309→      if (topic) {
   310→        await openTopic(topic)
   311→        // Clean the hash so refresh doesn't reopen the topic
   312→        try { history.replaceState(null, '', window.location.pathname + window.location.search) } catch {}
   313→      } else {
   314→        // Topic might be in another category — fetch directly
   315→        try {
   316→          const r = await fetch(`/api/forum-topics?topic_id=${tId}`)
   317→          const d = await r.json()
   318→          if (d && d.id) {
   319→            await openTopic(d)
   320→            try { history.replaceState(null, '', window.location.pathname + window.location.search) } catch {}
   321→          }
   322→        } catch {}
   323→      }
   324→    }
   325→    applyTopicDeepLink()
   326→    const handler = () => applyTopicDeepLink()
   327→    window.addEventListener('hashchange', handler)
   328→    return () => window.removeEventListener('hashchange', handler)
   329→  }, [topics])
   330→
   331→  const openTopic = async (t: any) => {
   332→    setViewingTopic(t)
   333→    const res = await fetch(`/api/forum?topic_id=${t.id}`)
   334→    const data = await res.json()
   335→    setPosts(Array.isArray(data) ? data : [])
   336→  }
   337→
   338→  const submitReply = async () => {
   339→    if (!replyText.trim() || !viewingTopic || !user?.id) return
   340→    setReplying(true)
   341→    try {
   342→      const res = await fetch('/api/forum', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({ topic_id: viewingTopic.id, user_id: user.id, body: replyText }),
   346→      })
   347→      const data = await res.json()
   348→      if (!res.ok) {
   349→        toast.error(data.error || 'فشل الإرسال')
   350→        return
   351→      }
   352→      // Refresh posts (with author profiles)
   353→      const postsRes = await fetch(`/api/forum?topic_id=${viewingTopic.id}`)
   354→      const postsData = await postsRes.json()
   355→      setPosts(Array.isArray(postsData) ? postsData : [])
   356→      setReplyText('')
   357→      toast.success('تم نشر ردك')
   358→    } finally { setReplying(false) }
   359→  }
   360→
   361→  const handleBan = async () => {
   362→    if (!banTarget || !user?.id) return
   363→    setBanning(true)
   364→    try {
   365→      const res = await fetch('/api/forum-bans', {
   366→        method: 'POST',
   367→        headers: { 'Content-Type': 'application/json' },
   368→        body: JSON.stringify({
   369→          banned_user_id: banTarget.userId,
   370→          banned_by_user_id: user.id,
   371→          reason: banReason || 'مخالفة قواعد المنتدى',
   372→          scope: 'all',
   373→        }),
   374→      })
   375→      const data = await res.json()
   376→      if (!res.ok) { toast.error(data.error || 'فشل الحظر'); return }
   377→      toast.success(`تم حظر ${banTarget.name} من المنتدى`)
   378→      setBanTarget(null); setBanReason('')
   379→    } finally { setBanning(false) }
   380→  }
   381→
   382→  if (banInfo) {
   383→    return (
   384→      <SectionCard title="المنتدى" description="تم حظرك من المنتدى">
   385→        <div className="text-center py-12 max-w-md mx-auto">
   386→          <div className="w-20 h-20 mx-auto rounded-full bg-red-500/10 flex items-center justify-center mb-4">
   387→            <Ban className="w-10 h-10 text-red-500" />
   388→          </div>
   389→          <h3 className="text-xl font-bold text-navy mb-2">تم حظرك من المنتدى</h3>
   390→          <p className="text-sm text-muted-foreground mb-4">{banInfo.reason || 'لتواصلك مع الإدارة لمعرفة السبب'}</p>
   391→          {banInfo.expires_at && (
   392→            <p className="text-xs text-muted-foreground">ينتهي الحظر في: {new Date(banInfo.expires_at).toLocaleDateString('ar-EG')}</p>
   393→          )}
   394→        </div>
   395→      </SectionCard>
   396→    )
   397→  }
   398→
   399→  if (viewingTopic) {
   400→    return (
   401→      <SectionCard
   402→        title={viewingTopic.title}
   403→        description={`${posts.length} رد`}
   404→        action={
   405→          <Button variant="outline" onClick={() => setViewingTopic(null)}>
   406→            <ArrowRight className="w-4 h-4 ml-1" /> رجوع للقائمة
   407→          </Button>
   408→        }
   409→      >
   410→        <div className="space-y-3">
   411→          {/* Original post (the topic body is the first "post") */}
   412→          {viewingTopic.body && (
   413→            <Card className="p-4 border-gold/30 bg-gold/5">
   414→              <div className="flex items-center justify-between mb-3">
   415→                <AuthorChip author={viewingTopic.author} />
   416→                <span className="text-xs text-muted-foreground">
   417→                  <Clock className="w-3 h-3 inline" /> {new Date(viewingTopic.created_at).toLocaleDateString('ar-EG')}
   418→                </span>
   419→              </div>
   420→              <p className="text-sm text-navy whitespace-pre-wrap leading-relaxed">{viewingTopic.body}</p>
   421→              {/* Reaction buttons for the topic itself */}
   422→              <div className="mt-3 pt-2 border-t border-gold/20">
   423→                <ReactionButtons type="topic" targetId={viewingTopic.id} userId={user?.id} initialLikes={viewingTopic.likes_count || 0} initialDislikes={viewingTopic.dislikes_count || 0} />
   424→              </div>
   425→            </Card>
   426→          )}
   427→
   428→          {posts.map((p) => (
   429→            <Card key={p.id} className={`p-4 ${p.is_solution ? 'border-green-500/40 bg-green-500/5' : ''}`}>
   430→              <div className="flex items-start justify-between gap-3 mb-3">
   431→                <AuthorChip author={p.author} />
   432→                <div className="flex items-center gap-2">
   433→                  <span className="text-xs text-muted-foreground">
   434→                    <Clock className="w-3 h-3 inline" /> {new Date(p.created_at).toLocaleDateString('ar-EG')}
   435→                  </span>
   436→                  {/* Ban button — only for trainers/admin/team, not for self */}
   437→                  {canBan && p.user_id !== user?.id && p.author?.role !== 'admin' && (
   438→                    <button
   439→                      onClick={() => setBanTarget({ userId: p.user_id, name: p.author?.name || 'مستخدم' })}
   440→                      className="text-xs text-red-500 hover:bg-red-500/10 px-2 py-1 rounded"
   441→                      title="حظر من المنتدى"
   442→                    >
   443→                      <Ban className="w-3.5 h-3.5" />
   444→                    </button>
   445→                  )}
   446→                  {p.is_solution && <Badge className="bg-green-500/15 text-green-600 border-green-500/30 text-[10px]">✓ الحل المختار</Badge>}
   447→                </div>
   448→              </div>
   449→              <p className="text-sm text-gray-700 whitespace-pre-wrap leading-relaxed">{p.body}</p>
   450→              {/* Reaction buttons for forum posts */}
   451→              <div className="mt-3 pt-2 border-t border-border">
   452→                <ReactionButtons type="post" targetId={p.id} userId={user?.id} initialLikes={p.likes_count || 0} initialDislikes={p.dislikes_count || 0} />
   453→              </div>
   454→            </Card>
   455→          ))}
   456→
   457→          {/* Reply box */}
   458→          <Card className="p-4 bg-muted/30">
   459→            <Label className="mb-2 block">أضف رداً</Label>
   460→            <Textarea
   461→              value={replyText}
   462→              onChange={(e) => setReplyText(e.target.value)}

   803→export function AdminForumBans() {
   804→  const [bans, setBans] = useState<any[]>([])
   805→  const [profiles, setProfiles] = useState<Record<string, any>>({})
   806→  const [searchQ, setSearchQ] = useState('')
   807→  const [searchResults, setSearchResults] = useState<any[]>([])
   808→  const [searching, setSearching] = useState(false)
   809→  const [banReason, setBanReason] = useState('')
   810→  const [deleteTarget, setDeleteTarget] = useState<any | null>(null)
   811→  const [deleting, setDeleting] = useState(false)
   812→
   813→  const loadBans = useCallback(async () => {
   814→    const res = await fetch('/api/forum-bans')
   815→    const data = await res.json()
   816→    const arr = Array.isArray(data) ? data : []
   817→    setBans(arr)
   818→    // Fetch profiles for all banned users
   819→    const ids = arr.map((b: any) => b.banned_user_id).join(',')
   820→    if (ids) {
   821→      const pRes = await fetch(`/api/public-profiles?ids=${ids}`)
   822→      const pData = await pRes.json()
   823→      const map: Record<string, any> = {}
   824→      for (const p of Array.isArray(pData) ? pData : []) map[p.user_id] = p
   825→      setProfiles(map)
   826→    }
   827→  }, [])
   828→
   829→  useEffect(() => { loadBans() }, [loadBans])
   830→
   831→  const searchUsers = async (q: string) => {
   832→    setSearchQ(q)
   833→    if (q.length < 2) { setSearchResults([]); return }
   834→    setSearching(true)
   835→    try {
   836→      const res = await fetch(`/api/users-search?q=${encodeURIComponent(q)}`)
   837→      const data = await res.json()
   838→      setSearchResults(Array.isArray(data) ? data : [])
   839→    } finally { setSearching(false) }
   840→  }

   843→    const adminUser = useAuth.getState().user
   844→    if (!adminUser) return

   846→      method: 'POST',
   847→      headers: { 'Content-Type': 'application/json' },
   848→      body: JSON.stringify({
   849→        banned_user_id: user.id,
   850→        banned_by_user_id: adminUser.id,
   851→        reason: banReason || 'مخالفة قواعد المنتدى',
   852→        scope: 'all',
   853→      }),
   854→    })
   855→    const data = await res.json()
   856→    if (!res.ok) { toast.error(data.error || 'فشل الحظر'); return }
   857→    toast.success(`تم حظر ${user.name} من المنتدى`)
   858→    setBanReason('')
   859→    setSearchQ(''); setSearchResults([])
   860→    loadBans()
   861→  }
   862→
   863→  const handleUnban = async () => {
   864→    if (!deleteTarget) return
   865→    setDeleting(true)

   867→      await fetch('/api/forum-bans', {
   868→        method: 'DELETE',
   869→        headers: { 'Content-Type': 'application/json' },
   870→        body: JSON.stringify({ id: deleteTarget.id }),
   871→      })
   872→      toast.success('تم رفع الحظر')
   873→      setDeleteTarget(null)
   874→      loadBans()
   875→    } finally { setDeleting(false) }
   876→  }
   877→
   878→  return (
   879→    <SectionCard title="حظر المنتدى" description={`${bans.length} حظر نشط`}>
   880→      <div className="space-y-6">
   881→        {/* Search + ban section */}
   882→        <Card className="p-4 bg-navy/5 border-gold/20">
   883→          <h4 className="font-bold text-navy mb-3 flex items-center gap-2">
   884→            <Ban className="w-4 h-4 text-red-500" /> حظر مستخدم جديد

   888→              <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />

   890→                value={searchQ}
   891→                onChange={(e) => searchUsers(e.target.value)}
   892→                placeholder="ابحث بالاسم عن متدرب أو مدرب..."
   893→                className="pr-10"
   894→              />
   895→            </div>
   896→            {searching && <div className="text-xs text-muted-foreground"><Loader2 className="w-3 h-3 inline ml-1 animate-spin" /> جاري البحث...</div>}

   900→                  <div key={u.id} className="flex items-center justify-between p-2 rounded bg-white border">
   901→                    <div className="flex items-center gap-2">

   913→                      <Ban className="w-3.5 h-3.5 ml-1" /> حظر
