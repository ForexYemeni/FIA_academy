    32→    32→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    33→    33→// Show only the human-readable text in the UI, but preserve the extra data on save.
    34→    34→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    35→    35→const splitBio = (raw: string) => {
    36→    36→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    37→    37→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    38→    38→}
    39→    39→const joinBio = (display: string, original?: string) => {
    40→    40→  if (!original) return display
    41→    41→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    42→    42→  if (i === -1) return display
    43→    43→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    44→    44→}

   378→   378→function TrainerStudents({ students, courses }: any) {

   380→   380→
   381→   381→  useEffect(() => {
   382→   382→    fetch('/api/enrollments').then((r) => r.json()).then((d) => {
   383→   383→      const all = Array.isArray(d) ? d : []
   384→   384→      // Filter to ONLY enrollments in this trainer's courses
   385→   385→      const trainerCourseIds = courses.map((c: any) => c.id)
   386→   386→      const mine = all.filter((e: any) => trainerCourseIds.includes(e.course_id))
   387→   387→      setEnrollments(mine)
   388→   388→    }).catch(() => {})
   389→   389→  }, [courses])
   390→   390→
   391→   391→  // Helper: get all enrollments for a student
   392→   392→  const studentEnrollments = (studentId: string) => enrollments.filter((e) => e.user_id === studentId)
   393→   393→
   394→   394→  return (
