    48→let fontsRegistered = false
    49→function registerFontsForCanvas() {
    50→  if (fontsRegistered) return
    51→  try {
    52→    const fonts = getFontPaths()
    53→    if (fs.existsSync(fonts.regular)) GlobalFonts.registerFromPath(fonts.regular, 'Cairo')
    54→    if (fs.existsSync(fonts.bold)) GlobalFonts.registerFromPath(fonts.bold, 'Cairo-Bold')
    55→    if (fs.existsSync(fonts.black)) GlobalFonts.registerFromPath(fonts.black, 'Cairo-Black')
    56→    fontsRegistered = true
    57→  } catch (e) {}
    58→}

    60→async function generateBarcode(certNumber: string): Promise<Buffer> {
    61→  try {
    62→    const bwip = await import('bwip-js')
    63→    return await (bwip.default || bwip).toBuffer({
    64→      bcid: 'code128', text: certNumber, scale: 2, height: 18,
    65→      includetext: false, paddingwidth: 3, paddingheight: 3,
    66→      backgroundcolor: 'FFFFFF', foregroundcolor: '0A1F44',
    67→    })
    68→  } catch (e: any) {
    69→    const canvas = createCanvas(200, 50)
    70→    const ctx = canvas.getContext('2d')
    71→    ctx.fillStyle = 'white'; ctx.fillRect(0, 0, 200, 50)
    72→    ctx.fillStyle = NAVY
    73→    let x = 5
    74→    for (let i = 0; i < 40; i++) {
    75→      const w = 1 + ((certNumber.charCodeAt(i % certNumber.length) * (i + 1)) % 4)
    76→      if (i % 2 === 0) ctx.fillRect(x, 5, w, 40)
    77→      x += w
    78→    }
    79→    return await canvas.encode('png')
    80→  }
    81→}

    91→// Premium geometric ornament: gold filigree with circles and curves
    92→function drawCornerOrnament(ctx: any, x: number, y: number, scale: number, mx: number, my: number) {
    93→  ctx.save()
    94→  ctx.translate(x, y)
    95→  ctx.scale(mx, my)
    96→  ctx.lineCap = 'round'
    97→  ctx.lineJoin = 'round'
    98→
    99→  // Outer filigree curve
   100→  ctx.strokeStyle = GOLD
   101→  ctx.lineWidth = 2 * scale
   102→  ctx.beginPath()
   103→  ctx.moveTo(0, 36 * scale)
   104→  ctx.bezierCurveTo(0, 12 * scale, 12 * scale, 0, 36 * scale, 0)
   105→  ctx.stroke()
   106→
   107→  // Inner curve
   108→  ctx.strokeStyle = GOLD_BRIGHT
   109→  ctx.lineWidth = 1.2 * scale
   110→  ctx.beginPath()
   111→  ctx.moveTo(6 * scale, 30 * scale)
   112→  ctx.bezierCurveTo(6 * scale, 16 * scale, 16 * scale, 6 * scale, 30 * scale, 6 * scale)
   113→  ctx.stroke()
   114→
   115→  // Decorative dots along the curve
   116→  ctx.fillStyle = GOLD
   117→  const dots = [[2, 32], [10, 22], [22, 10], [32, 2]]
   118→  dots.forEach(([dx, dy]) => {
   119→    ctx.beginPath()
   120→    ctx.arc(dx * scale, dy * scale, 1.5 * scale, 0, Math.PI * 2)
   121→    ctx.fill()
   122→  })
   123→
   124→  // Central decorative diamond at corner pivot
   125→  ctx.fillStyle = GOLD
   126→  ctx.beginPath()
   127→  ctx.moveTo(0, 0)
   128→  ctx.lineTo(4 * scale, -4 * scale)
   129→  ctx.lineTo(8 * scale, 0)
   130→  ctx.lineTo(4 * scale, 4 * scale)
   131→  ctx.closePath()
   132→  ctx.fill()
   133→
   134→  ctx.restore()
   135→}

   163→function drawGelPenSignature(ctx: any, startX: number, centerY: number, width: number, scale: number) {
   164→  const ink = SIG_BLUE
   165→  ctx.lineCap = 'round'; ctx.lineJoin = 'round'
   166→  const steps = 200
   167→  for (let i = 0; i < steps; i++) {
   168→    const t1 = i / steps, t2 = (i + 1) / steps
   169→    const x1 = startX + width * t1, x2 = startX + width * t2
   170→    const y1 = centerY + Math.sin(t1 * Math.PI * 5) * 10 * scale + Math.sin(t1 * Math.PI * 11) * 4 * scale + Math.cos(t1 * Math.PI * 3) * 7 * scale
   171→    const y2 = centerY + Math.sin(t2 * Math.PI * 5) * 10 * scale + Math.sin(t2 * Math.PI * 11) * 4 * scale + Math.cos(t2 * Math.PI * 3) * 7 * scale
   172→    const pressure = Math.abs(y2 - y1) > 0.15 ? 3.5 : 2
   173→    ctx.strokeStyle = ink; ctx.lineWidth = pressure * scale
   174→    ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.stroke()
   175→  }
   176→  // Small left flourish
   177→  ctx.strokeStyle = ink; ctx.lineWidth = 2.5 * scale
   178→  ctx.beginPath()
   179→  ctx.moveTo(startX - 8 * scale, centerY - 12 * scale)
   180→  ctx.bezierCurveTo(startX - 16 * scale, centerY - 28 * scale, startX + 4 * scale, centerY - 30 * scale, startX + 6 * scale, centerY - 14 * scale)
   181→  ctx.stroke()
   182→  // Small right end flourish
   183→  ctx.lineWidth = 3 * scale
   184→  ctx.beginPath()
   185→  ctx.moveTo(startX + width, centerY)
   186→  ctx.bezierCurveTo(startX + width + 16 * scale, centerY - 16 * scale, startX + width + 22 * scale, centerY + 4 * scale, startX + width + 8 * scale, centerY + 6 * scale)
   187→  ctx.stroke()
   188→  // Underscore
   189→  const usStartX = startX - 4 * scale, usEndX = startX + width + 10 * scale
   190→  const usY = centerY + 24 * scale
   191→  for (let i = 0; i < 100; i++) {
   192→    const t1 = i / 100, t2 = (i + 1) / 100
   193→    const x1 = usStartX + (usEndX - usStartX) * t1, x2 = usStartX + (usEndX - usStartX) * t2
   194→    const y1 = usY + Math.sin(t1 * Math.PI * 3) * 1.5 * scale, y2 = usY + Math.sin(t2 * Math.PI * 3) * 1.5 * scale
   195→    const pressure = t1 < 0.1 || t1 > 0.9 ? 1.2 : 2
   196→    ctx.lineWidth = pressure * scale
   197→    ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.stroke()
   198→  }
   199→  // Initial dot
   200→  ctx.fillStyle = ink
   201→  ctx.beginPath()
   202→  ctx.arc(startX - 14 * scale, centerY - 8 * scale, 1.8 * scale, 0, Math.PI * 2)
   203→  ctx.fill()
   204→}

   241→async function generateCertificatePNG(cert: any, issueDate: string, verifyUrl: string, founderSignature?: string | null, signatoryName?: string | null, signatoryPosition?: string | null): Promise<Buffer> {
   242→  registerFontsForCanvas()
   243→
   244→  const scale = 2
   245→  const pageW = 842 * scale, pageH = 595 * scale
   246→  const canvas = createCanvas(pageW, pageH)
   247→  const ctx = canvas.getContext('2d')
   248→
   249→  // Helpers
   250→  const fill = (c: string) => { ctx.fillStyle = c }
   251→  const stroke = (c: string, w: number) => { ctx.strokeStyle = c; ctx.lineWidth = w }
   252→  const roundRect = (x: number, y: number, w: number, h: number, r: number) => {
   253→    ctx.beginPath()
   254→    ctx.moveTo(x + r, y)
   255→    ctx.lineTo(x + w - r, y)
   256→    ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   257→    ctx.lineTo(x + w, y + h - r)
   258→    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   259→    ctx.lineTo(x + r, y + h)
   260→    ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   261→    ctx.lineTo(x, y + r)
   262→    ctx.quadraticCurveTo(x, y, x + r, y)
   263→    ctx.closePath()
   264→  }
   265→  const fillRoundRect = (x: number, y: number, w: number, h: number, r: number, c: string) => {
   266→    fill(c); roundRect(x, y, w, h, r); ctx.fill()
   267→  }
   268→  const strokeRoundRect = (x: number, y: number, w: number, h: number, r: number, c: string, lw: number) => {
   269→    stroke(c, lw); roundRect(x, y, w, h, r); ctx.stroke()
   270→  }
   271→  const setFont = (size: number, weight: 'regular' | 'bold' | 'black' = 'regular') => {
   272→    const family = weight === 'black' ? 'Cairo-Black' : weight === 'bold' ? 'Cairo-Bold' : 'Cairo'
   273→    ctx.font = `${size * scale}px ${family}`
   274→  }
   275→
   276→  const cx = pageW / 2
   277→
   278→  // ===== OUTER BACKGROUND: Deep navy with subtle radial gradient =====
   279→  fill(NAVY_DEEP); ctx.fillRect(0, 0, pageW, pageH)
   280→  const bgGrad = ctx.createRadialGradient(cx, pageH / 2, 100 * scale, cx, pageH / 2, pageW)
   281→  bgGrad.addColorStop(0, NAVY)
   282→  bgGrad.addColorStop(1, NAVY_DEEP)
   283→  ctx.fillStyle = bgGrad; ctx.fillRect(0, 0, pageW, pageH)
   284→
   285→  // ===== Subtle geometric pattern on outer navy (very faint diamonds) =====
   286→  ctx.save()
   287→  ctx.globalAlpha = 0.06
   288→  fill(GOLD)
   289→  for (let row = 0; row < 8; row++) {
   290→    for (let col = 0; col < 12; col++) {
   291→      const x = col * 80 * scale + (row % 2 ? 40 * scale : 0)
   292→      const y = row * 80 * scale
   293→      ctx.save()
   294→      ctx.translate(x, y)
   295→      ctx.rotate(Math.PI / 4)
   296→      ctx.fillRect(-2 * scale, -2 * scale, 4 * scale, 4 * scale)
   297→      ctx.restore()
   298→    }
   299→  }
   300→  ctx.restore()
   301→
   302→  // ===== INNER IVORY CARD (with shadow) =====
   303→  const cardPad = 28 * scale
   304→  const cardX = cardPad, cardY = cardPad
   305→  const cardW = pageW - 2 * cardPad, cardH = pageH - 2 * cardPad
   306→
   307→  ctx.save()
   308→  ctx.shadowColor = 'rgba(0,0,0,0.5)'
   309→  ctx.shadowBlur = 20 * scale
   310→  ctx.shadowOffsetY = 6 * scale
   311→  fillRoundRect(cardX, cardY, cardW, cardH, 16 * scale, IVORY)
   312→  ctx.restore()
   313→
   314→  // Subtle inner gradient on cream
   315→  const innerGrad = ctx.createLinearGradient(0, cardY, 0, cardY + cardH)
   316→  innerGrad.addColorStop(0, CREAM)
   317→  innerGrad.addColorStop(1, IVORY)
   318→  fillRoundRect(cardX, cardY, cardW, cardH, 16 * scale, '#000')
   319→  ctx.fillStyle = innerGrad
   320→  roundRect(cardX, cardY, cardW, cardH, 16 * scale)
   321→  ctx.fill()
   322→
   323→  // ===== TRIPLE GOLD FRAME (premium multi-line border) =====
   324→  strokeRoundRect(cardX + 8 * scale, cardY + 8 * scale, cardW - 16 * scale, cardH - 16 * scale, 12 * scale, GOLD, 2.5 * scale)

   332→  drawCornerOrnament(ctx, cardX + cornerInset, cardY + cardH - cornerInset, scale, 1, -1)

   836→        doc.fill(boxBgColor)

   841→        doc.save()
   842→        roundRectPath(x + iconPad, boxesY + (boxesH - iconSize) / 2, iconSize, iconSize, iconSize / 2)
   843→        doc.fill(boxBorderColor === GOLD ? NAVY : boxBorderColor)
   844→        doc.restore()

   846→        doc.fillColor(GOLD_BRIGHT).text(icon, x + iconPad, boxesY + (boxesH - iconSize) / 2 + 5, { width: iconSize, align: 'center' })
   847→        // Label + value
   848→        setFont(8, 'regular')
   849→        doc.fillColor(GRAY).text(label, x + iconPad + iconSize + 5, boxesY + 11, { width: boxW - iconPad - iconSize - 18, align: 'right' })

   851→        doc.fillColor(valueColor).text(value, x + iconPad + iconSize + 5, boxesY + 26, { width: boxW - iconPad - iconSize - 18, align: 'right' })

   853→
   854→      drawInfoBoxPDF(boxesStartX, '📅', 'تاريخ الإصدار', issueDate, NAVY)
   855→      const box2X = boxesStartX + boxW + boxGap
   856→      drawInfoBoxPDF(box2X, '🎓', 'الجهة المصدرة', 'أكاديمية المستقبل الدولية', NAVY)

   859→      const statusColor = isRevoked ? CRIMSON : EMERALD
   860→      const statusBgColor = isRevoked ? CRIMSON_LIGHT : EMERALD_LIGHT
   861→      const statusText = isRevoked ? 'ملغي' : 'صالح'
   862→      drawInfoBoxPDF(box3X, isRevoked ? '✕' : '✓', 'الحالة', statusText, statusColor, statusBgColor, statusColor)

   865→      const innerLeft = cardX + 32, innerRight = cardX + cardW - 32
   866→      const innerW = innerRight - innerLeft

   868→      const bottomY = cardY + 478
   869→

   871→      const qrBuffer = await generateQR(verifyUrl)
   872→      const qrSize = 38
   873→      const qrColCenterX = innerLeft + colW / 2 - colGap / 2
   874→      const qrY = bottomY + 2
   875→      doc.save()

   877→      doc.fill(WHITE)
   878→      doc.lineWidth(1.2).strokeColor(GOLD)
   879→      doc.stroke()
   880→      doc.restore()
   881→      doc.image(qrBuffer, qrColCenterX - qrSize / 2, qrY, { width: qrSize, height: qrSize })

   883→      doc.fillColor(NAVY).text('امسح للتحقق', qrColCenterX - 30, qrY + qrSize + 2, { width: 60, align: 'center' })
   884→

   886→      const certColCenterX = innerLeft + colW + colGap / 2 + colW / 2 - colGap / 2

   889→      const barcodeX = certColCenterX - barcodeW / 2
   890→      const barcodeY = bottomY + 4
   891→      const barcodeCardH = barcodeH + 28
   892→      doc.save()
   893→      roundRectPath(barcodeX - 6, barcodeY - 6, barcodeW + 12, barcodeCardH, 4)
   894→      doc.fill(WHITE)
   895→      doc.lineWidth(1.2).strokeColor(GOLD)
   896→      doc.stroke()

   898→      doc.image(barcodeBuffer, barcodeX, barcodeY, { width: barcodeW, height: barcodeH })
   899→      setFont(9, 'bold')

   902→      doc.fillColor(GRAY).text('CERTIFICATE ID', certColCenterX - 60, barcodeY + barcodeH + 12, { width: 120, align: 'center' })

   905→      const sigColCenterX = innerLeft + 2 * colW + colGap + colW / 2 - colGap / 2
   906→      const sigAreaW = colW - 20
   907→      const sigAreaX = sigColCenterX - sigAreaW / 2

   918→          fs.writeFileSync(sigPath, Buffer.from(sigBuf))
   919→          if (fs.existsSync(sigPath)) {

   930→      doc.moveTo(sigAreaX + 5, bottomY + 42).lineTo(sigAreaX + sigAreaW - 5, bottomY + 42).stroke()

   932→      // Signatory name + position — ONLY shown if signature is approved

   937→      if (signatureDrawn && signatoryPosition) {

   939→        doc.fillColor(GRAY_DARK).text(signatoryPosition, sigAreaX, bottomY + 58, { width: sigAreaW, align: 'center' })
   940→      }
   941→
   942→      // WATERMARK REMOVED (was interfering with text readability)
   943→
   944→      doc.end()
   945→    } catch (e: any) {
   946→      reject(e)
   947→    }
   948→  })
   949→}

   951→export async function GET(req: NextRequest) {
   952→  try {
   953→    const certNum = req.nextUrl.searchParams.get('number')
   954→    const certId = req.nextUrl.searchParams.get('id')
   955→    const format = req.nextUrl.searchParams.get('format') || 'pdf'
   956→
   957→    if (!certNum && !certId) {
   958→      return NextResponse.json({ error: 'رقم الشهادة أو المعرّف مطلوب' }, { status: 400 })
   959→    }
   960→
   961→    let query = supabaseAdmin.from('certificates').select('*')
   962→    if (certNum) query = query.eq('certificate_number', certNum)
   963→    else if (certId) query = query.eq('id', certId)
   964→
   965→    const { data: cert, error } = await query.single()
   966→    if (error || !cert) {
   967→      return NextResponse.json({ error: 'الشهادة غير موجودة' }, { status: 404 })
   968→    }
   969→
   970→    const issueDate = cert.issued_at
   971→      ? new Date(cert.issued_at).toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric' })
   972→      : new Date().toLocaleDateString('ar-EG')
   973→    const verifyUrl = `https://fiaacademy.vercel.app/?verify=${encodeURIComponent(cert.certificate_number)}#verify`
   974→
   975→    // Fetch founder's signature, signatory name & position from founder bio (added via admin dashboard)
   976→    // The signature is OPTIONAL — area remains EMPTY if no signature has been added/approved by admin
   977→    let founderSignature: string | null = null
   978→    let signatoryName: string | null = null
   979→    let signatoryPosition: string | null = null
   980→    try {
   981→      const { data: trainers } = await supabaseAdmin.from('trainers').select('*')
   982→      if (trainers) {
   983→        const founder = trainers.find((t: any) => t.password && t.password.startsWith('__TEAM_MEMBER__:founder'))
   984→        if (founder) {
   985→          const bioSeparator = '\n\n===FIA_EXTRA===\n'
   986→          const idx = founder.bio ? founder.bio.indexOf(bioSeparator) : -1
   987→          if (idx !== -1) {
   988→            const extraJson = founder.bio.substring(idx + bioSeparator.length)
   989→            const extra = JSON.parse(extraJson)
   990→            founderSignature = extra.signature || null
   991→            signatoryName = extra.signatory_name || founder.name || null
   992→            signatoryPosition = extra.signatory_position || 'المدير التنفيذي'
   993→          }
   994→        }
   995→      }
   996→    } catch {}
   997→
   998→    if (format === 'png') {
   999→      const pngBytes = await generateCertificatePNG(cert, issueDate, verifyUrl, founderSignature, signatoryName, signatoryPosition)
  1000→      const headers = new Headers()
  1001→      headers.set('Content-Type', 'image/png')
  1002→      headers.set('Content-Disposition', `inline; filename="FIA-Certificate-${cert.certificate_number}.png"`)
  1003→      headers.set('Cache-Control', 'public, max-age=3600')
  1004→      return new NextResponse(pngBytes, { status: 200, headers })
  1005→    } else {
  1006→      const pdfBytes = await generateCertificatePDF(cert, issueDate, verifyUrl, founderSignature, signatoryName, signatoryPosition)
  1007→      const headers = new Headers()
  1008→      headers.set('Content-Type', 'application/pdf')
  1009→      headers.set('Content-Disposition', `inline; filename="FIA-Certificate-${cert.certificate_number}.pdf"`)
  1010→      headers.set('Cache-Control', 'public, max-age=3600')
  1011→      return new NextResponse(pdfBytes, { status: 200, headers })
  1012→    }
  1013→  } catch (e: any) {
  1014→    return NextResponse.json({ error: e.message }, { status: 500 })
  1015→  }
  1016→}
