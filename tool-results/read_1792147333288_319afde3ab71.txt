    39→    39→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    40→    40→// Show only the human-readable text in the UI, but preserve the extra data on save.
    41→    41→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    42→    42→const splitBio = (raw: string) => {
    43→    43→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    44→    44→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    45→    45→}
    46→    46→const joinBio = (display: string, original?: string) => {
    47→    47→  if (!original) return display
    48→    48→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    49→    49→  if (i === -1) return display
    50→    50→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    51→    51→}

   285→   285→function BrowseCourses({ courses, enrollments, userId, onEnrolled }: any) {

   287→   287→  const [payDialog, setPayDialog] = useState<{ course: any; open: boolean } | null>(null)
   288→   288→  const [enrolling, setEnrolling] = useState<string | null>(null)
   289→   289→
   290→   290→  useEffect(() => {
   291→   291→    fetch('/api/payment-methods').then((r) => r.json()).then((d) => setPaymentMethods(Array.isArray(d) ? d.filter((p:any)=>p.enabled) : [])).catch(() => {})
   292→   292→  }, [])
   293→   293→
   294→   294→  const isEnrolled = (courseId: string) => enrollments?.some((e: any) => e.course_id === courseId)
   295→   295→
   296→   296→  const enrollFree = async (course: any) => {
   297→   297→    // Double-check client-side before calling API
   298→   298→    if (isEnrolled(course.id)) {
   299→   299→      toast.info('أنت مسجّل في هذه الدورة')
   300→   300→      return
   301→   301→    }
   302→   302→    setEnrolling(course.id)
   303→   303→    try {
   304→   304→      const res = await fetch('/api/enrollments', {
   305→   305→        method: 'POST',
   306→   306→        headers: { 'Content-Type': 'application/json' },
   307→   307→        body: JSON.stringify({ user_id: userId, course_id: course.id, payment_status: 'free' }),
   308→   308→      })
   309→   309→      const data = await res.json()
   310→   310→      if (data.already_enrolled) {
   311→   311→        toast.info('أنت مسجّل في هذه الدورة')
   312→   312→      } else if (res.ok) {
   313→   313→        toast.success(`تم التسجيل في دورة: ${course.title}`)
   314→   314→        onEnrolled()
   315→   315→      } else toast.error('فشل التسجيل')
   316→   316→    } catch {
   317→   317→      toast.error('خطأ')
   318→   318→    } finally {
   319→   319→      setEnrolling(null)
   320→   320→    }
   321→   321→  }
   322→   322→
   323→   323→  const enrollPaid = async (course: any, method: any, reference: string, proofUrl?: string | null) => {
   324→   324→    // Double-check client-side before calling API
   325→   325→    if (isEnrolled(course.id)) {
   326→   326→      toast.info('أنت مسجّل في هذه الدورة')
   327→   327→      return
   328→   328→    }
   329→   329→    setEnrolling(course.id)
   330→   330→    try {
   331→   331→      // Combine reference with proof URL
   332→   332→      const fullReference = proofUrl ? `${reference} | PROOF: ${proofUrl}` : reference
   333→   333→
   334→   334→      // 1. Create enrollment with pending payment
   335→   335→      const res = await fetch('/api/enrollments', {
   336→   336→        method: 'POST',
   337→   337→        headers: { 'Content-Type': 'application/json' },
   338→   338→        body: JSON.stringify({
   339→   339→          user_id: userId,
   340→   340→          course_id: course.id,
   341→   341→          payment_status: 'pending',
   342→   342→          amount_paid: course.price,
   343→   343→          payment_method: method.name,
   344→   344→          payment_reference: fullReference,
   345→   345→        }),
   346→   346→      })
   347→   347→      const data = await res.json()
   348→   348→      if (data.already_enrolled) {
   349→   349→        toast.info('أنت مسجّل في هذه الدورة')
   350→   350→      } else if (res.ok) {
   351→   351→        // 2. Create payment record with proof URL
   352→   352→        await fetch('/api/payments', {
   353→   353→          method: 'POST',
   354→   354→          headers: { 'Content-Type': 'application/json' },
   355→   355→          body: JSON.stringify({
   356→   356→            amount: course.price,
   357→   357→            currency: course.currency || 'ر.ي',
   358→   358→            method: method.name,
   359→   359→            status: 'قيد الانتظار',
   360→   360→            reference: fullReference,
   361→   361→            user_id: userId,
   362→   362→            course_id: course.id,
   363→   363→          }),
   364→   364→        })
   365→   365→        toast.success('تم تسجيل طلبك! سيتم تفعيل الدورة بعد تأكيد الدفع')
   366→   366→        onEnrolled()
   367→   367→        setPayDialog(null)
   368→   368→      } else toast.error('فشل التسجيل')
   369→   369→    } catch {
   370→   370→      toast.error('خطأ')
   371→   371→    } finally {
   372→   372→      setEnrolling(null)
   373→   373→    }
   374→   374→  }
   375→   375→
   376→   376→  return (
