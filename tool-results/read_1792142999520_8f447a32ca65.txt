    33→    33→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    34→    34→// Show only the human-readable text in the UI, but preserve the extra data on save.
    35→    35→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    36→    36→const splitBio = (raw: string) => {
    37→    37→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    38→    38→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    39→    39→}
    40→    40→const joinBio = (display: string, original?: string) => {
    41→    41→  if (!original) return display
    42→    42→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    43→    43→  if (i === -1) return display
    44→    44→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    45→    45→}

    47→    47→export function TrainerDashboard() {
    48→    48→  const [active, setActive] = useState('overview')
    49→    49→  const [contentCourseId, setContentCourseId] = useState<string | null>(null)
    50→    50→  const user = useAuth((s) => s.user)
    51→    51→  const updateUser = useAuth((s) => s.updateUser)
    52→    52→  const logout = useAuth((s) => s.logout)
    53→    53→  const [courses, setCourses] = useState<any[]>([])
    54→    54→  const [students, setStudents] = useState<any[]>([])
    55→    55→  const [editing, setEditing] = useState<any | null>(null)
    56→    56→  const [isDialogOpen, setIsDialogOpen] = useState(false)
    57→    57→  const [profile, setProfile] = useState({ name: user?.name || '', phone: user?.phone || '', title: user?.title || '', bio: splitBio(user?.bio || ''), specialization: user?.title || '' })
    58→    58→
    59→    59→  const loadCourses = () => {
    60→    60→    fetch('/api/courses').then((r) => r.json()).then((d) => {
    61→    61→      const all = Array.isArray(d) ? d : []
    62→    62→      // Filter to ONLY this trainer's courses
    63→    63→      const mine = all.filter((c: any) => c.trainer_id === user?.id)
    64→    64→      setCourses(mine)
    65→    65→    }).catch(() => {})
    66→    66→  }

    68→    68→    fetch('/api/students?role=student').then((r) => r.json()).then((d) => setStudents(Array.isArray(d) ? d : [])).catch(() => {})
    69→    69→  }
    70→    70→
    71→    71→  useEffect(() => {
    72→    72→    loadCourses()
    73→    73→    loadStudents()
    74→    74→  }, [user?.id])
    75→    75→
    76→    76→  const navigateToContent = (courseId: string) => {

    78→    78→    setActive('content')
    79→    79→  }

    81→    81→  return (
    82→    82→    <DashboardShell
    83→    83→      nav={NAV}
    84→    84→      active={active}
    85→    85→      onNavigate={(id) => { setActive(id); if (id !== 'content') setContentCourseId(null) }}
    86→    86→      roleLabel="مدرب"

    91→    91→      {active === 'content' && contentCourseId && (
    92→    92→        <div className="space-y-4">
    93→    93→          <Button variant="ghost" onClick={() => setActive('courses')} className="text-navy hover:bg-navy/5">
    94→    94→            <ArrowRight className="w-4 h-4 ml-1" /> العودة للدورات
    95→    95→          </Button>
    96→    96→          <CourseContentManager courseId={contentCourseId} />
    97→    97→        </div>
    98→    98→      )}
    99→    99→      {active === 'students' && <TrainerStudents students={students} courses={courses} />}
   100→   100→      {active === 'exams' && <TrainerExams courses={courses} />}
   101→   101→      {active === 'profile' && <TrainerProfile profile={profile} setProfile={setProfile} user={user} updateUser={updateUser} />}
   102→   102→      {active === 'notifications' && <TrainerNotifs courses={courses} />}
   103→   103→      {active === 'settings' && <TrainerAccountSettings user={user} updateUser={updateUser} logout={logout} />}
   104→   104→
   105→   105→      {isDialogOpen && (
   106→   106→        <CourseDialog
   107→   107→          open={isDialogOpen}
   108→   108→          onOpenChange={setIsDialogOpen}
   109→   109→          course={editing}
   110→   110→          trainerId={user?.id}
   111→   111→          onSaved={() => {
   112→   112→            setIsDialogOpen(false)
   113→   113→            loadCourses()
   114→   114→          }}
   115→   115→        />
   116→   116→      )}

   155→   155→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   156→   156→  const [deleting, setDeleting] = useState<string | null>(null)
   157→   157→
   158→   158→  const handleDelete = async (id: string) => {
   159→   159→    if (!confirm('هل أنت متأكد من حذف هذه الدورة؟')) return
   160→   160→    setDeleting(id)
   161→   161→    try {
   162→   162→      const res = await fetch('/api/courses', {
   163→   163→        method: 'DELETE',
   164→   164→        headers: { 'Content-Type': 'application/json' },
   165→   165→        body: JSON.stringify({ id }),
   166→   166→      })
   167→   167→      if (res.ok) {
   168→   168→        toast.success('تم حذف الدورة')
   169→   169→        onRefresh()
   170→   170→      } else {
   171→   171→        toast.error('فشل الحذف')
   172→   172→      }
   173→   173→    } catch {
   174→   174→      toast.error('خطأ في الاتصال')
   175→   175→    } finally {
   176→   176→      setDeleting(null)
   177→   177→    }
   178→   178→  }
   179→   179→
   180→   180→  return (
   181→   181→    <SectionCard
   182→   182→      title="دوراتي التدريبية"
   183→   183→      description="إدارة الدورات التي تدرّسها"
   184→   184→      action={
   185→   185→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   186→   186→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   187→   187→        </Button>
   188→   188→      }
   189→   189→    >
   190→   190→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   191→   191→        {courses.map((c: any) => (
   192→   192→          <Card key={c.id} className="overflow-hidden card-hover">
   193→   193→            <div className="h-32 bg-navy relative">
   194→   194→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   195→   195→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   196→   196→            </div>
   197→   197→            <div className="p-4">
   198→   198→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   199→   199→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   200→   200→                <span>{c.enrolled || 0} متدرب</span>
   201→   201→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   202→   202→              </div>
   203→   203→              <div className="space-y-2">
   204→   204→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   205→   205→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   206→   206→                </Button>
   207→   207→                <div className="flex gap-2">
   208→   208→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   209→   209→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   210→   210→                  </Button>
   211→   211→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => handleDelete(c.id)} disabled={deleting === c.id}>
   212→   212→                    <Trash2 className="w-3.5 h-3.5" />
   213→   213→                  </Button>
   214→   214→                </div>
   215→   215→              </div>
   216→   216→            </div>
   217→   217→          </Card>
   218→   218→        ))}
   219→   219→      </div>
   220→   220→    </SectionCard>

   224→   224→function CourseDialog({ open, onOpenChange, course, trainerId, onSaved }: any) {
   225→   225→  const [form, setForm] = useState<any>(
   226→   226→    course || {
   227→   227→      title: '',
   228→   228→      short_description: '',
   229→   229→      description: '',
   230→   230→      level: 'متوسط',
   231→   231→      price_type: 'مجانية',
   232→   232→      price: 0,
   233→   233→      currency: 'ر.ي',
   234→   234→      duration_hours: 0,
   235→   235→      pass_score: 60,
   236→   236→      cover: '',
   237→   237→      status: 'منشورة',
   238→   238→    }
   239→   239→  )
   240→   240→  const [saving, setSaving] = useState(false)
   241→   241→  const [uploadingCover, setUploadingCover] = useState(false)
   242→   242→
   243→   243→  // Keep the form in sync with the `course` prop whenever the dialog opens or
   244→   244→  // the course being edited changes. Without this, a dialog that stays mounted
   245→   245→  // across course switches would show stale state (and could overwrite the
   246→   246→  // wrong course's `cover` field).
   247→   247→  useEffect(() => {
   248→   248→    if (open) {
   249→   249→      setForm(
   250→   250→        course || {
   251→   251→          title: '',
   252→   252→          short_description: '',
   253→   253→          description: '',
   254→   254→          level: 'متوسط',
   255→   255→          price_type: 'مجانية',
   256→   256→          price: 0,
   257→   257→          currency: 'ر.ي',
   258→   258→          duration_hours: 0,
   259→   259→          pass_score: 60,
   260→   260→          cover: '',
   261→   261→          status: 'منشورة',
   262→   262→        }
   263→   263→      )
   264→   264→    }
   265→   265→  }, [open, course])
   266→   266→
   267→   267→  const handleUploadCover = async (file: File) => {
   268→   268→    setUploadingCover(true)
   269→   269→    try {
   270→   270→      const fd = new FormData()
   271→   271→      fd.append('file', file)
   272→   272→      fd.append('folder', 'courses')
   273→   273→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   274→   274→      const data = await res.json().catch(() => ({}))
   275→   275→      if (data.url) {
   276→   276→        // Store the Telegram-hosted URL on form.cover — it is later persisted
   277→   277→        // to the courses.cover column by the save() call below.
   278→   278→        setForm((prev: any) => ({ ...prev, cover: data.url }))
   279→   279→        toast.success('تم رفع صورة الغلاف')
   280→   280→      } else {
   281→   281→        toast.error('فشل رفع الصورة: ' + (data.error || ''))
   282→   282→      }
   283→   283→    } catch {
   284→   284→      toast.error('خطأ في الاتصال')
   285→   285→    } finally {
   286→   286→      setUploadingCover(false)
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  const save = async () => {
   291→   291→    if (!form.title) {
   292→   292→      toast.error('أدخل عنوان الدورة')
   293→   293→      return
   294→   294→    }
   295→   295→    setSaving(true)
   296→   296→    try {
   297→   297→      const slug = form.title.toLowerCase().replace(/\s+/g, '-').slice(0, 80)
   298→   298→      // Build the payload. Always include `cover` (string URL or null) so the
   299→   299→      // courses.cover column is updated on every save — even when the trainer
   300→   300→      // cleared the field.
   301→   301→      const body: any = {
   302→   302→        title: form.title,
   303→   303→        short_description: form.short_description ?? '',
   304→   304→        description: form.description ?? '',
   305→   305→        level: form.level,
   306→   306→        price_type: form.price_type,
   307→   307→        price: Number(form.price) || 0,
   308→   308→        currency: form.currency || 'ر.ي',
   309→   309→        duration_hours: Number(form.duration_hours) || 0,
   310→   310→        pass_score: Number(form.pass_score) || 60,
   311→   311→        status: form.status || 'منشورة',
   312→   312→        cover: form.cover ?? null,
   313→   313→        slug,
   314→   314→        trainer_id: trainerId,
   315→   315→        modules_count: form.modules_count || 0,
   316→   316→        lessons_count: form.lessons_count || 0,
   317→   317→      }
   318→   318→      const res = await fetch('/api/courses', {
   319→   319→        method: course ? 'PUT' : 'POST',
   320→   320→        headers: { 'Content-Type': 'application/json' },
   321→   321→        body: JSON.stringify(course ? { id: course.id, ...body } : body),
   322→   322→      })
   323→   323→      if (res.ok) {
   324→   324→        toast.success(course ? 'تم تحديث الدورة' : 'تم إنشاء الدورة')
   325→   325→        onSaved()
   326→   326→      } else {
   327→   327→        const err = await res.json().catch(() => ({}))
   328→   328→        toast.error(err.error || 'فشل الحفظ')
   329→   329→      }
   330→   330→    } catch {
   331→   331→      toast.error('خطأ')
   332→   332→    } finally {
   333→   333→      setSaving(false)
   334→   334→    }
   335→   335→  }
   336→   336→
   337→   337→  return (
   338→   338→    <Dialog open={open} onOpenChange={onOpenChange}>
   339→   339→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   340→   340→        <DialogHeader>
   341→   341→          <DialogTitle>{course ? 'تعديل الدورة' : 'دورة جديدة'}</DialogTitle>
   342→   342→        </DialogHeader>
   343→   343→        <div className="space-y-4 py-4">
   344→   344→          <div>
   345→   345→            <Label>عنوان الدورة</Label>
   346→   346→            <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} className="mt-1" />
   347→   347→          </div>
   348→   348→          <div>
   349→   349→            <Label>وصف مختصر</Label>
   350→   350→            <Input value={form.short_description || ''} onChange={(e) => setForm({ ...form, short_description: e.target.value })} className="mt-1" />
   351→   351→          </div>
   352→   352→          <div>
   353→   353→            <Label>الوصف الكامل</Label>
   354→   354→            <Textarea value={form.description || ''} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} className="mt-1" />
   355→   355→          </div>
   356→   356→          <div className="grid grid-cols-2 gap-3">
   357→   357→            <div>
   358→   358→              <Label>المستوى</Label>
   359→   359→              <select
   360→   360→                value={form.level}
   361→   361→                onChange={(e) => setForm({ ...form, level: e.target.value })}
   362→   362→                className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background"
   363→   363→              >
   364→   364→                <option value="مبتدئ">مبتدئ</option>
   365→   365→                <option value="متوسط">متوسط</option>
   366→   366→                <option value="متقدم">متقدم</option>
   367→   367→              </select>
   368→   368→            </div>
   369→   369→            <div>
   370→   370→              <Label>السعر</Label>
   371→   371→              <select
   372→   372→                value={form.price_type}
   373→   373→                onChange={(e) => setForm({ ...form, price_type: e.target.value })}
   374→   374→                className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background"
   375→   375→              >
   376→   376→                <option value="مجانية">مجانية</option>
   377→   377→                <option value="مدفوعة">مدفوعة</option>
   378→   378→              </select>
   379→   379→            </div>
   380→   380→          </div>
   381→   381→          {form.price_type === 'مدفوعة' && (
   382→   382→            <div>
   383→   383→              <Label>المبلغ</Label>
   384→   384→              <Input type="number" value={form.price || 0} onChange={(e) => setForm({ ...form, price: parseInt(e.target.value) || 0 })} className="mt-1" dir="ltr" />
   385→   385→            </div>
   386→   386→          )}
   387→   387→          <div className="grid grid-cols-2 gap-3">
   388→   388→            <div>
   389→   389→              <Label>عدد الساعات</Label>
   390→   390→              <Input type="number" value={form.duration_hours || 0} onChange={(e) => setForm({ ...form, duration_hours: parseInt(e.target.value) || 0 })} className="mt-1" dir="ltr" />
   391→   391→            </div>
   392→   392→            <div>
   393→   393→              <Label>درجة النجاح (%)</Label>
   394→   394→              <Input type="number" value={form.pass_score || 60} onChange={(e) => setForm({ ...form, pass_score: parseInt(e.target.value) || 60 })} className="mt-1" dir="ltr" />
   395→   395→            </div>
   396→   396→          </div>
   397→   397→          <div>
   398→   398→            <Label>صورة الغلاف</Label>

   939→   939→function TrainerNotifs({ courses }: { courses: any[] }) {

   946→   946→      if (!courses || courses.length === 0) {
   947→   947→        setNotifs([])

   950→   950→      }
   951→   951→      setLoading(true)
   952→   952→      try {
   953→   953→        const courseIds = courses.map((c) => c.id)
   954→   954→        const courseById = new Map(courses.map((c) => [c.id, c]))
   955→   955→
   956→   956→        // Fetch enrollments for all courses (we'll filter on the client)
   957→   957→        const [enrRes, stuRes] = await Promise.all([
   958→   958→          fetch('/api/enrollments'),

   960→   960→        ])

   968→   968→

   998→   998→        if (!cancelled) setNotifs(built)
   999→   999→      } catch {
  1000→  1000→        if (!cancelled) setNotifs([])
  1001→  1001→      } finally {
  1002→  1002→        if (!cancelled) setLoading(false)
  1003→  1003→      }
  1004→  1004→    }
  1005→  1005→    loadNotifs()
  1006→  1006→    return () => { cancelled = true }
  1007→  1007→  }, [courses])
