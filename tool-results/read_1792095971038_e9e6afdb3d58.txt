    31→function getFontPaths() {
    32→  const candidates = [
    33→    path.join(process.cwd(), '.fonts'),
    34→    '/home/z/my-project/.fonts',
    35→  ].filter(Boolean)
    36→  const fontsDir = candidates.find((p) => fs.existsSync(p)) || candidates[0]
    37→  return {
    38→    regular: path.join(fontsDir, 'Cairo-Regular.ttf'),
    39→    bold: path.join(fontsDir, 'Cairo-Bold.ttf'),
    40→    black: path.join(fontsDir, 'Cairo-Black.ttf'),
    41→  }

    44→let fontsRegistered = false
    45→function registerFontsForCanvas() {
    46→  if (fontsRegistered) return
    47→  try {
    48→    const fonts = getFontPaths()
    49→    if (fs.existsSync(fonts.regular)) GlobalFonts.registerFromPath(fonts.regular, 'Cairo')
    50→    if (fs.existsSync(fonts.bold)) GlobalFonts.registerFromPath(fonts.bold, 'Cairo-Bold')
    51→    if (fs.existsSync(fonts.black)) GlobalFonts.registerFromPath(fonts.black, 'Cairo-Black')
    52→    fontsRegistered = true
    53→  } catch (e) {}
    54→}

    56→async function generateBarcode(certNumber: string): Promise<Buffer> {
    57→  try {
    58→    return await bwip.toBuffer({
    59→      bcid: 'code128',
    60→      text: certNumber,
    61→      scale: 2,
    62→      height: 20,
    63→      includetext: false,
    64→      paddingwidth: 3,
    65→      paddingheight: 3,
    66→      backgroundcolor: 'FFFFFF',
    67→      foregroundcolor: '0F172A',
    68→    })
    69→  } catch (e: any) {
    70→    const canvas = createCanvas(200, 50)
    71→    const ctx = canvas.getContext('2d')
    72→    ctx.fillStyle = 'white'
    73→    ctx.fillRect(0, 0, 200, 50)
    74→    ctx.fillStyle = NAVY
    75→    let x = 5
    76→    for (let i = 0; i < 40; i++) {
    77→      const w = 1 + ((certNumber.charCodeAt(i % certNumber.length) * (i + 1)) % 4)
    78→      if (i % 2 === 0) ctx.fillRect(x, 5, w, 40)
    79→      x += w
    80→    }
    81→    return await canvas.encode('png')
    82→  }
    83→}

   105→  // Helpers
   106→  const fill = (c: string) => { ctx.fillStyle = c }
   107→  const stroke = (c: string, w: number) => { ctx.strokeStyle = c; ctx.lineWidth = w }
   108→  const roundRect = (x: number, y: number, w: number, h: number, r: number) => {
   109→    ctx.beginPath()
   110→    ctx.moveTo(x + r, y)
   111→    ctx.lineTo(x + w - r, y)
   112→    ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   113→    ctx.lineTo(x + w, y + h - r)
   114→    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   115→    ctx.lineTo(x + r, y + h)
   116→    ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   117→    ctx.lineTo(x, y + r)

   397→  setFont(140, 'black')
   398→  fill(NAVY)
   399→  ctx.textAlign = 'center'
   400→  ctx.textBaseline = 'middle'
   401→  ctx.fillText('FIA', 0, 0)

   403→
   404→  return await canvas.encode('png')
   405→}

   407→// ============== PDF Certificate (matching PNG design) ==============
   408→async function generateCertificatePDF(cert: any, issueDate: string, verifyUrl: string, founderSignature?: string | null): Promise<Buffer> {
   409→  return new Promise(async (resolve, reject) => {
   410→    try {
   411→      const fonts = getFontPaths()
   412→      const doc = new PDFDocument({
   413→        size: [842, 595],
   414→        margins: { top: 0, bottom: 0, left: 0, right: 0 },
   415→        info: {
   416→          Title: `FIA Certificate — ${cert.certificate_number}`,
   417→          Author: 'FIA — Future International Academy',
   418→          Subject: 'Verified Professional Certificate',
   419→        },
   420→      })
   421→
   422→      const fontExists = (weight: 'regular' | 'bold' | 'black' = 'regular') => {
   423→        const p = weight === 'black' ? fonts.black : weight === 'bold' ? fonts.bold : fonts.regular
   424→        return fs.existsSync(p) ? p : undefined
   425→      }
   426→      const setFont = (size: number, weight: 'regular' | 'bold' | 'black' = 'regular') => {
   427→        const fp = fontExists(weight)
   428→        if (fp) doc.font(fp, size)
   429→        else {
   430→          doc.font(weight === 'regular' ? 'Helvetica' : 'Helvetica-Bold', size)
   431→        }
   432→      }
   433→
   434→      const chunks: Buffer[] = []
   435→      doc.on('data', (c: Buffer) => chunks.push(c))
   436→      doc.on('end', () => resolve(Buffer.concat(chunks)))
   437→
   438→      const pageW = 842
   439→      const pageH = 595
   440→      const cx = pageW / 2
   441→
   442→      const roundRectPath = (x: number, y: number, w: number, h: number, r: number) => {
   443→        // pdfkit rounded path
   444→        doc.path(`M ${x + r},${y} L ${x + w - r},${y} Q ${x + w},${y} ${x + w},${y + r} L ${x + w},${y + h - r} Q ${x + w},${y + h} ${x + w - r},${y + h} L ${x + r},${y + h} Q ${x},${y + h} ${x},${y + h - r} L ${x},${y + r} Q ${x},${y} ${x + r},${y} Z`)
   445→      }
   446→
   447→      // ===== Background: cream =====
   448→      doc.rect(0, 0, pageW, pageH).fill(CREAM)
   449→
   450→      // Outer soft green border (container)
   451→      doc.save()
   452→      doc.lineWidth(2).strokeColor(SAGE_GREEN)
   453→      roundRectPath(15, 15, pageW - 30, pageH - 30, 14)
   454→      doc.stroke()
   455→      doc.restore()
   456→
   457→      // ===== Inner certificate frame =====
   458→      const innerPad = 30
   459→      const innerX = innerPad
   460→      const innerY = innerPad
   461→      const innerW = pageW - 2 * innerPad
   462→      const innerH = pageH - 2 * innerPad
   463→
   464→      // Fill cream for inner card
   465→      doc.save()
   466→      roundRectPath(innerX, innerY, innerW, innerH, 12)
   467→      doc.fill(CREAM)
   468→      doc.restore()
   469→
   470→      // Gold border (thick outer line)
   471→      doc.save()
   472→      doc.lineWidth(3).strokeColor(GOLD)
   473→      roundRectPath(innerX, innerY, innerW, innerH, 12)
   474→      doc.stroke()
   475→      doc.restore()
   476→      // Gold border (thin inner line)
   477→      doc.save()
   478→      doc.lineWidth(0.7).strokeColor(GOLD)
   479→      roundRectPath(innerX + 5, innerY + 5, innerW - 10, innerH - 10, 10)
   480→      doc.stroke()

   693→export async function GET(req: NextRequest) {
   694→  try {
   695→    const certNum = req.nextUrl.searchParams.get('number')
   696→    const certId = req.nextUrl.searchParams.get('id')
   697→    const format = req.nextUrl.searchParams.get('format') || 'pdf'
   698→
   699→    if (!certNum && !certId) {
   700→      return NextResponse.json({ error: 'رقم الشهادة أو المعرّف مطلوب' }, { status: 400 })
   701→    }
   702→
   703→    let query = supabaseAdmin.from('certificates').select('*')
   704→    if (certNum) query = query.eq('certificate_number', certNum)
   705→    else if (certId) query = query.eq('id', certId)
   706→
   707→    const { data: cert, error } = await query.single()
   708→    if (error || !cert) {
   709→      return NextResponse.json({ error: 'الشهادة غير موجودة' }, { status: 404 })
   710→    }
   711→
   712→    // Fetch founder's signature
   713→    let founderSignature: string | null = null
   714→    try {
   715→      const { data: trainers } = await supabaseAdmin.from('trainers').select('*')
   716→      if (trainers) {
   717→        const founder = trainers.find((t: any) => t.password && t.password.startsWith('__TEAM_MEMBER__:founder'))
   718→        if (founder) {
   719→          // Parse bio for signature
   720→          const bioSeparator = '\n\n===FIA_EXTRA===\n'
   721→          const idx = founder.bio ? founder.bio.indexOf(bioSeparator) : -1
   722→          if (idx !== -1) {
   723→            const extraJson = founder.bio.substring(idx + bioSeparator.length)
   724→            const extra = JSON.parse(extraJson)
   725→            founderSignature = extra.signature || null
   726→          }
   727→        }
   728→      }
   729→    } catch {}
   730→
   731→    const issueDate = cert.issued_at
   732→      ? new Date(cert.issued_at).toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric' })
   733→      : new Date().toLocaleDateString('ar-EG')
   734→    const verifyUrl = `https://fiaacademy.vercel.app/?verify=${encodeURIComponent(cert.certificate_number)}#verify`
   735→
   736→    if (format === 'png') {
   737→      const pngBytes = await generateCertificatePNG(cert, issueDate, verifyUrl, founderSignature)
   738→      const headers = new Headers()
   739→      headers.set('Content-Type', 'image/png')
   740→      headers.set('Content-Disposition', `inline; filename="FIA-Certificate-${cert.certificate_number}.png"`)
   741→      headers.set('Cache-Control', 'public, max-age=3600')
   742→      return new NextResponse(pngBytes, { status: 200, headers })
   743→    } else {
   744→      const pdfBytes = await generateCertificatePDF(cert, issueDate, verifyUrl, founderSignature)
   745→      const headers = new Headers()
   746→      headers.set('Content-Type', 'application/pdf')
   747→      headers.set('Content-Disposition', `inline; filename="FIA-Certificate-${cert.certificate_number}.pdf"`)
   748→      headers.set('Cache-Control', 'public, max-age=3600')
   749→      return new NextResponse(pdfBytes, { status: 200, headers })
   750→    }
   751→  } catch (e: any) {
   752→    return NextResponse.json({ error: e.message }, { status: 500 })
   753→  }
   754→}
