   160→function AdminOverview() {
   161→  const [stats, setStats] = useState({ courses: 0, trainers: 0, students: 0, payments: 0, totalRevenue: 0, pendingPayments: 0, completedEnrollments: 0, activeEnrollments: 0 })
   162→
   163→  useEffect(() => {
   164→    Promise.all([
   165→      fetch('/api/courses').then((r) => r.json()),
   166→      fetch('/api/trainers').then((r) => r.json()),
   167→      fetch('/api/students?role=student').then((r) => r.json()),
   168→      fetch('/api/payments').then((r) => r.json()),
   169→      fetch('/api/enrollments').then((r) => r.json()),
   170→    ]).then(([c, t, s, p, e]) => {
   171→      const courses = Array.isArray(c) ? c : []
   172→      const trainers = Array.isArray(t) ? t : []
   173→      const students = Array.isArray(s) ? s : []
   174→      const payments = Array.isArray(p) ? p : []
   175→      const enrollments = Array.isArray(e) ? e : []
   176→
   177→      // Calculate REAL revenue from completed payments only
   178→      const completedPayments = payments.filter(p => p.status === 'مكتمل')
   179→      const totalRevenue = completedPayments.reduce((sum, p) => sum + (p.amount || 0), 0)
   180→      const pendingPayments = payments.filter(p => p.status === 'قيد الانتظار').length
   181→      const completedEnrollments = enrollments.filter(e => e.status === 'completed' || e.status === 'مكتمل').length
   182→      const activeEnrollments = enrollments.filter(e => e.status === 'active').length
   183→
   184→      setStats({
   185→        courses: courses.length,
   186→        trainers: trainers.length,
   187→        students: students.length,
   188→        payments: payments.length,
   189→        totalRevenue,
   190→        pendingPayments,
   191→        completedEnrollments,
   192→        activeEnrollments,
   193→      })
   194→    }).catch(() => {})
   195→  }, [])
   196→
   197→  return (
   198→    <div className="space-y-6">
   199→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   200→        <h2 className="text-2xl font-bold mb-1">لوحة تحكم الإدارة 👑</h2>
   201→        <p className="text-white/80">نظرة شاملة على المنصة</p>
   202→      </div>
   203→
   204→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   205→        <StatCard icon={BookOpen} label="إجمالي الدورات" value={stats.courses} color="navy" />
   206→        <StatCard icon={GraduationCap} label="المدربون" value={stats.trainers} color="gold" />
   207→        <StatCard icon={Users} label="المتدربون" value={stats.students} color="blue" />
   208→        <StatCard icon={CreditCard} label="المعاملات" value={stats.payments} color="green" />
   209→      </div>
   210→
   211→      <SectionCard title="نظرة عامة مالية" description="إحصائيات حقيقية من قاعدة البيانات">
   212→        <div className="grid sm:grid-cols-2 lg:grid-cols-4 gap-4">
   213→          <div className="bg-gradient-to-br from-navy to-navy-light text-white p-5 rounded-xl">
   214→            <DollarSign className="w-6 h-6 text-gold mb-2" />
   215→            <div className="text-2xl font-bold">{stats.totalRevenue.toLocaleString()}</div>
   216→            <div className="text-white/70 text-sm">ر.ي إجمالي الإيرادات (مكتمل)</div>
   217→          </div>
   218→          <div className="bg-gradient-to-br from-yellow-500 to-yellow-600 text-white p-5 rounded-xl">
   219→            <Clock className="w-6 h-6 mb-2" />
   220→            <div className="text-2xl font-bold">{stats.pendingPayments}</div>
   221→            <div className="text-white/80 text-sm">مدفوعات قيد الانتظار</div>
   222→          </div>
   223→          <div className="bg-gradient-to-br from-green-500 to-green-600 text-white p-5 rounded-xl">
   224→            <CheckCircle2 className="w-6 h-6 mb-2" />
   225→            <div className="text-2xl font-bold">{stats.completedEnrollments}</div>
   226→            <div className="text-white/80 text-sm">دورات مكتملة</div>
   227→          </div>
   228→          <div className="bg-gradient-to-br from-blue-500 to-blue-600 text-white p-5 rounded-xl">
   229→            <Activity className="w-6 h-6 mb-2" />
   230→            <div className="text-2xl font-bold">{stats.activeEnrollments}</div>
   231→            <div className="text-white/80 text-sm">تسجيلات نشطة</div>
   232→          </div>
   233→        </div>
   234→      </SectionCard>
   235→    </div>
   236→  )
   237→}

   240→function AdminCourses() {
   241→  const [items, setItems] = useState<any[]>([])
   242→  const [loading, setLoading] = useState(true)
   243→  const [editing, setEditing] = useState<any | null>(null)
   244→  const [open, setOpen] = useState(false)
   245→
   246→  const load = () => {
   247→    setLoading(true)
   248→    fetch('/api/courses').then((r) => r.json()).then((d) => setItems(Array.isArray(d) ? d : [])).finally(() => setLoading(false))
   249→  }
   250→  useEffect(() => { load() }, [])
   251→
   252→  const del = async (id: string) => {
   253→    if (!confirm('حذف الدورة؟')) return
   254→    await fetch('/api/courses', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id }) })
   255→    toast.success('تم الحذف')
   256→    load()
   257→  }
   258→
   259→  return (
   260→    <SectionCard
   261→      title="إدارة الدورات"
   262→      description={`${items.length} دورة`}
   263→      action={<Button className="bg-navy hover:bg-navy-light" onClick={() => { setEditing(null); setOpen(true) }}><Plus className="w-4 h-4 ml-1" /> جديد</Button>}
   264→    >
   265→      <div className="overflow-x-auto">
   266→        <table className="w-full text-sm">
   267→          <thead>
   268→            <tr className="border-b border-border text-right text-muted-foreground">
   269→              <th className="py-3 px-2">العنوان</th>
   270→              <th className="py-3 px-2">المستوى</th>
   271→              <th className="py-3 px-2">السعر</th>
   272→              <th className="py-3 px-2">الحالة</th>
   273→              <th className="py-3 px-2">إجراءات</th>
   274→            </tr>
   275→          </thead>
   276→          <tbody>
   277→            {items.map((c) => (
   278→              <tr key={c.id} className="border-b border-border">
   279→                <td className="py-3 px-2 font-medium text-navy">{c.title}</td>
   280→                <td className="py-3 px-2">{c.level}</td>
   281→                <td className="py-3 px-2">{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency}`}</td>
   282→                <td className="py-3 px-2"><Badge className="bg-green-500/15 text-green-600 border-green-500/30">{c.status}</Badge></td>
   283→                <td className="py-3 px-2">
   284→                  <div className="flex gap-1">
   285→                    <Button size="icon" variant="ghost" onClick={() => { setEditing(c); setOpen(true) }}><Edit className="w-4 h-4 text-navy" /></Button>
   286→                    <Button size="icon" variant="ghost" onClick={() => del(c.id)}><Trash2 className="w-4 h-4 text-red-500" /></Button>
   287→                  </div>
   288→                </td>
   289→              </tr>
   290→            ))}
   291→            {!items.length && !loading && <tr><td colSpan={5} className="text-center py-8 text-muted-foreground">لا توجد دورات</td></tr>}
   292→          </tbody>
   293→        </table>
   294→      </div>
   295→      {open && <SimpleDialog open={open} onOpenChange={setOpen} title={editing ? 'تعديل' : 'إضافة'} onSaved={load} initial={editing} fields={['title', 'short_description', 'description', 'level', 'price', 'duration_hours', 'pass_score']} apiPath="/api/courses" />}
   296→    </SectionCard>
   297→  )
   298→}

   312→function AdminStudents() {

   326→    if (!confirm('حذف المتدرب نهائياً؟')) return

   361→                  </div>
   362→                  <div className="flex-1 min-w-0">
   363→                    <div className="font-bold text-sm truncate">{s.name}</div>
   364→                    <div className="text-xs text-white/70 truncate" dir="ltr">{s.email}</div>
   365→                  </div>
   366→                </div>
   367→              </div>
   368→              <div className="p-4 space-y-2">

   370→                  <div className="flex items-center gap-2 text-xs text-muted-foreground">

  1022→  const handleAvatarUpload = async (file: File) => {
  1023→    setUploadingAvatar(true)
  1024→    try {
  1025→      const compressedFile = await new Promise<File>((resolve, reject) => {
  1026→        const reader = new FileReader()
  1027→        reader.onload = (e) => {
  1028→          const img = new Image()
  1029→          img.onload = () => {
  1030→            const maxW = 512, maxH = 512
  1031→            let { width, height } = img
  1032→            if (width > maxW) { height = (height * maxW) / width; width = maxW }
  1033→            if (height > maxH) { width = (width * maxH) / height; height = maxH }
  1034→            const canvas = document.createElement('canvas')
  1035→            canvas.width = width; canvas.height = height
  1036→            const ctx = canvas.getContext('2d')!
  1037→            ctx.drawImage(img, 0, 0, width, height)
  1038→            canvas.toBlob((blob) => {
  1039→              if (!blob) { reject(new Error('Compression failed')); return }
  1040→              resolve(new File([blob], 'avatar.jpg', { type: 'image/jpeg' }))
  1041→            }, 'image/jpeg', 0.85)
  1042→          }
  1043→          img.onerror = () => reject(new Error('Image load failed'))
  1044→          img.src = e.target?.result as string
  1045→        }
  1046→        reader.onerror = () => reject(new Error('File read failed'))

  1052→      fd.append('folder', 'trainer-avatars')

  1082→        const d = await res.json(); toast.error(d.error || 'فشل')
  1083→      }
  1084→    } catch { toast.error('خطأ') } finally { setSaving(false) }
  1085→  }
  1086→

  1786→      enrollments.forEach(enr => {
  1787→        const course = courses.find(co => co.id === enr.course_id)
  1788→        if (enr.status === 'completed' || enr.status === 'مكتمل') {
  1789→          realLogs.push({ action: `إكمال دورة: ${course?.title || 'دورة'}`, time: enr.completed_at || enr.enrolled_at, user: enr.user_id || 'متدرب', type: 'complete' })
  1790→        } else if (enr.status === 'active') {
  1791→          realLogs.push({ action: `تسجيل في دورة: ${course?.title || 'دورة'}`, time: enr.enrolled_at, user: enr.user_id || 'متدرب', type: 'enroll' })
  1792→        }
  1793→      })
  1794→
  1795→      // Sort by date descending
  1796→      realLogs.sort((a, b) => new Date(b.time || 0).getTime() - new Date(a.time || 0).getTime())
  1797→
  1798→      setLogs(realLogs.slice(0, 50))
  1799→    })
  1800→  }, [])
  1801→
  1802→  const iconMap: any = {
  1803→    payment: { icon: CheckCircle2, color: 'bg-green-500', text: 'text-green-600' },
  1804→    cancel: { icon: XCircle, color: 'bg-red-500', text: 'text-red-600' },
  1805→    pending: { icon: Clock, color: 'bg-yellow-500', text: 'text-yellow-600' },
  1806→    complete: { icon: Award, color: 'bg-navy', text: 'text-navy' },
  1807→    enroll: { icon: Users, color: 'bg-blue-500', text: 'text-blue-600' },
  1808→  }
  1809→
  1810→  return (
  1811→    <SectionCard title="سجل العمليات" description={`${logs.length} عملية حقيقية`}>
  1812→      <div className="space-y-2 max-h-[600px] overflow-y-auto fia-scroll">
  1813→        {logs.map((l, i) => {
  1814→          const cfg = iconMap[l.type] || iconMap.enroll
  1815→          const Icon = cfg.icon
  1816→          return (
  1817→            <div key={i} className="flex items-center justify-between p-3 bg-muted/30 rounded-lg border border-border">
  1818→              <div className="flex items-center gap-3 min-w-0">
  1819→                <div className={`w-8 h-8 rounded-lg ${cfg.color} flex items-center justify-center flex-shrink-0`}>
  1820→                  <Icon className="w-4 h-4 text-white" />
  1821→                </div>
  1822→                <div className="min-w-0">
  1823→                  <div className="text-sm font-medium text-navy truncate">{l.action}</div>
  1824→                  <div className="text-xs text-muted-foreground" dir="ltr">{l.user}</div>
  1825→                </div>
  1826→              </div>
  1827→              <div className="text-xs text-muted-foreground flex-shrink-0" dir="ltr">{l.time ? new Date(l.time).toLocaleString('ar-EG') : '-'}</div>
  1828→            </div>
  1829→          )
  1830→        })}
  1831→        {logs.length === 0 && <div className="text-center py-8 text-muted-foreground">لا توجد عمليات مسجّلة</div>}
  1832→      </div>
  1833→    </SectionCard>
  1834→  )
  1835→}

  1837→// Generic simple dialog
  1838→function SimpleDialog({ open, onOpenChange, title, onSaved, initial, fields, apiPath }: any) {
  1839→  const [form, setForm] = useState<any>(initial || {})
  1840→  const [saving, setSaving] = useState(false)
  1841→
  1842→  useEffect(() => { setForm(initial || {}) }, [initial])
  1843→
  1844→  const save = async () => {
  1845→    setSaving(true)
  1846→    try {
  1847→      const res = await fetch(apiPath, {
  1848→        method: initial ? 'PUT' : 'POST',
  1849→        headers: { 'Content-Type': 'application/json' },
  1850→        body: JSON.stringify(initial ? { id: initial.id, ...form } : form),
  1851→      })
  1852→      if (res.ok) { toast.success('تم الحفظ'); onSaved(); onOpenChange(false) }
  1853→      else { const d = await res.json(); toast.error(d.error || 'فشل') }
  1854→    } catch { toast.error('خطأ') } finally { setSaving(false) }
  1855→  }
  1856→
  1857→  const fieldLabels: Record<string, string> = {
  1858→    title: 'العنوان', short_description: 'وصف مختصر', description: 'الوصف', level: 'المستوى',
  1859→    price: 'السعر', duration_hours: 'عدد الساعات', pass_score: 'درجة النجاح', name: 'الاسم',
  1860→    icon: 'الأيقونة (إيموجي)', type: 'النوع', wallet_id: 'معرف المحفظة', instructions: 'التعليمات',
  1861→    student_name: 'اسم المتدرب', course_title: 'عنوان الدورة', status: 'الحالة', issued_by: 'جهة الإصدار',
  1862→    external_accreditation_body: 'جهة الاعتماد الخارجي', logo: 'الشعار (إيموجي)', website: 'الموقع',
  1863→    since: 'منذ', issuing_body: 'جهة الإصدار', number: 'الرقم', issued_at: 'تاريخ الإصدار',
  1864→    expires_at: 'تاريخ الانتهاء', verification_url: 'رابط التحقق',
  1865→  }
  1866→
  1867→  return (
  1868→    <Dialog open={open} onOpenChange={onOpenChange}>
  1869→      <DialogContent className="sm:max-w-md max-h-[90vh] overflow-y-auto">
  1870→        <DialogHeader><DialogTitle>{title}</DialogTitle></DialogHeader>
  1871→        <div className="space-y-3 py-4">
  1872→          {fields.map((f: string) => (
  1873→            <div key={f}>
  1874→              <Label>{fieldLabels[f] || f}</Label>
  1875→              {['description', 'instructions'].includes(f) ? (
  1876→                <Textarea value={form[f] || ''} onChange={(e) => setForm({ ...form, [f]: e.target.value })} rows={2} className="mt-1" />
  1877→              ) : ['price', 'duration_hours', 'pass_score'].includes(f) ? (
  1878→                <Input type="number" value={form[f] || ''} onChange={(e) => setForm({ ...form, [f]: parseInt(e.target.value) || 0 })} dir="ltr" className="mt-1" />
  1879→              ) : (
  1880→                <Input value={form[f] || ''} onChange={(e) => setForm({ ...form, [f]: e.target.value })} dir={f === 'number' || f === 'website' || f === 'verification_url' || f === 'wallet_id' ? 'ltr' : 'rtl'} className="mt-1" />
  1881→              )}
  1882→            </div>
  1883→          ))}
  1884→        </div>
  1885→        <DialogFooter>
  1886→          <Button variant="outline" onClick={() => onOpenChange(false)}>إلغاء</Button>
  1887→          <Button onClick={save} disabled={saving} className="bg-navy hover:bg-navy-light">{saving ? 'جاري...' : 'حفظ'}</Button>
  1888→        </DialogFooter>
  1889→      </DialogContent>
  1890→    </Dialog>
  1891→  )
  1892→}
