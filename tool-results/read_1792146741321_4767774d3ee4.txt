    33→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    34→// Show only the human-readable text in the UI, but preserve the extra data on save.
    35→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    36→const splitBio = (raw: string) => {
    37→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    38→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    39→}
    40→const joinBio = (display: string, original?: string) => {
    41→  if (!original) return display
    42→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    43→  if (i === -1) return display
    44→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    45→}

    47→export function TrainerDashboard() {
    48→  const [active, setActive] = useState('overview')
    49→  const [contentCourseId, setContentCourseId] = useState<string | null>(null)
    50→  const user = useAuth((s) => s.user)
    51→  const updateUser = useAuth((s) => s.updateUser)
    52→  const logout = useAuth((s) => s.logout)
    53→  const [courses, setCourses] = useState<any[]>([])
    54→  const [students, setStudents] = useState<any[]>([])
    55→  const [enrolledStudentIds, setEnrolledStudentIds] = useState<Set<string>>(new Set())
    56→  const [editing, setEditing] = useState<any | null>(null)
    57→  const [isDialogOpen, setIsDialogOpen] = useState(false)
    58→  const [profile, setProfile] = useState({ name: user?.name || '', phone: user?.phone || '', title: user?.title || '', bio: splitBio(user?.bio || ''), specialization: user?.title || '' })
    59→
    60→  const loadCourses = () => {
    61→    fetch('/api/courses').then((r) => r.json()).then((d) => {
    62→      const all = Array.isArray(d) ? d : []
    63→      // Filter to ONLY this trainer's courses
    64→      const mine = all.filter((c: any) => c.trainer_id === user?.id)
    65→      setCourses(mine)
    66→    }).catch(() => {})
    67→  }
    68→  const loadStudents = () => {
    69→    fetch('/api/students?role=student').then((r) => r.json()).then((d) => setStudents(Array.isArray(d) ? d : [])).catch(() => {})
    70→  }
    71→
    72→  // Fetch enrollments and filter to ONLY students enrolled in this trainer's courses
    73→  useEffect(() => {
    74→    if (courses.length === 0) {
    75→      setEnrolledStudentIds(new Set())
    76→      return
    77→    }
    78→    fetch('/api/enrollments').then((r) => r.json()).then((d) => {
    79→      const all = Array.isArray(d) ? d : []
    80→      const courseIds = courses.map((c: any) => c.id)
    81→      const myEnrollments = all.filter((e: any) => courseIds.includes(e.course_id))
    82→      const ids = new Set(myEnrollments.map((e: any) => e.user_id))
    83→      setEnrolledStudentIds(ids)
    84→    }).catch(() => {})
    85→  }, [courses])
    86→
    87→  // Filtered students — only those enrolled in this trainer's courses
    88→  const myStudents = students.filter((s: any) => enrolledStudentIds.has(s.id))
    89→
    90→  useEffect(() => {
    91→    loadCourses()
    92→    loadStudents()
    93→  }, [user?.id])
    94→
    95→  const navigateToContent = (courseId: string) => {
    96→    setContentCourseId(courseId)
    97→    setActive('content')
    98→  }
    99→
   100→  return (
   101→    <DashboardShell
   102→      nav={NAV}
   103→      active={active}
   104→      onNavigate={(id) => { setActive(id); if (id !== 'content') setContentCourseId(null) }}
   105→      roleLabel="مدرب"
   106→      roleColor="bg-green-500/15 text-green-600 border-green-500/30"
   107→    >
   108→      {active === 'overview' && <TrainerOverview user={user} courses={courses} students={myStudents} />}
   109→      {active === 'courses' && <TrainerCourses courses={courses} onAdd={() => { setEditing(null); setIsDialogOpen(true) }} onEdit={(c) => { setEditing(c); setIsDialogOpen(true) }} onRefresh={loadCourses} onManageContent={navigateToContent} />}
   110→      {active === 'content' && contentCourseId && (
   111→        <div className="space-y-4">
   112→          <Button variant="ghost" onClick={() => setActive('courses')} className="text-navy hover:bg-navy/5">
   113→            <ArrowRight className="w-4 h-4 ml-1" /> العودة للدورات
   114→          </Button>
   115→          <CourseContentManager courseId={contentCourseId} />
   116→        </div>
   117→      )}
   118→      {active === 'students' && <TrainerStudents students={myStudents} courses={courses} />}
   119→      {active === 'exams' && <TrainerExams courses={courses} />}
   120→      {active === 'profile' && <TrainerProfile profile={profile} setProfile={setProfile} user={user} updateUser={updateUser} />}
   121→      {active === 'notifications' && <TrainerNotifs courses={courses} />}
   122→      {active === 'settings' && <TrainerAccountSettings user={user} updateUser={updateUser} logout={logout} />}
   123→
   124→      {isDialogOpen && (
   125→        <CourseDialog
   126→          open={isDialogOpen}
   127→          onOpenChange={setIsDialogOpen}
   128→          course={editing}
   129→          trainerId={user?.id}
   130→          onSaved={() => {
   131→            setIsDialogOpen(false)
   132→            loadCourses()
   133→          }}
   134→        />
   135→      )}
   136→    </DashboardShell>
   137→  )
   138→}

   140→function TrainerOverview({ user, courses, students }: any) {
   141→  // Compute the real average rating from courses that actually carry a
   142→  // numeric rating. If no ratings exist yet, fall back to a clear message
   143→  // instead of a hardcoded fake number.
   144→  const ratedCourses = (courses || []).filter(
   145→    (c: any) => typeof c.rating === 'number' && c.rating > 0
   146→  )
   147→  const avgRating =
   148→    ratedCourses.length > 0
   149→      ? (ratedCourses.reduce((acc, c) => acc + c.rating, 0) / ratedCourses.length).toFixed(1)
   150→      : 'لا توجد تقييمات بعد'
   151→
   152→  // Real "active exams" count: lessons with lesson_type === 'exam' across
   153→  // the trainer's own courses (same data source as TrainerExams).
   154→  const [examsCount, setExamsCount] = useState<number>(0)
   155→  const [examsLoading, setExamsLoading] = useState(false)
   156→
   157→  useEffect(() => {
   158→    let cancelled = false
   159→    const load = async () => {
   160→      if (!courses || courses.length === 0) {
   161→        setExamsCount(0)
   162→        return
   163→      }
   164→      setExamsLoading(true)
   165→      try {
   166→        const results = await Promise.all(
   167→          courses.map(async (c: any) => {
   168→            try {
   169→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   170→              const data = await res.json()
   171→              const lessons = Array.isArray(data) ? data : []
   172→              return lessons.filter((l: any) => l.lesson_type === 'exam').length
   173→            } catch {
   174→              return 0
   175→            }
   176→          })
   177→        )
   178→        if (cancelled) return
   179→        setExamsCount(results.reduce((a, b) => a + b, 0))
   180→      } catch {
   181→        if (!cancelled) setExamsCount(0)
   182→      } finally {
   183→        if (!cancelled) setExamsLoading(false)
   184→      }
   185→    }
   186→    load()
   187→    return () => { cancelled = true }
   188→  }, [courses])
   189→
   190→  return (
   191→    <div className="space-y-6">
   192→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   193→        <h2 className="text-2xl font-bold mb-1">مرحباً، {user?.name} 👋</h2>
   194→        <p className="text-white/80">نظرة عامة على نشاطك التدريبي</p>
   195→      </div>
   196→
   197→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   198→        <StatCard icon={BookOpen} label="الدورات التدريبية" value={courses.length} color="navy" />
   199→        <StatCard icon={Users} label="إجمالي المتدربين" value={String(students.length)} color="gold" />
   200→        <StatCard icon={Star} label="متوسط التقييم" value={avgRating} color="green" />
   201→        <StatCard
   202→          icon={FileText}
   203→          label="الاختبارات النشطة"
   204→          value={examsLoading ? '—' : examsCount}
   205→          color="blue"
   206→        />
   207→      </div>
   208→
   209→      <SectionCard title="أحدث الدورات" description="الدورات التي تدرّسها">
   210→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   211→          {courses.slice(0, 3).map((c: any) => (
   212→            <Card key={c.id} className="overflow-hidden card-hover">
   213→              <div className="h-32 bg-navy">
   214→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   215→              </div>
   216→              <div className="p-4">
   217→                <h3 className="font-bold text-navy text-sm mb-1 line-clamp-1">{c.title}</h3>
   218→                <div className="text-xs text-muted-foreground">{c.enrolled || 0} متدرب • {c.rating || 0} ⭐</div>
   219→              </div>
   220→            </Card>
   221→          ))}
   222→        </div>
   223→      </SectionCard>
   224→    </div>
   225→  )
   226→}

   228→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   229→  const [deleting, setDeleting] = useState<string | null>(null)
   230→  // Course currently awaiting delete confirmation. Null means the dialog is closed.
   231→  const [deleteTarget, setDeleteTarget] = useState<any | null>(null)
   232→
   233→  const handleDelete = async () => {
   234→    if (!deleteTarget) return
   235→    const id = deleteTarget.id
   236→    setDeleting(id)
   237→    try {
   238→      const res = await fetch('/api/courses', {
   239→        method: 'DELETE',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({ id }),
   242→      })
   243→      if (res.ok) {
   244→        toast.success('تم حذف الدورة')
   245→        setDeleteTarget(null)
   246→        onRefresh()
   247→      } else {
   248→        const err = await res.json().catch(() => ({}))
   249→        toast.error(err.error || 'فشل الحذف')
   250→      }
   251→    } catch {
   252→      toast.error('خطأ في الاتصال')
   253→    } finally {
   254→      setDeleting(null)
   255→    }
   256→  }
   257→
   258→  return (
   259→    <SectionCard
   260→      title="دوراتي التدريبية"
   261→      description="إدارة الدورات التي تدرّسها"
   262→      action={
   263→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   264→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   265→        </Button>
   266→      }
   267→    >
   268→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   269→        {courses.map((c: any) => (
   270→          <Card key={c.id} className="overflow-hidden card-hover">
   271→            <div className="h-32 bg-navy relative">
   272→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   273→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   274→            </div>
   275→            <div className="p-4">
   276→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   277→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   278→                <span>{c.enrolled || 0} متدرب</span>
   279→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   280→              </div>
   281→              <div className="space-y-2">
   282→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   283→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   284→                </Button>
   285→                <div className="flex gap-2">
   286→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   287→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   288→                  </Button>
   289→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => setDeleteTarget(c)} disabled={deleting === c.id}>
   290→                    <Trash2 className="w-3.5 h-3.5" />
   291→                  </Button>
   292→                </div>
   293→              </div>
   294→            </div>
   295→          </Card>
   296→        ))}
   297→      </div>
   298→
   299→      {/* Professional delete-confirmation card */}
   300→      <Dialog
   301→        open={!!deleteTarget}
   302→        onOpenChange={(open) => { if (!open && !deleting) setDeleteTarget(null) }}
   303→      >
   304→        <DialogContent className="sm:max-w-md">
   305→          <DialogHeader>
   306→            <DialogTitle className="flex items-center gap-2 text-red-600">
   307→              <AlertTriangle className="w-5 h-5" />
   308→              تأكيد الحذف
   309→            </DialogTitle>
   310→          </DialogHeader>
   311→          <div className="py-2 space-y-3">
   312→            <div className="flex items-start gap-3 p-3 bg-red-500/5 border border-red-500/20 rounded-lg">
   313→              <Trash2 className="w-5 h-5 text-red-500 flex-shrink-0 mt-0.5" />
   314→              <div className="min-w-0">
   315→                <div className="text-xs text-muted-foreground mb-1">سيتم حذف الدورة التالية:</div>
   316→                <div className="font-bold text-navy line-clamp-2 break-words">{deleteTarget?.title}</div>
   317→              </div>
   318→            </div>
   319→            <p className="text-sm text-muted-foreground text-center">
   320→              هل أنت متأكد من الحذف؟ لا يمكن التراجع عن هذا الإجراء.
   321→            </p>
   322→          </div>
   323→          <DialogFooter>
   324→            <Button
   325→              variant="outline"
   326→              onClick={() => setDeleteTarget(null)}
   327→              disabled={!!deleting}
   328→            >
   329→              إلغاء
   330→            </Button>
   331→            <Button
   332→              onClick={handleDelete}
   333→              disabled={!!deleting}
   334→              className="bg-red-600 hover:bg-red-700 text-white"
   335→            >
   336→              {deleting
   337→                ? <><Loader2 className="w-4 h-4 ml-1 animate-spin" /> جاري الحذف...</>
   338→                : <><Trash2 className="w-4 h-4 ml-1" /> حذف الدورة</>}
   339→            </Button>
   340→          </DialogFooter>
   341→        </DialogContent>
   342→      </Dialog>
   343→    </SectionCard>
   344→  )
   345→}

   347→function CourseDialog({ open, onOpenChange, course, trainerId, onSaved }: any) {
   348→  const [form, setForm] = useState<any>(
   349→    course || {
   350→      title: '',
   351→      short_description: '',
   352→      description: '',
   353→      level: 'متوسط',
   354→      price_type: 'مجانية',
   355→      price: 0,
   356→      currency: 'ر.ي',
   357→      duration_hours: 0,
   358→      pass_score: 60,
   359→      cover: '',
   360→      status: 'منشورة',
   361→    }
   362→  )
   363→  const [saving, setSaving] = useState(false)
   364→  const [uploadingCover, setUploadingCover] = useState(false)
   365→
   366→  // Keep the form in sync with the `course` prop whenever the dialog opens or
   367→  // the course being edited changes. Without this, a dialog that stays mounted
   368→  // across course switches would show stale state (and could overwrite the
   369→  // wrong course's `cover` field).
   370→  useEffect(() => {
   371→    if (open) {
   372→      setForm(
   373→        course || {
   374→          title: '',
   375→          short_description: '',
   376→          description: '',
   377→          level: 'متوسط',
   378→          price_type: 'مجانية',
   379→          price: 0,
   380→          currency: 'ر.ي',
   381→          duration_hours: 0,
   382→          pass_score: 60,
   383→          cover: '',
   384→          status: 'منشورة',
   385→        }
   386→      )
   387→    }
   388→  }, [open, course])
   389→
   390→  const handleUploadCover = async (file: File) => {
   391→    setUploadingCover(true)
   392→    try {
   393→      const fd = new FormData()
   394→      fd.append('file', file)
   395→      fd.append('folder', 'courses')
   396→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   397→      const data = await res.json().catch(() => ({}))
   398→      if (data.url) {
   399→        // Store the Telegram-hosted URL on form.cover — it is later persisted
   400→        // to the courses.cover column by the save() call below.
   401→        setForm((prev: any) => ({ ...prev, cover: data.url }))
   402→        toast.success('تم رفع صورة الغلاف')
   403→      } else {
   404→        toast.error('فشل رفع الصورة: ' + (data.error || ''))
   405→      }
   406→    } catch {
   407→      toast.error('خطأ في الاتصال')
   408→    } finally {
   409→      setUploadingCover(false)
   410→    }
   411→  }
   412→
   413→  const save = async () => {
   414→    if (!form.title) {
   415→      toast.error('أدخل عنوان الدورة')
   416→      return
   417→    }
   418→    setSaving(true)
   419→    try {
   420→      const slug = form.title.toLowerCase().replace(/\s+/g, '-').slice(0, 80)
   421→      // Build the payload. Always include `cover` (string URL or null) so the
   422→      // courses.cover column is updated on every save — even when the trainer
   423→      // cleared the field.
   424→      const body: any = {
   425→        title: form.title,
   426→        short_description: form.short_description ?? '',
   427→        description: form.description ?? '',
   428→        level: form.level,
   429→        price_type: form.price_type,
   430→        price: Number(form.price) || 0,
   431→        currency: form.currency || 'ر.ي',
   432→        duration_hours: Number(form.duration_hours) || 0,
   433→        pass_score: Number(form.pass_score) || 60,
   434→        status: form.status || 'منشورة',
   435→        cover: form.cover ?? null,
   436→        slug,
   437→        trainer_id: trainerId,
   438→        modules_count: form.modules_count || 0,
   439→        lessons_count: form.lessons_count || 0,
   440→      }
   441→      const res = await fetch('/api/courses', {
   442→        method: course ? 'PUT' : 'POST',
   443→        headers: { 'Content-Type': 'application/json' },
   444→        body: JSON.stringify(course ? { id: course.id, ...body } : body),
   445→      })
   446→      if (res.ok) {
   447→        toast.success(course ? 'تم تحديث الدورة' : 'تم إنشاء الدورة')
   448→        onSaved()
   449→      } else {
   450→        const err = await res.json().catch(() => ({}))
   451→        toast.error(err.error || 'فشل الحفظ')
   452→      }
   453→    } catch {
   454→      toast.error('خطأ')
   455→    } finally {
   456→      setSaving(false)
   457→    }
   458→  }
   459→
   460→  return (
   461→    <Dialog open={open} onOpenChange={onOpenChange}>
   462→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   463→        <DialogHeader>
   464→          <DialogTitle>{course ? 'تعديل الدورة' : 'دورة جديدة'}</DialogTitle>
   465→        </DialogHeader>
   466→        <div className="space-y-4 py-4">
   467→          <div>
   468→            <Label>عنوان الدورة</Label>
   469→            <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} className="mt-1" />
   470→          </div>
   471→          <div>
   472→            <Label>وصف مختصر</Label>
   473→            <Input value={form.short_description || ''} onChange={(e) => setForm({ ...form, short_description: e.target.value })} className="mt-1" />
   474→          </div>
   475→          <div>
   476→            <Label>الوصف الكامل</Label>
   477→            <Textarea value={form.description || ''} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} className="mt-1" />
   478→          </div>
   479→          <div className="grid grid-cols-2 gap-3">
   480→            <div>
   481→              <Label>المستوى</Label>
   482→              <select
   483→                value={form.level}
   484→                onChange={(e) => setForm({ ...form, level: e.target.value })}

   690→function TrainerExams({ courses }: { courses: any[] }) {
   691→  const [loading, setLoading] = useState(true)
   692→  const [exams, setExams] = useState<any[]>([])
   693→
   694→  useEffect(() => {
   695→    let cancelled = false
   696→    const loadExams = async () => {
   697→      if (!courses || courses.length === 0) {
   698→        setExams([])
   699→        setLoading(false)
   700→        return
   701→      }
   702→      setLoading(true)
   703→      try {
   704→        // Fetch lessons for each course in parallel
   705→        const results = await Promise.all(
   706→          courses.map(async (c) => {
   707→            try {
   708→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   709→              const data = await res.json()
   710→              const lessons = Array.isArray(data) ? data : []
   711→              // Filter exam-type lessons only
   712→              return lessons
   713→                .filter((l: any) => l.lesson_type === 'exam')
   714→                .map((l: any) => {
   715→                  // Parse question count from content JSON
   716→                  let questions: any[] = []
   717→                  if (l.content) {
   718→                    try {
   719→                      const parsed = JSON.parse(l.content)
   720→                      if (Array.isArray(parsed)) questions = parsed
   721→                    } catch {}
   722→                  }
   723→                  return {
   724→                    ...l,
   725→                    course_id: c.id,
   726→                    course_title: c.title,
   727→                    course_pass_score: c.pass_score,
   728→                    questions_count: questions.length,
   729→                    questions,
   730→                  }
   731→                })
   732→            } catch {
   733→              return []
   734→            }
   735→          })
   736→        )
   737→        if (cancelled) return
   738→        const flat = results.flat()
   739→        setExams(flat)
   740→      } catch {
   741→        if (!cancelled) setExams([])
   742→      } finally {
   743→        if (!cancelled) setLoading(false)
   744→      }
   745→    }
   746→    loadExams()
   747→    return () => { cancelled = true }
   748→  }, [courses])
   749→
   750→  const totalQuestions = exams.reduce((acc, e) => acc + (e.questions_count || 0), 0)
   751→  const totalCoursesWithExams = new Set(exams.map((e) => e.course_id)).size
   752→
   753→  return (
   754→    <div className="space-y-6">
   755→      <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
   756→        <StatCard icon={FileText} label="الاختبارات" value={exams.length} color="blue" />
   757→        <StatCard icon={HelpCircle} label="إجمالي الأسئلة" value={totalQuestions} color="navy" />
   758→        <StatCard icon={BookOpen} label="دورات بها اختبارات" value={totalCoursesWithExams} color="gold" />
   759→      </div>
   760→
   761→      <SectionCard
   762→        title="الاختبارات"
   763→        description="الاختبارات المستندة إلى محتوى دوراتك التدريبية"
   764→      >
   765→        {loading ? (
   766→          <div className="text-center py-12">
   767→            <Loader2 className="w-8 h-8 text-navy mx-auto mb-3 animate-spin" />
   768→            <p className="text-muted-foreground text-sm">جاري تحميل الاختبارات...</p>
   769→          </div>
   770→        ) : exams.length === 0 ? (
   771→          <div className="text-center py-12">
   772→            <FileText className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   773→            <p className="text-muted-foreground mb-2">لا توجد اختبارات بعد</p>
   774→            <p className="text-xs text-muted-foreground">
   775→              أنشئ دروساً من نوع "اختبار" في محتوى دوراتك لتظهر هنا

   894→function TrainerAccountSettings({ user, updateUser, logout }: any) {
   895→  const [currentPassword, setCurrentPassword] = useState('')
   896→  const [newEmail, setNewEmail] = useState('')
   897→  const [newPassword, setNewPassword] = useState('')
   898→  const [confirmPassword, setConfirmPassword] = useState('')
   899→  const [showCurrent, setShowCurrent] = useState(false)
   900→  const [showNew, setShowNew] = useState(false)
   901→  const [loading, setLoading] = useState(false)
   902→
   903→  const save = async () => {
   904→    if (!currentPassword) {
   905→      toast.error('أدخل كلمة المرور الحالية للتأكيد')
   906→      return
   907→    }
   908→    if (newPassword && newPassword !== confirmPassword) {
   909→      toast.error('كلمتا المرور الجديدتان غير متطابقتين')
   910→      return
   911→    }
   912→    if (newPassword && newPassword.length < 6) {
   913→      toast.error('كلمة المرور الجديدة يجب أن تكون 6 أحرف على الأقل')
   914→      return
   915→    }
   916→    if (!newEmail && !newPassword) {
   917→      toast.error('لا توجد تغييرات للحفظ')
   918→      return
   919→    }
   920→    setLoading(true)
   921→    try {
   922→      const res = await fetch('/api/auth/change-credentials', {
   923→        method: 'POST',
   924→        headers: { 'Content-Type': 'application/json' },
   925→        body: JSON.stringify({
   926→          userId: user.id,
   927→          currentEmail: user.email,
   928→          currentPassword,
   929→          newEmail: newEmail || undefined,
   930→          newPassword: newPassword || undefined,
   931→          role: 'trainer',
   932→        }),
   933→      })
   934→      const data = await res.json()
   935→      if (!res.ok) {
   936→        toast.error(data.error || 'فشل التحديث')
   937→        return
   938→      }
   939→      toast.success('تم تحديث بيانات الدخول بنجاح')
   940→      updateUser({
   941→        email: newEmail || user.email,
   942→        password: newPassword || user.password,
   943→      })
   944→      setCurrentPassword('')
   945→      setNewEmail('')
   946→      setNewPassword('')
   947→      setConfirmPassword('')
   948→      if (newEmail) {
   949→        setTimeout(() => {
   950→          logout()
   951→          toast.info('تم تحديث البريد. يرجى تسجيل الدخول بالبيانات الجديدة')
   952→        }, 2000)
   953→      }
   954→    } catch {
   955→      toast.error('خطأ في الاتصال')
   956→    } finally {
   957→      setLoading(false)
   958→    }
   959→  }

  1075→function TrainerNotifs({ courses }: { courses: any[] }) {
  1076→  const [loading, setLoading] = useState(true)
  1077→  const [notifs, setNotifs] = useState<any[]>([])
  1078→
  1079→  useEffect(() => {
  1080→    let cancelled = false
  1081→    const loadNotifs = async () => {
  1082→      if (!courses || courses.length === 0) {
  1083→        setNotifs([])
  1084→        setLoading(false)
  1085→        return
  1086→      }
  1087→      setLoading(true)
  1088→      try {
  1089→        const courseIds = courses.map((c) => c.id)
  1090→        const courseById = new Map(courses.map((c) => [c.id, c]))
  1091→
  1092→        // Fetch enrollments for all courses (we'll filter on the client)
  1093→        const [enrRes, stuRes] = await Promise.all([
  1094→          fetch('/api/enrollments'),
  1095→          fetch('/api/students?role=student'),
  1096→        ])
  1097→        const [enrollments, students] = await Promise.all([enrRes.json(), stuRes.json()])
  1098→
  1099→        if (cancelled) return
  1100→
  1101→        const enrList = Array.isArray(enrollments) ? enrollments : []
  1102→        const stuList = Array.isArray(students) ? students : []
  1103→        const studentById = new Map(stuList.map((s: any) => [s.id, s]))
  1104→
  1105→        // Only enrollments for this trainer's courses
  1106→        const mine = enrList.filter((e: any) => courseIds.includes(e.course_id))
  1107→
  1108→        // Sort by enrolled_at desc
  1109→        mine.sort((a: any, b: any) => {
  1110→          const aT = a.enrolled_at ? new Date(a.enrolled_at).getTime() : 0
  1111→          const bT = b.enrolled_at ? new Date(b.enrolled_at).getTime() : 0
  1112→          return bT - aT
  1113→        })
  1114→
  1115→        // Build notification objects — newest 25
  1116→        const built = mine.slice(0, 25).map((e: any) => {
  1117→          const student = studentById.get(e.user_id)
  1118→          const course = courseById.get(e.course_id)
  1119→          const isFree = !e.amount_paid || e.amount_paid === 0
  1120→          return {
  1121→            id: e.id,
  1122→            type: 'enrollment',
  1123→            studentName: student?.name || 'متدرب',
  1124→            studentAvatar: student?.avatar,
  1125→            courseTitle: course?.title || 'دورة',
  1126→            enrolledAt: e.enrolled_at,
  1127→            isFree,
  1128→            amountPaid: e.amount_paid,
  1129→            currency: e.currency,
  1130→            paymentStatus: e.payment_status,
  1131→          }
  1132→        })
  1133→
  1134→        if (!cancelled) setNotifs(built)
  1135→      } catch {
  1136→        if (!cancelled) setNotifs([])
  1137→      } finally {
  1138→        if (!cancelled) setLoading(false)
  1139→      }
  1140→    }
  1141→    loadNotifs()
  1142→    return () => { cancelled = true }
  1143→  }, [courses])
  1144→
  1145→  const formatRelativeTime = (iso: string | null | undefined) => {
  1146→    if (!iso) return ''
  1147→    try {
  1148→      const date = new Date(iso)
  1149→      const diffMs = Date.now() - date.getTime()
  1150→      const diffMin = Math.floor(diffMs / 60000)
  1151→      if (diffMin < 1) return 'الآن'
  1152→      if (diffMin < 60) return `منذ ${diffMin} دقيقة`
  1153→      const diffHr = Math.floor(diffMin / 60)
  1154→      if (diffHr < 24) return `منذ ${diffHr} ساعة`
  1155→      const diffDay = Math.floor(diffHr / 24)
  1156→      if (diffDay < 30) return `منذ ${diffDay} يوم`
  1157→      return date.toLocaleDateString('ar-EG')
  1158→    } catch {
  1159→      return ''
  1160→    }
  1161→  }
  1162→
  1163→  return (
  1164→    <SectionCard
  1165→      title="الإشعارات"
  1166→      description="آخر التسجيلات في دوراتك التدريبية"
  1167→    >
  1168→      {loading ? (
  1169→        <div className="text-center py-12">
  1170→          <Loader2 className="w-8 h-8 text-navy mx-auto mb-3 animate-spin" />
  1171→          <p className="text-muted-foreground text-sm">جاري تحميل الإشعارات...</p>
  1172→        </div>
  1173→      ) : notifs.length === 0 ? (
  1174→        <div className="text-center py-12">
  1175→          <Bell className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
  1176→          <p className="text-muted-foreground">لا توجد إشعارات</p>
  1177→          <p className="text-xs text-muted-foreground mt-1">
  1178→            ستظهر هنا إشعارات التسجيل الجديدة في دوراتك
  1179→          </p>
  1180→        </div>
  1181→      ) : (
  1182→        <div className="space-y-3">
  1183→          {notifs.map((n) => (
  1184→            <div
  1185→              key={n.id}
  1186→              className="p-4 bg-gold/5 border border-gold/20 rounded-lg flex items-start gap-3"
  1187→            >
  1188→              <div className="w-10 h-10 rounded-full bg-navy flex items-center justify-center text-gold font-bold flex-shrink-0 border border-gold/30 overflow-hidden">
  1189→                {n.studentAvatar ? (

  1191→                ) : (
  1192→                  <UserPlus className="w-5 h-5" />
  1193→                )}
  1194→              </div>
  1195→              <div className="flex-1 min-w-0">
  1196→                <div className="font-semibold text-navy text-sm">
  1197→                  متدرب جديد في دورتك
  1198→                </div>
  1199→                <div className="text-xs text-muted-foreground mt-1">
  1200→                  <span className="font-medium text-navy">{n.studentName}</span>
  1201→                  {' '}انضم إلى{' '}
  1202→                  <span className="font-medium text-navy">{n.courseTitle}</span>
  1203→                </div>
  1204→                <div className="flex items-center gap-2 mt-2 flex-wrap">

  1207→                      n.isFree
  1208→                        ? 'bg-green-500/15 text-green-600 border-green-500/30 text-xs'
  1209→                        : 'bg-gold/15 text-gold border-gold/30 text-xs'
  1210→                    }
  1211→                  >
  1212→                    {n.isFree ? 'مجانية' : `${(n.amountPaid || 0).toLocaleString()} ${n.currency || 'ر.ي'}`}
  1213→                  </Badge>

  1219→                  {n.enrolledAt && (
  1220→                    <span className="text-xs text-muted-foreground">

  1224→                </div>
