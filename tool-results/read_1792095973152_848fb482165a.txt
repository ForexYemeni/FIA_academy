    45→    45→function registerFontsForCanvas() {
    46→    46→  if (fontsRegistered) return
    47→    47→  try {
    48→    48→    const fonts = getFontPaths()
    49→    49→    if (fs.existsSync(fonts.regular)) GlobalFonts.registerFromPath(fonts.regular, 'Cairo')
    50→    50→    if (fs.existsSync(fonts.bold)) GlobalFonts.registerFromPath(fonts.bold, 'Cairo-Bold')
    51→    51→    if (fs.existsSync(fonts.black)) GlobalFonts.registerFromPath(fonts.black, 'Cairo-Black')
    52→    52→    fontsRegistered = true
    53→    53→  } catch (e) {}
    54→    54→}

    56→    56→async function generateBarcode(certNumber: string): Promise<Buffer> {
    57→    57→  try {
    58→    58→    return await bwip.toBuffer({
    59→    59→      bcid: 'code128',
    60→    60→      text: certNumber,
    61→    61→      scale: 2,
    62→    62→      height: 20,
    63→    63→      includetext: false,
    64→    64→      paddingwidth: 3,
    65→    65→      paddingheight: 3,
    66→    66→      backgroundcolor: 'FFFFFF',
    67→    67→      foregroundcolor: '0F172A',
    68→    68→    })
    69→    69→  } catch (e: any) {
    70→    70→    const canvas = createCanvas(200, 50)
    71→    71→    const ctx = canvas.getContext('2d')
    72→    72→    ctx.fillStyle = 'white'
    73→    73→    ctx.fillRect(0, 0, 200, 50)
    74→    74→    ctx.fillStyle = NAVY
    75→    75→    let x = 5
    76→    76→    for (let i = 0; i < 40; i++) {
    77→    77→      const w = 1 + ((certNumber.charCodeAt(i % certNumber.length) * (i + 1)) % 4)
    78→    78→      if (i % 2 === 0) ctx.fillRect(x, 5, w, 40)
    79→    79→      x += w
    80→    80→    }
    81→    81→    return await canvas.encode('png')
    82→    82→  }
    83→    83→}

   103→   103→  const ctx = canvas.getContext('2d')
   104→   104→
   105→   105→  // Helpers
   106→   106→  const fill = (c: string) => { ctx.fillStyle = c }
   107→   107→  const stroke = (c: string, w: number) => { ctx.strokeStyle = c; ctx.lineWidth = w }
   108→   108→  const roundRect = (x: number, y: number, w: number, h: number, r: number) => {
   109→   109→    ctx.beginPath()
   110→   110→    ctx.moveTo(x + r, y)
   111→   111→    ctx.lineTo(x + w - r, y)
   112→   112→    ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   113→   113→    ctx.lineTo(x + w, y + h - r)
   114→   114→    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   115→   115→    ctx.lineTo(x + r, y + h)
   116→   116→    ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   117→   117→    ctx.lineTo(x, y + r)
   118→   118→    ctx.quadraticCurveTo(x, y, x + r, y)

   135→   135→

   168→   168→  ctx.lineTo(innerX + cornerInset + cornerLen, innerY + cornerInset)

   693→   693→export async function GET(req: NextRequest) {
   694→   694→  try {
   695→   695→    const certNum = req.nextUrl.searchParams.get('number')
   696→   696→    const certId = req.nextUrl.searchParams.get('id')
   697→   697→    const format = req.nextUrl.searchParams.get('format') || 'pdf'
   698→   698→
   699→   699→    if (!certNum && !certId) {
   700→   700→      return NextResponse.json({ error: 'رقم الشهادة أو المعرّف مطلوب' }, { status: 400 })
   701→   701→    }
   702→   702→
   703→   703→    let query = supabaseAdmin.from('certificates').select('*')
   704→   704→    if (certNum) query = query.eq('certificate_number', certNum)
   705→   705→    else if (certId) query = query.eq('id', certId)
   706→   706→
   707→   707→    const { data: cert, error } = await query.single()
   708→   708→    if (error || !cert) {
   709→   709→      return NextResponse.json({ error: 'الشهادة غير موجودة' }, { status: 404 })
   710→   710→    }
   711→   711→
   712→   712→    // Fetch founder's signature
   713→   713→    let founderSignature: string | null = null
   714→   714→    try {
   715→   715→      const { data: trainers } = await supabaseAdmin.from('trainers').select('*')
   716→   716→      if (trainers) {
   717→   717→        const founder = trainers.find((t: any) => t.password && t.password.startsWith('__TEAM_MEMBER__:founder'))
   718→   718→        if (founder) {
   719→   719→          // Parse bio for signature
   720→   720→          const bioSeparator = '\n\n===FIA_EXTRA===\n'
   721→   721→          const idx = founder.bio ? founder.bio.indexOf(bioSeparator) : -1
   722→   722→          if (idx !== -1) {
   723→   723→            const extraJson = founder.bio.substring(idx + bioSeparator.length)
   724→   724→            const extra = JSON.parse(extraJson)
   725→   725→            founderSignature = extra.signature || null
   726→   726→          }
   727→   727→        }
   728→   728→      }
   729→   729→    } catch {}
   730→   730→
   731→   731→    const issueDate = cert.issued_at
   732→   732→      ? new Date(cert.issued_at).toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric' })
   733→   733→      : new Date().toLocaleDateString('ar-EG')
   734→   734→    const verifyUrl = `https://fiaacademy.vercel.app/?verify=${encodeURIComponent(cert.certificate_number)}#verify`
   735→   735→
   736→   736→    if (format === 'png') {
   737→   737→      const pngBytes = await generateCertificatePNG(cert, issueDate, verifyUrl, founderSignature)
   738→   738→      const headers = new Headers()
   739→   739→      headers.set('Content-Type', 'image/png')
   740→   740→      headers.set('Content-Disposition', `inline; filename="FIA-Certificate-${cert.certificate_number}.png"`)
   741→   741→      headers.set('Cache-Control', 'public, max-age=3600')
   742→   742→      return new NextResponse(pngBytes, { status: 200, headers })
   743→   743→    } else {
   744→   744→      const pdfBytes = await generateCertificatePDF(cert, issueDate, verifyUrl, founderSignature)
   745→   745→      const headers = new Headers()
   746→   746→      headers.set('Content-Type', 'application/pdf')
   747→   747→      headers.set('Content-Disposition', `inline; filename="FIA-Certificate-${cert.certificate_number}.pdf"`)
   748→   748→      headers.set('Cache-Control', 'public, max-age=3600')
   749→   749→      return new NextResponse(pdfBytes, { status: 200, headers })
   750→   750→    }
   751→   751→  } catch (e: any) {
   752→   752→    return NextResponse.json({ error: e.message }, { status: 500 })
   753→   753→  }
