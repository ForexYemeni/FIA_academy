    66→            </div>

   105→function CouponDialog({ open, onOpenChange, initial, onSaved }: any) {
   106→  const [form, setForm] = useState<any>(initial || { code: '', description: '', discount_type: 'percentage', discount_value: 10, max_uses: '', expires_at: '', status: 'نشط' })
   107→  const [saving, setSaving] = useState(false)
   108→  useEffect(() => {
   109→    setForm(initial || { code: '', description: '', discount_type: 'percentage', discount_value: 10, max_uses: '', expires_at: '', status: 'نشط' })
   110→  }, [initial])
   111→
   112→  const save = async () => {
   113→    if (!form.discount_value) { toast.error('قيمة الخصم مطلوبة'); return }
   114→    setSaving(true)
   115→    try {
   116→      const payload = { ...form, max_uses: form.max_uses ? parseInt(form.max_uses) : null, expires_at: form.expires_at || null }
   117→      const res = await fetch('/api/coupons', {
   118→        method: initial ? 'PUT' : 'POST',
   119→        headers: { 'Content-Type': 'application/json' },
   120→        body: JSON.stringify(initial ? { id: initial.id, ...payload } : payload),
   121→      })
   122→      if (res.ok) {
   123→        toast.success(initial ? 'تم التحديث' : 'تم إنشاء الكوبون')
   124→        onSaved(); onOpenChange(false)
   125→      } else {
   126→        const d = await res.json(); toast.error(d.error || 'فشل')
   127→      }
   128→    } finally { setSaving(false) }
   129→  }
   130→
   131→  return (
   132→    <Dialog open={open} onOpenChange={onOpenChange}>
   133→      <DialogContent className="sm:max-w-md max-h-[90vh] overflow-y-auto">
   134→        <DialogHeader><DialogTitle>{initial ? 'تعديل كوبون' : 'كوبون جديد'}</DialogTitle></DialogHeader>
   135→        <div className="space-y-4 py-4">
   136→          <div>
   137→            <Label>الكود (اتركه فارغاً للتوليد التلقائي)</Label>
   138→            <Input value={form.code} onChange={(e) => setForm({ ...form, code: e.target.value })} dir="ltr" className="mt-1 font-mono" placeholder="FIA-XXXXXX" />
   139→          </div>
   140→          <div>
   141→            <Label>الوصف</Label>
   142→            <Input value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} className="mt-1" />
   143→          </div>
   144→          <div className="grid grid-cols-2 gap-3">
   145→            <div>
   146→              <Label>نوع الخصم</Label>
   147→              <select value={form.discount_type} onChange={(e) => setForm({ ...form, discount_type: e.target.value })} className="w-full mt-1 px-3 py-2 border border-input rounded-md bg-background text-sm">
   148→                <option value="percentage">نسبة مئوية %</option>
   149→                <option value="fixed">مبلغ ثابت</option>
   150→              </select>
   151→            </div>
   152→            <div>
   153→              <Label>القيمة</Label>
   154→              <Input type="number" value={form.discount_value || ''} onChange={(e) => setForm({ ...form, discount_value: e.target.value === '' ? 0 : parseFloat(e.target.value) })} className="mt-1" />
   155→            </div>
   156→          </div>
   157→          <div className="grid grid-cols-2 gap-3">
   158→            <div>
   159→              <Label>حد الاستخدام</Label>
   160→              <Input type="number" value={form.max_uses || ''} onChange={(e) => setForm({ ...form, max_uses: e.target.value })} className="mt-1" placeholder="فارغ = غير محدود" />
   161→            </div>
   162→            <div>
   163→              <Label>تاريخ الانتهاء</Label>
   164→              <Input type="date" value={form.expires_at ? form.expires_at.split('T')[0] : ''} onChange={(e) => setForm({ ...form, expires_at: e.target.value ? new Date(e.target.value).toISOString() : '' })} className="mt-1" />
   165→            </div>
   166→          </div>
   167→        </div>
   168→        <DialogFooter>
   169→          <Button variant="outline" onClick={() => onOpenChange(false)}>إلغاء</Button>
   170→          <Button onClick={save} disabled={saving} className="bg-navy hover:bg-navy-light">
   171→            {saving ? <Loader2 className="w-4 h-4 ml-1 animate-spin" /> : null} حفظ
   172→          </Button>
   173→        </DialogFooter>
   174→      </DialogContent>
   175→    </Dialog>
   176→  )
   177→}

   353→  const [open, setOpen] = useState(false)

   356→  const [deleting, setDeleting] = useState(false)
   357→
   358→  useEffect(() => {
   359→    fetch('/api/courses').then((r) => r.json()).then((d) => {
   360→      const arr = Array.isArray(d) ? d : []
   361→      setCourses(arr)
   362→      if (arr.length > 0 && !selectedCourse) setSelectedCourse(arr[0].id)
   363→    }).catch(() => {})
   364→  }, [])
   365→
   366→  useEffect(() => {
   367→    if (selectedCourse) {
   368→      fetch(`/api/exam-questions?course_id=${selectedCourse}`).then((r) => r.json()).then((d) => setItems(Array.isArray(d) ? d : [])).catch(() => setItems([]))
   369→    } else {
   370→      setItems([])
   371→    }
   372→  }, [selectedCourse])
   373→
   374→  const handleDelete = async () => {
   375→    if (!deleteTarget) return
   376→    setDeleting(true)
   377→    try {
   378→      await fetch('/api/exam-questions', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: deleteTarget.id }) })
   379→      toast.success('تم حذف السؤال')
   380→      setDeleteTarget(null)
   381→      if (selectedCourse) {
   382→        fetch(`/api/exam-questions?course_id=${selectedCourse}`).then((r) => r.json()).then((d) => setItems(Array.isArray(d) ? d : []))
   383→      }
   384→    } finally { setDeleting(false) }
   385→  }
   386→

  1619→function FaqDialog({ open, onOpenChange, initial, onSaved }: any) {
  1620→  const [form, setForm] = useState<any>(initial || { question: '', answer: '', category: 'عام', display_order: 0 })
  1621→  const [saving, setSaving] = useState(false)
  1622→  useEffect(() => { setForm(initial || { question: '', answer: '', category: 'عام', display_order: 0 }) }, [initial])
  1623→
  1624→  const save = async () => {
  1625→    if (!form.question || !form.answer) { toast.error('السؤال والإجابة مطلوبان'); return }

  1627→    try {
  1628→      const res = await fetch('/api/faqs', { method: initial ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(initial ? { id: initial.id, ...form } : form) })

  1630→      else { const d = await res.json(); toast.error(d.error || 'فشل') }
  1631→    } finally { setSaving(false) }
  1632→  }

  1635→    <Dialog open={open} onOpenChange={onOpenChange}>
  1636→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
