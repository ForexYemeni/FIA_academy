   119→function AdminContent({ contentCourseId, setContentCourseId }: any) {
   120→  const [courses, setCourses] = useState<any[]>([])
   121→  const [enrollmentsByCourse, setEnrollmentsByCourse] = useState<Record<string, number>>({})
   122→
   123→  // Load courses + ALL enrollments (no user filter) so we can show the
   124→  // real enrolled-students count per course. The legacy `c.enrolled`
   125→  // column is never updated when a student enrolls, so we have to
   126→  // compute the count from the payments/enrollments table.
   127→  useEffect(() => {
   128→    Promise.all([
   129→      fetch('/api/courses').then((r) => r.json()).catch(() => []),
   130→      fetch('/api/enrollments').then((r) => r.json()).catch(() => []),
   131→    ]).then(([c, e]: any) => {
   132→      setCourses(Array.isArray(c) ? c : [])
   133→      const list: any[] = Array.isArray(e) ? e : []
   134→      // Count active enrollments — only those that have real access:
   135→      //   payment_status in ['free', 'confirmed', 'completed']
   136→      // Skip 'pending' (payment not yet approved) and any
   137→      // cancelled/failed enrollments.
   138→      const counts: Record<string, number> = {}
   139→      for (const enr of list) {
   140→        if (!enr?.course_id) continue
   141→        const ps = enr.payment_status
   142→        // Defensive: accept the canonical normalized values, AND
   143→        // the raw status 'مكتمل'/'completed' as a fallback (in case
   144→        // the toEnrollment normalization in the API missed an edge
   145→        // case, the count should still reflect reality).
   146→        const rawStatus = (enr.status || '').trim()
   147→        const isApproved =
   148→          ps === 'free' ||
   149→          ps === 'confirmed' ||
   150→          ps === 'completed' ||
   151→          rawStatus === 'مكتمل' ||
   152→          rawStatus === 'completed' ||
   153→          rawStatus === 'paid'
   154→        if (!isApproved) continue
   155→        counts[enr.course_id] = (counts[enr.course_id] || 0) + 1
   156→      }
   157→      setEnrollmentsByCourse(counts)
   158→    })
   159→  }, [])
   160→
   161→  // Helper to get actual enrolled count for a course
   162→  const enrolledCount = (courseId: string) => enrollmentsByCourse[courseId] || 0
   163→
   164→  if (contentCourseId) {
   165→    return (
   166→      <div className="space-y-4">
   167→        <Button variant="ghost" onClick={() => setContentCourseId(null)} className="text-navy hover:bg-navy/5">
   168→          <ArrowRight className="w-4 h-4 ml-1" /> العودة لقائمة الدورات
   169→        </Button>
   170→        <CourseContentManager courseId={contentCourseId} />
   171→      </div>
   172→    )
   173→  }
   174→
   175→  return (
   176→    <SectionCard
   177→      title="محتوى الدورات"
   178→      description="تصفّح كل الدورات والدروس"
   179→    >
   180→      {courses.length === 0 ? (
   181→        <div className="text-center py-12">
   182→          <BookOpen className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   183→          <p className="text-muted-foreground">لا توجد دورات بعد</p>
   184→        </div>
   185→      ) : (
   186→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   187→          {courses.map((c: any) => {
   188→            const isFree = c.price_type === 'مجانية' || c.price === 0
   189→            return (
   190→              <Card key={c.id} className="overflow-hidden card-hover">
   191→                <div className="h-32 bg-navy relative">
   192→                  {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   193→                  <div className="absolute top-2 right-2 flex gap-1">
   194→                    <Badge className="bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   195→                  </div>
   196→                  <div className="absolute top-2 left-2">
   197→                    <Badge className={isFree ? 'bg-green-500 text-white' : 'bg-gold text-navy'}>
   198→                      {isFree ? 'مجانية' : 'مدفوعة'}
   199→                    </Badge>
   200→                  </div>
   201→                </div>
   202→                <div className="p-4">
   203→                  <h3 className="font-bold text-navy text-sm mb-2 line-clamp-2 min-h-[2.5rem]">{c.title}</h3>
   204→                  <div className="flex justify-between text-xs text-muted-foreground mb-3">
   205→                    <span>{c.modules_count || 0} وحدة</span>
   206→                    <span>{c.lessons_count || 0} درس</span>
   207→                    <span>{c.duration_hours || 0} ساعة</span>
   208→                  </div>
   209→                  <div className="text-xs text-muted-foreground mb-3 flex items-center gap-1.5">
   210→                    <span className="inline-flex items-center justify-center w-5 h-5 rounded-full bg-navy/10 text-navy font-bold text-[10px]">
   211→                      {enrolledCount(c.id)}
   212→                    </span>
   213→                    <span>متدرب مسجّل فعلياً</span>
   214→                  </div>
   215→                  <Button
   216→                    size="sm"
   217→                    className="w-full bg-navy hover:bg-navy-light"
   218→                    onClick={() => setContentCourseId(c.id)}
   219→                  >
   220→                    <FileText className="w-3.5 h-3.5 ml-1" /> عرض المحتوى
   221→                  </Button>
   222→                </div>
   223→              </Card>
   224→            )
   225→          })}
   226→        </div>
   227→      )}
   228→    </SectionCard>
   229→  )
   230→}

   232→function AdminOverview() {
   233→  const [stats, setStats] = useState({
   234→    courses: 0, trainers: 0, students: 0, payments: 0,
   235→    pendingPayments: 0, completedEnrollments: 0, activeEnrollments: 0,
   236→    // Revenue grouped by currency. Each entry: { currency, total, count, paid_count, pending_count, pending_total }
   237→    revenueByCurrency: [] as any[],
   238→  })
   239→
   240→  // Get the course lookup map so we can resolve a payment's currency
   241→  // from the course (source of truth) when the payment row itself
   242→  // has the wrong/missing currency (legacy rows default to 'ر.ي').
   243→  const [courses, setCourses] = useState<any[]>([])
   244→
   245→  useEffect(() => {
   246→    Promise.all([
   247→      fetch('/api/courses').then((r) => r.json()),
   248→      fetch('/api/trainers').then((r) => r.json()),
   249→      fetch('/api/students?role=student').then((r) => r.json()),
   250→      fetch('/api/payments').then((r) => r.json()),
   251→      fetch('/api/enrollments').then((r) => r.json()),
   252→    ]).then(([c, t, s, p, e]) => {
   253→      const coursesArr = Array.isArray(c) ? c : []
   254→      const trainers = Array.isArray(t) ? t : []
   255→      const students = Array.isArray(s) ? s : []
   256→      const payments = Array.isArray(p) ? p : []
   257→      const enrollments = Array.isArray(e) ? e : []
   258→
   259→      setCourses(coursesArr)
   260→
   261→      // Build a course-id → course map for currency resolution
   262→      const courseMap = new Map<string, any>()
   263→      for (const c of coursesArr) courseMap.set(c.id, c)
   264→
   265→      // Determine currency for a payment: prefer the course's currency,
   266→      // then the payment's stored currency, then default 'ر.ي'.
   267→      const resolveCurrency = (pay: any): string => {
   268→        if (pay.course_id) {
   269→          const c = courseMap.get(pay.course_id)
   270→          if (c?.currency) return c.currency
   271→        }
   272→        return pay.currency || 'ر.ي'
   273→      }
   274→
   275→      // Group all payments by currency. For each currency, we compute:
   276→      //   total            = sum of amounts for COMPLETED payments
   277→      //   pending_total    = sum of amounts for PENDING payments
   278→      //   paid_count       = number of completed payments
   279→      //   pending_count    = number of pending payments
   280→      const groups: Record<string, { total: number; pending_total: number; paid_count: number; pending_count: number }> = {}
   281→      for (const pay of payments) {
   282→        const cur = resolveCurrency(pay)
   283→        if (!groups[cur]) groups[cur] = { total: 0, pending_total: 0, paid_count: 0, pending_count: 0 }
   284→        if (pay.status === 'مكتمل' || pay.status === 'completed' || pay.status === 'paid') {
   285→          groups[cur].total += Number(pay.amount) || 0
   286→          groups[cur].paid_count += 1
   287→        } else if (pay.status === 'قيد الانتظار' || pay.status === 'pending') {
   288→          groups[cur].pending_total += Number(pay.amount) || 0
   289→          groups[cur].pending_count += 1
   290→        }
   291→      }
   292→      const revenueByCurrency = Object.entries(groups).map(([currency, g]) => ({
   293→        currency, ...g,
   294→      })).sort((a, b) => b.total - a.total)
   295→
   296→      setStats({
   297→        courses: coursesArr.length,
   298→        trainers: trainers.length,
   299→        students: students.length,
   300→        payments: payments.length,
   301→        pendingPayments: payments.filter(p => p.status === 'قيد الانتظار' || p.status === 'pending').length,
   302→        completedEnrollments: enrollments.filter(e => e.status === 'completed' || e.status === 'مكتمل').length,
   303→        activeEnrollments: enrollments.filter(e => e.status === 'active').length,
   304→        revenueByCurrency,
   305→      })
   306→    }).catch(() => {})
   307→  }, [])
   308→
   309→  // Pretty currency formatter — group digits, append the currency code
   310→  const fmt = (amount: number, currency: string) => `${amount.toLocaleString('en-US')} ${currency}`
   311→
   312→  return (
   313→    <div className="space-y-6">
   314→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   315→        <h2 className="text-2xl font-bold mb-1">لوحة تحكم الإدارة 👑</h2>
   316→        <p className="text-white/80">نظرة شاملة على المنصة</p>
   317→      </div>
   318→
   319→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   320→        <StatCard icon={BookOpen} label="إجمالي الدورات" value={stats.courses} color="navy" />
   321→        <StatCard icon={GraduationCap} label="المدربون" value={stats.trainers} color="gold" />
   322→        <StatCard icon={Users} label="المتدربون" value={stats.students} color="blue" />
   323→        <StatCard icon={CreditCard} label="المعاملات" value={stats.payments} color="green" />
   324→      </div>
   325→
   326→      {/* ========= PROFESSIONAL FINANCIAL OVERVIEW ========= */}
   327→      <SectionCard title="نظرة عامة مالية" description="الإيرادات الفعلية مجمّعة حسب العملة">
   328→        {stats.revenueByCurrency.length === 0 ? (
   329→          <div className="text-center py-10 text-muted-foreground">
   330→            <DollarSign className="w-12 h-12 mx-auto mb-3 opacity-30" />
   331→            <p>لا توجد معاملات بعد</p>
   332→          </div>
   333→        ) : (
   334→          <div className="space-y-6">
   335→            {/* Currency cards — one per currency, professional layout */}
   336→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   337→              {stats.revenueByCurrency.map((rev) => {
   338→                const isUSD = rev.currency === '$' || rev.currency === 'USD'
   339→                const isSAR = rev.currency === 'ر.س'
   340→                const gradient = isUSD
   341→                  ? 'from-emerald-600 to-emerald-800'
   342→                  : isSAR
   343→                    ? 'from-blue-700 to-blue-900'
   344→                    : 'from-navy to-navy-light'
   345→                const accent = isUSD ? 'text-emerald-300' : isSAR ? 'text-blue-300' : 'text-gold'
   346→                return (
   347→                  <div
   348→                    key={rev.currency}
   349→                    className={`relative bg-gradient-to-br ${gradient} text-white p-5 rounded-2xl shadow-lg overflow-hidden`}
   350→                  >
   351→                    {/* Decorative currency icon in background */}
   352→                    <div className="absolute -left-4 -bottom-4 text-white/10 text-7xl font-bold select-none">
   353→                      {rev.currency}
   354→                    </div>
   355→                    <div className="relative">
   356→                      <div className="flex items-center justify-between mb-3">
   357→                        <div className="flex items-center gap-2">
   358→                          <DollarSign className={`w-5 h-5 ${accent}`} />
   359→                          <span className="text-xs font-semibold text-white/70 uppercase tracking-wide">العملة</span>
   360→                        </div>
   361→                        <span className="text-base font-extrabold bg-white/15 px-2 py-0.5 rounded-md" dir="ltr">
   362→                          {rev.currency}
   363→                        </span>
   364→                      </div>
   365→                      {/* Total completed revenue — the headline number */}
   366→                      <div className="mb-3">
   367→                        <div className="text-[11px] text-white/70 mb-0.5">إجمالي الإيرادات المكتملة</div>
   368→                        <div className="text-3xl font-extrabold tracking-tight" dir="ltr">
   369→                          {fmt(rev.total, rev.currency)}
   370→                        </div>
   371→                      </div>
   372→                      {/* Pending row */}
   373→                      <div className="border-t border-white/15 pt-2 grid grid-cols-2 gap-2 text-xs">
   374→                        <div>
   375→                          <div className="text-white/70">قيد الانتظار</div>
   376→                          <div className="font-bold" dir="ltr">{fmt(rev.pending_total, rev.currency)}</div>
   377→                          <div className="text-[10px] text-white/60">{rev.pending_count} معاملة</div>
   378→                        </div>
   379→                        <div>
   380→                          <div className="text-white/70">مكتملة</div>
   381→                          <div className="font-bold" dir="ltr">{fmt(rev.total, rev.currency)}</div>
   382→                          <div className="text-[10px] text-white/60">{rev.paid_count} معاملة</div>
   383→                        </div>
   384→                      </div>
   385→                    </div>
   386→                  </div>
   387→                )
   388→              })}
   389→            </div>
   390→
   391→            {/* Summary cards — count-based metrics (currency-agnostic) */}
   392→            <div className="grid sm:grid-cols-3 gap-4">
   393→              <div className="bg-gradient-to-br from-yellow-500 to-yellow-600 text-white p-5 rounded-xl">
   394→                <Clock className="w-6 h-6 mb-2" />
   395→                <div className="text-2xl font-bold">{stats.pendingPayments}</div>
   396→                <div className="text-white/80 text-sm">مدفوعات قيد الانتظار</div>
   397→              </div>
   398→              <div className="bg-gradient-to-br from-green-500 to-green-600 text-white p-5 rounded-xl">
   399→                <CheckCircle2 className="w-6 h-6 mb-2" />
   400→                <div className="text-2xl font-bold">{stats.completedEnrollments}</div>
   401→                <div className="text-white/80 text-sm">دورات مكتملة</div>
   402→              </div>
   403→              <div className="bg-gradient-to-br from-blue-500 to-blue-600 text-white p-5 rounded-xl">
   404→                <Activity className="w-6 h-6 mb-2" />
   405→                <div className="text-2xl font-bold">{stats.activeEnrollments}</div>

   409→          </div>
   410→        )}
   411→      </SectionCard>
   412→    </div>
   413→  )

   417→function AdminCourses() {
   418→  const [items, setItems] = useState<any[]>([])
   419→  const [loading, setLoading] = useState(true)
   420→  const [editing, setEditing] = useState<any | null>(null)
   421→  const [open, setOpen] = useState(false)
   422→  const [deleteTarget, setDeleteTarget] = useState<any | null>(null)
   423→  const [deleting, setDeleting] = useState(false)
   424→
   425→  const load = () => {
   426→    setLoading(true)
   427→    fetch('/api/courses').then((r) => r.json()).then((d) => setItems(Array.isArray(d) ? d : [])).finally(() => setLoading(false))
   428→  }
   429→  useEffect(() => { load() }, [])
   430→
   431→  const handleDelete = async () => {
   432→    if (!deleteTarget) return

   734→        toast.error(err.error || 'فشل تحديث حالة الدفع')
   735→        return
   736→      }

   741→      toast.error('خطأ في الاتصال')

   746→    if (statusFilter !== 'all' && p.status !== statusFilter) return false

   748→      const q = search.toLowerCase()
   749→      return (p.reference || '').toLowerCase().includes(q) || (p.method || '').toLowerCase().includes(q)
   750→    }
   751→    return true

   755→  const extractProofUrl = (ref: string) => {

   768→    if (p?.course_id) {
   769→      const course = getCourse(p.course_id)

   771→    }

   779→    if (!ref) return null

   804→

   806→        {filtered.map((p) => {
   807→          const course = getCourse(p.course_id)
   808→          const proofUrl = extractProofUrl(p.reference)
   809→          const isPending = p.status === 'قيد الانتظار'
   810→          const couponCode = getCouponCode(p)
   811→          const ref = cleanReference(p.reference)
   812→          const currency = getCurrency(p)
