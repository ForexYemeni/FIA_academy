    41→export function TrainerDashboard() {
    42→  const [active, setActive] = useState('overview')
    43→  useHashTab(NAV.map((n) => n.id), setActive)
    44→  const [contentCourseId, setContentCourseId] = useState<string | null>(null)
    45→  const user = useAuth((s) => s.user)
    46→  const updateUser = useAuth((s) => s.updateUser)
    47→  const logout = useAuth((s) => s.logout)
    48→  const [courses, setCourses] = useState<any[]>([])
    49→  const [students, setStudents] = useState<any[]>([])
    50→  const [enrolledStudentIds, setEnrolledStudentIds] = useState<Set<string>>(new Set())
    51→  const [editing, setEditing] = useState<any | null>(null)
    52→  const [isDialogOpen, setIsDialogOpen] = useState(false)
    53→  // Profile state is seeded from auth-store user; refreshTrainerProfile() below
    54→  // keeps it in sync with the latest DB record fetched from /api/trainers.
    55→  const [profile, setProfile] = useState({
    56→    name: user?.name || '',
    57→    phone: user?.phone || '',
    58→    title: user?.title || '',
    59→    specialization: user?.title || '',
    60→    bio: user?.bio || '',
    61→    avatar: user?.avatar || '',
    62→  })
    63→
    64→  const refreshTrainerProfile = useCallback(async () => {
    65→    if (!user?.id) return
    66→    try {
    67→      const res = await fetch(`/api/trainers?id=${user.id}`)
    68→      const data = await res.json()
    69→      if (data && !data.error) {
    70→        setProfile({
    71→          name: data.name || '',
    72→          phone: data.phone || '',
    73→          title: data.title || '',
    74→          specialization: data.specialization || data.title || '',
    75→          bio: data.bio || '',
    76→          avatar: data.avatar || '',
    77→        })
    78→        // Keep auth-store in sync too so other parts of the dashboard reflect
    79→        // any admin-side edits.
    80→        updateUser({
    81→          name: data.name,
    82→          phone: data.phone,
    83→          title: data.title,
    84→          bio: data.bio,
    85→          avatar: data.avatar,
    86→        })
    87→      }
    88→    } catch {
    89→      // silent
    90→    }
    91→  }, [user?.id, updateUser])
    92→
    93→  useEffect(() => {
    94→    refreshTrainerProfile()
    95→  }, [refreshTrainerProfile])
    96→
    97→  const loadCourses = () => {
    98→    fetch('/api/courses').then((r) => r.json()).then((d) => {
    99→      const all = Array.isArray(d) ? d : []
   100→      // Filter to ONLY this trainer's courses
   101→      const mine = all.filter((c: any) => c.trainer_id === user?.id)
   102→      setCourses(mine)
   103→    }).catch(() => {})
   104→  }
   105→  const loadStudents = () => {
   106→    fetch('/api/students?role=student').then((r) => r.json()).then((d) => setStudents(Array.isArray(d) ? d : [])).catch(() => {})
   107→  }
   108→
   109→  // Fetch enrollments and filter to ONLY students enrolled in this trainer's courses
   110→  useEffect(() => {
   111→    if (courses.length === 0) {
   112→      setEnrolledStudentIds(new Set())
   113→      return
   114→    }
   115→    fetch('/api/enrollments').then((r) => r.json()).then((d) => {
   116→      const all = Array.isArray(d) ? d : []
   117→      const courseIds = courses.map((c: any) => c.id)
   118→      const myEnrollments = all.filter((e: any) => courseIds.includes(e.course_id))
   119→      const ids = new Set(myEnrollments.map((e: any) => e.user_id))
   120→      setEnrolledStudentIds(ids)
   121→    }).catch(() => {})
   122→  }, [courses])
   123→
   124→  // Filtered students — only those enrolled in this trainer's courses
   125→  const myStudents = students.filter((s: any) => enrolledStudentIds.has(s.id))
   126→
   127→  useEffect(() => {
   128→    loadCourses()
   129→    loadStudents()
   130→  }, [user?.id])
   131→
   132→  const navigateToContent = (courseId: string) => {
   133→    setContentCourseId(courseId)
   134→    setActive('content')
   135→  }
   136→
   137→  return (
   138→    <DashboardShell
   139→      nav={NAV}
   140→      active={active}
   141→      onNavigate={(id) => { setActive(id); if (id !== 'content') setContentCourseId(null) }}
   142→      roleLabel="مدرب"
   143→      roleColor="bg-green-500/15 text-green-600 border-green-500/30"
   144→    >
   145→      {active === 'overview' && <TrainerOverview user={user} courses={courses} students={myStudents} />}
   146→      {active === 'courses' && <TrainerCourses courses={courses} onAdd={() => { setEditing(null); setIsDialogOpen(true) }} onEdit={(c) => { setEditing(c); setIsDialogOpen(true) }} onRefresh={loadCourses} onManageContent={navigateToContent} />}
   147→      {active === 'content' && contentCourseId && (
   148→        <div className="space-y-4">
   149→          <Button variant="ghost" onClick={() => setActive('courses')} className="text-navy hover:bg-navy/5">
   150→            <ArrowRight className="w-4 h-4 ml-1" /> العودة للدورات
   151→          </Button>
   152→          <CourseContentManager courseId={contentCourseId} />
   153→        </div>
   154→      )}
   155→      {active === 'students' && <TrainerStudents students={myStudents} courses={courses} />}
   156→      {active === 'exams' && <TrainerExams courses={courses} />}
   157→      {active === 'forum' && <SharedForum user={user} />}
   158→      {active === 'blog' && <SharedBlog />}
   159→      {active === 'faqs' && <SharedFAQ />}
   160→      {active === 'affiliate' && (
   161→        <SectionCard title="برنامج العمولة" description="شارك أصدقاءك واكسب عمولة">
   162→          <AffiliateDashboard />
   163→        </SectionCard>
   164→      )}
   165→      {active === 'profile' && <TrainerProfile profile={profile} setProfile={setProfile} user={user} updateUser={updateUser} refreshProfile={refreshTrainerProfile} />}
   166→      {active === 'notifications' && <TrainerNotifs courses={courses} />}
   167→      {active === 'settings' && <TrainerAccountSettings user={user} updateUser={updateUser} logout={logout} />}
   168→
   169→      {isDialogOpen && (
   170→        <ProCourseDialog
   171→          open={isDialogOpen}
   172→          onOpenChange={setIsDialogOpen}
   173→          course={editing}
   174→          trainerId={user?.id}
   175→          onSaved={() => {
   176→            setIsDialogOpen(false)
   177→            loadCourses()
   178→          }}
   179→        />
   180→      )}
   181→    </DashboardShell>
   182→  )
   183→}

   185→function TrainerOverview({ user, courses, students }: any) {
   186→  // Compute the real average rating from courses that actually carry a
   187→  // numeric rating. If no ratings exist yet, fall back to a clear message
   188→  // instead of a hardcoded fake number.
   189→  const ratedCourses = (courses || []).filter(
   190→    (c: any) => typeof c.rating === 'number' && c.rating > 0
   191→  )
   192→  const avgRating =
   193→    ratedCourses.length > 0
   194→      ? (ratedCourses.reduce((acc, c) => acc + c.rating, 0) / ratedCourses.length).toFixed(1)
   195→      : 'لا توجد تقييمات بعد'
   196→
   197→  // Real "active exams" count: lessons with lesson_type === 'exam' across
   198→  // the trainer's own courses (same data source as TrainerExams).
   199→  const [examsCount, setExamsCount] = useState<number>(0)
   200→  const [examsLoading, setExamsLoading] = useState(false)
   201→
   202→  useEffect(() => {
   203→    let cancelled = false
   204→    const load = async () => {
   205→      if (!courses || courses.length === 0) {
   206→        setExamsCount(0)
   207→        return
   208→      }
   209→      setExamsLoading(true)
   210→      try {
   211→        const results = await Promise.all(
   212→          courses.map(async (c: any) => {
   213→            try {
   214→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   215→              const data = await res.json()
   216→              const lessons = Array.isArray(data) ? data : []
   217→              return lessons.filter((l: any) => l.lesson_type === 'exam').length
   218→            } catch {
   219→              return 0
   220→            }
   221→          })
   222→        )
   223→        if (cancelled) return
   224→        setExamsCount(results.reduce((a, b) => a + b, 0))
   225→      } catch {
   226→        if (!cancelled) setExamsCount(0)
   227→      } finally {
   228→        if (!cancelled) setExamsLoading(false)
   229→      }
   230→    }
   231→    load()
   232→    return () => { cancelled = true }
   233→  }, [courses])
   234→
   235→  return (
   236→    <div className="space-y-6">
   237→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   238→        <h2 className="text-2xl font-bold mb-1">مرحباً، {user?.name} 👋</h2>
   239→        <p className="text-white/80">نظرة عامة على نشاطك التدريبي</p>
   240→      </div>
   241→
   242→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   243→        <StatCard icon={BookOpen} label="الدورات التدريبية" value={courses.length} color="navy" />
   244→        <StatCard icon={Users} label="إجمالي المتدربين" value={String(students.length)} color="gold" />
   245→        <StatCard icon={Star} label="متوسط التقييم" value={avgRating} color="green" />
   246→        <StatCard
   247→          icon={FileText}
   248→          label="الاختبارات النشطة"
   249→          value={examsLoading ? '—' : examsCount}
   250→          color="blue"
   251→        />
   252→      </div>
   253→
   254→      <SectionCard title="أحدث الدورات" description="الدورات التي تدرّسها">
   255→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   256→          {courses.slice(0, 3).map((c: any) => (
   257→            <Card key={c.id} className="overflow-hidden card-hover">
   258→              <div className="h-32 bg-navy">
   259→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   260→              </div>
   261→              <div className="p-4">
   262→                <h3 className="font-bold text-navy text-sm mb-1 line-clamp-1">{c.title}</h3>
   263→                <div className="text-xs text-muted-foreground">{c.enrolled || 0} متدرب • {c.rating || 0} ⭐</div>
   264→              </div>
   265→            </Card>
   266→          ))}
   267→        </div>
   268→      </SectionCard>
   269→    </div>
   270→  )
   271→}

   273→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   274→  const [deleting, setDeleting] = useState<string | null>(null)
   275→  // Course currently awaiting delete confirmation. Null means the dialog is closed.
   276→  const [deleteTarget, setDeleteTarget] = useState<any | null>(null)
   277→
   278→  const handleDelete = async () => {
   279→    if (!deleteTarget) return
   280→    const id = deleteTarget.id
   281→    setDeleting(id)
   282→    try {
   283→      const res = await fetch('/api/courses', {
   284→        method: 'DELETE',
   285→        headers: { 'Content-Type': 'application/json' },
   286→        body: JSON.stringify({ id }),
   287→      })
   288→      if (res.ok) {
   289→        toast.success('تم حذف الدورة')
   290→        setDeleteTarget(null)
   291→        onRefresh()
   292→      } else {
   293→        const err = await res.json().catch(() => ({}))
   294→        toast.error(err.error || 'فشل الحذف')
   295→      }
   296→    } catch {
   297→      toast.error('خطأ في الاتصال')
   298→    } finally {
   299→      setDeleting(null)
   300→    }
   301→  }
   302→
   303→  return (
   304→    <SectionCard
   305→      title="دوراتي التدريبية"
   306→      description="إدارة الدورات التي تدرّسها"
   307→      action={
   308→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   309→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   310→        </Button>
   311→      }
   312→    >
   313→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   314→        {courses.map((c: any) => (
   315→          <Card key={c.id} className="overflow-hidden card-hover">
   316→            <div className="h-32 bg-navy relative">
   317→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   318→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   319→            </div>
   320→            <div className="p-4">
   321→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   322→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   323→                <span>{c.enrolled || 0} متدرب</span>
   324→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   325→              </div>
   326→              <div className="space-y-2">
   327→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   328→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   329→                </Button>
   330→                <div className="flex gap-2">
   331→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   332→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   333→                  </Button>
   334→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => setDeleteTarget(c)} disabled={deleting === c.id}>
   335→                    <Trash2 className="w-3.5 h-3.5" />
   336→                  </Button>
   337→                </div>
   338→              </div>
   339→            </div>
   340→          </Card>
   341→        ))}
   342→      </div>
   343→
   344→      {/* Professional delete-confirmation card */}
   345→      <Dialog
   346→        open={!!deleteTarget}
   347→        onOpenChange={(open) => { if (!open && !deleting) setDeleteTarget(null) }}
   348→      >
   349→        <DialogContent className="sm:max-w-md">
   350→          <DialogHeader>
   351→            <DialogTitle className="flex items-center gap-2 text-red-600">
   352→              <AlertTriangle className="w-5 h-5" />
   353→              تأكيد الحذف
   354→            </DialogTitle>
   355→          </DialogHeader>
   356→          <div className="py-2 space-y-3">
   357→            <div className="flex items-start gap-3 p-3 bg-red-500/5 border border-red-500/20 rounded-lg">
   358→              <Trash2 className="w-5 h-5 text-red-500 flex-shrink-0 mt-0.5" />
   359→              <div className="min-w-0">
   360→                <div className="text-xs text-muted-foreground mb-1">سيتم حذف الدورة التالية:</div>
   361→                <div className="font-bold text-navy line-clamp-2 break-words">{deleteTarget?.title}</div>
   362→              </div>
   363→            </div>
   364→            <p className="text-sm text-muted-foreground text-center">
   365→              هل أنت متأكد من الحذف؟ لا يمكن التراجع عن هذا الإجراء.
   366→            </p>
   367→          </div>
   368→          <DialogFooter>
   369→            <Button
   370→              variant="outline"
   371→              onClick={() => setDeleteTarget(null)}
   372→              disabled={!!deleting}
   373→            >
   374→              إلغاء
   375→            </Button>
   376→            <Button
   377→              onClick={handleDelete}
   378→              disabled={!!deleting}
   379→              className="bg-red-600 hover:bg-red-700 text-white"
   380→            >
   381→              {deleting
   382→                ? <><Loader2 className="w-4 h-4 ml-1 animate-spin" /> جاري الحذف...</>
   383→                : <><Trash2 className="w-4 h-4 ml-1" /> حذف الدورة</>}
   384→            </Button>
   385→          </DialogFooter>
   386→        </DialogContent>
   387→      </Dialog>
   388→    </SectionCard>
   389→  )
   390→}

   392→function CourseDialog({ open, onOpenChange, course, trainerId, onSaved }: any) {
   393→  const [form, setForm] = useState<any>(
   394→    course || {
   395→      title: '',
   396→      short_description: '',
   397→      description: '',
   398→      level: 'متوسط',
   399→      price_type: 'مجانية',
   400→      price: 0,
   401→      currency: 'ر.ي',
   402→      duration_hours: 0,
   403→      pass_score: 60,
   404→      cover: '',
   405→      status: 'منشورة',
   406→    }
   407→  )
   408→  const [saving, setSaving] = useState(false)
   409→  const [uploadingCover, setUploadingCover] = useState(false)
   410→
   411→  // Keep the form in sync with the `course` prop whenever the dialog opens or
   412→  // the course being edited changes. Without this, a dialog that stays mounted
   413→  // across course switches would show stale state (and could overwrite the
   414→  // wrong course's `cover` field).
   415→  useEffect(() => {
   416→    if (open) {
   417→      setForm(
   418→        course || {
   419→          title: '',
   420→          short_description: '',
   421→          description: '',
   422→          level: 'متوسط',
   423→          price_type: 'مجانية',
   424→          price: 0,
   425→          currency: 'ر.ي',
   426→          duration_hours: 0,
   427→          pass_score: 60,
   428→          cover: '',
   429→          status: 'منشورة',
   430→        }
   431→      )
   432→    }
   433→  }, [open, course])
   434→
   435→  const handleUploadCover = async (file: File) => {
   436→    setUploadingCover(true)
   437→    try {
   438→      const fd = new FormData()
   439→      fd.append('file', file)
   440→      fd.append('folder', 'courses')
   441→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   442→      const data = await res.json().catch(() => ({}))
   443→      if (data.url) {
   444→        // Store the Telegram-hosted URL on form.cover — it is later persisted
   445→        // to the courses.cover column by the save() call below.
   446→        setForm((prev: any) => ({ ...prev, cover: data.url }))
   447→        toast.success('تم رفع صورة الغلاف')
   448→      } else {
   449→        toast.error('فشل رفع الصورة: ' + (data.error || ''))
   450→      }
   451→    } catch {
   452→      toast.error('خطأ في الاتصال')
   453→    } finally {
   454→      setUploadingCover(false)
   455→    }
   456→  }
   457→
   458→  const save = async () => {
   459→    if (!form.title) {
   460→      toast.error('أدخل عنوان الدورة')
   461→      return
   462→    }
   463→    setSaving(true)
   464→    try {
   465→      const slug = form.title.toLowerCase().replace(/\s+/g, '-').slice(0, 80)
   466→      // Build the payload. Always include `cover` (string URL or null) so the
   467→      // courses.cover column is updated on every save — even when the trainer
   468→      // cleared the field.
   469→      const body: any = {
   470→        title: form.title,
   471→        short_description: form.short_description ?? '',
   472→        description: form.description ?? '',
   473→        level: form.level,
   474→        price_type: form.price_type,
   475→        price: Number(form.price) || 0,
   476→        currency: form.currency || 'ر.ي',
   477→        duration_hours: Number(form.duration_hours) || 0,
   478→        pass_score: Number(form.pass_score) || 60,
   479→        status: form.status || 'منشورة',
   480→        cover: form.cover ?? null,
   481→        slug,
   482→        trainer_id: trainerId,
   483→        modules_count: form.modules_count || 0,
   484→        lessons_count: form.lessons_count || 0,
   485→      }
   486→      const res = await fetch('/api/courses', {
   487→        method: course ? 'PUT' : 'POST',
   488→        headers: { 'Content-Type': 'application/json' },
   489→        body: JSON.stringify(course ? { id: course.id, ...body } : body),
   490→      })
   491→      if (res.ok) {
   492→        toast.success(course ? 'تم تحديث الدورة' : 'تم إنشاء الدورة')

   750→function TrainerExams({ courses }: { courses: any[] }) {
   751→  const [loading, setLoading] = useState(true)
   752→  const [exams, setExams] = useState<any[]>([])
   753→
   754→  useEffect(() => {
   755→    let cancelled = false
   756→    const loadExams = async () => {
   757→      if (!courses || courses.length === 0) {
   758→        setExams([])
   759→        setLoading(false)
   760→        return
   761→      }
   762→      setLoading(true)
   763→      try {
   764→        // Fetch lessons for each course in parallel
   765→        const results = await Promise.all(
   766→          courses.map(async (c) => {
   767→            try {
   768→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   769→              const data = await res.json()
   770→              const lessons = Array.isArray(data) ? data : []
   771→              // Filter exam-type lessons only
   772→              return lessons
   773→                .filter((l: any) => l.lesson_type === 'exam')
   774→                .map((l: any) => {
   775→                  // Parse question count from content JSON
   776→                  let questions: any[] = []
   777→                  if (l.content) {
   778→                    try {
   779→                      const parsed = JSON.parse(l.content)
   780→                      if (Array.isArray(parsed)) questions = parsed
   781→                    } catch {}
   782→                  }
   783→                  return {
   784→                    ...l,
   785→                    course_id: c.id,
   786→                    course_title: c.title,
   787→                    course_pass_score: c.pass_score,
   788→                    questions_count: questions.length,
   789→                    questions,
   790→                  }
   791→                })
   792→            } catch {
   793→              return []
   794→            }
   795→          })
   796→        )
   797→        if (cancelled) return
   798→        const flat = results.flat()
   799→        setExams(flat)
   800→      } catch {
   801→        if (!cancelled) setExams([])
   802→      } finally {
   803→        if (!cancelled) setLoading(false)
   804→      }
   805→    }
   806→    loadExams()
   807→    return () => { cancelled = true }
   808→  }, [courses])
   809→
   810→  const totalQuestions = exams.reduce((acc, e) => acc + (e.questions_count || 0), 0)
   811→  const totalCoursesWithExams = new Set(exams.map((e) => e.course_id)).size
   812→
   813→  return (
   814→    <div className="space-y-6">
   815→      <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
   816→        <StatCard icon={FileText} label="الاختبارات" value={exams.length} color="blue" />
   817→        <StatCard icon={HelpCircle} label="إجمالي الأسئلة" value={totalQuestions} color="navy" />
   818→        <StatCard icon={BookOpen} label="دورات بها اختبارات" value={totalCoursesWithExams} color="gold" />
   819→      </div>
   820→
   821→      <SectionCard
   822→        title="الاختبارات"
   823→        description="الاختبارات المستندة إلى محتوى دوراتك التدريبية"
   824→      >
   825→        {loading ? (
   826→          <div className="text-center py-12">
   827→            <Loader2 className="w-8 h-8 text-navy mx-auto mb-3 animate-spin" />
   828→            <p className="text-muted-foreground text-sm">جاري تحميل الاختبارات...</p>
   829→          </div>
   830→        ) : exams.length === 0 ? (
   831→          <div className="text-center py-12">
   832→            <FileText className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />
   833→            <p className="text-muted-foreground mb-2">لا توجد اختبارات بعد</p>
   834→            <p className="text-xs text-muted-foreground">
   835→              أنشئ دروساً من نوع "اختبار" في محتوى دوراتك لتظهر هنا

   837→          </div>

   841→              <Card key={exam.id} className="overflow-hidden card-hover">
   842→                <div className="p-4 border-b border-border bg-purple-500/5">
   843→                  <div className="flex items-start justify-between gap-2 mb-2">

  1188→function TrainerNotifs({ courses }: { courses: any[] }) {
  1189→  const [loading, setLoading] = useState(true)
  1190→  const [notifs, setNotifs] = useState<any[]>([])
  1191→
  1192→  useEffect(() => {
  1193→    let cancelled = false
  1194→    const loadNotifs = async () => {
  1195→      if (!courses || courses.length === 0) {
  1196→        setNotifs([])
  1197→        setLoading(false)
  1198→        return
  1199→      }
  1200→      setLoading(true)
  1201→      try {
  1202→        const courseIds = courses.map((c) => c.id)
  1203→        const courseById = new Map(courses.map((c) => [c.id, c]))
  1204→
  1205→        // Fetch enrollments for all courses (we'll filter on the client)
  1206→        const [enrRes, stuRes] = await Promise.all([
  1207→          fetch('/api/enrollments'),
  1208→          fetch('/api/students?role=student'),
  1209→        ])
  1210→        const [enrollments, students] = await Promise.all([enrRes.json(), stuRes.json()])
  1211→
  1212→        if (cancelled) return
  1213→
  1214→        const enrList = Array.isArray(enrollments) ? enrollments : []
  1215→        const stuList = Array.isArray(students) ? students : []
  1216→        const studentById = new Map(stuList.map((s: any) => [s.id, s]))
  1217→
  1218→        // Only enrollments for this trainer's courses
  1219→        const mine = enrList.filter((e: any) => courseIds.includes(e.course_id))
  1220→
  1221→        // Sort by enrolled_at desc
  1222→        mine.sort((a: any, b: any) => {
  1223→          const aT = a.enrolled_at ? new Date(a.enrolled_at).getTime() : 0
  1224→          const bT = b.enrolled_at ? new Date(b.enrolled_at).getTime() : 0
  1225→          return bT - aT
  1226→        })
  1227→
  1228→        // Build notification objects — newest 25
  1229→        const built = mine.slice(0, 25).map((e: any) => {
  1230→          const student = studentById.get(e.user_id)
  1231→          const course = courseById.get(e.course_id)
  1232→          const isFree = !e.amount_paid || e.amount_paid === 0
  1233→          return {
  1234→            id: e.id,
  1235→            type: 'enrollment',
  1236→            studentName: student?.name || 'متدرب',
  1237→            studentAvatar: student?.avatar,
  1238→            courseTitle: course?.title || 'دورة',
  1239→            enrolledAt: e.enrolled_at,
  1240→            isFree,
  1241→            amountPaid: e.amount_paid,
  1242→            currency: e.currency,
  1243→            paymentStatus: e.payment_status,
  1244→          }
  1245→        })
  1246→
  1247→        if (!cancelled) setNotifs(built)
  1248→      } catch {
  1249→        if (!cancelled) setNotifs([])
  1250→      } finally {
  1251→        if (!cancelled) setLoading(false)
  1252→      }
  1253→    }
  1254→    loadNotifs()
  1255→    return () => { cancelled = true }
  1256→  }, [courses])
  1257→
  1258→  const formatRelativeTime = (iso: string | null | undefined) => {
  1259→    if (!iso) return ''
  1260→    try {
  1261→      const date = new Date(iso)
  1262→      const diffMs = Date.now() - date.getTime()
  1263→      const diffMin = Math.floor(diffMs / 60000)
  1264→      if (diffMin < 1) return 'الآن'
  1265→      if (diffMin < 60) return `منذ ${diffMin} دقيقة`
  1266→      const diffHr = Math.floor(diffMin / 60)
  1267→      if (diffHr < 24) return `منذ ${diffHr} ساعة`
  1268→      const diffDay = Math.floor(diffHr / 24)
  1269→      if (diffDay < 30) return `منذ ${diffDay} يوم`
  1270→      return date.toLocaleDateString('ar-EG')
  1271→    } catch {
  1272→      return ''
  1273→    }
  1274→  }
  1275→
  1276→  return (
  1277→    <SectionCard
  1278→      title="الإشعارات"
  1279→      description="آخر التسجيلات في دوراتك التدريبية"
  1280→    >
  1281→      {loading ? (
  1282→        <div className="text-center py-12">
  1283→          <Loader2 className="w-8 h-8 text-navy mx-auto mb-3 animate-spin" />
  1284→          <p className="text-muted-foreground text-sm">جاري تحميل الإشعارات...</p>
  1285→        </div>

  1287→        <div className="text-center py-12">
  1288→          <Bell className="w-16 h-16 text-muted-foreground/30 mx-auto mb-4" />

  1295→        <div className="space-y-3">
  1296→          {notifs.map((n) => (
  1297→            <div
  1298→              key={n.id}
  1299→              className="p-4 bg-gold/5 border border-gold/20 rounded-lg flex items-start gap-3"
