    33→    33→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    34→    34→// Show only the human-readable text in the UI, but preserve the extra data on save.
    35→    35→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    36→    36→const splitBio = (raw: string) => {
    37→    37→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    38→    38→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    39→    39→}
    40→    40→const joinBio = (display: string, original?: string) => {
    41→    41→  if (!original) return display
    42→    42→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    43→    43→  if (i === -1) return display
    44→    44→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    45→    45→}

    47→    47→export function TrainerDashboard() {

   140→   140→function TrainerOverview({ user, courses, students }: any) {
   141→   141→  // Compute the real average rating from courses that actually carry a
   142→   142→  // numeric rating. If no ratings exist yet, fall back to a clear message
   143→   143→  // instead of a hardcoded fake number.
   144→   144→  const ratedCourses = (courses || []).filter(
   145→   145→    (c: any) => typeof c.rating === 'number' && c.rating > 0
   146→   146→  )
   147→   147→  const avgRating =
   148→   148→    ratedCourses.length > 0
   149→   149→      ? (ratedCourses.reduce((acc, c) => acc + c.rating, 0) / ratedCourses.length).toFixed(1)
   150→   150→      : 'لا توجد تقييمات بعد'
   151→   151→
   152→   152→  // Real "active exams" count: lessons with lesson_type === 'exam' across
   153→   153→  // the trainer's own courses (same data source as TrainerExams).
   154→   154→  const [examsCount, setExamsCount] = useState<number>(0)
   155→   155→  const [examsLoading, setExamsLoading] = useState(false)
   156→   156→
   157→   157→  useEffect(() => {
   158→   158→    let cancelled = false
   159→   159→    const load = async () => {
   160→   160→      if (!courses || courses.length === 0) {
   161→   161→        setExamsCount(0)
   162→   162→        return
   163→   163→      }
   164→   164→      setExamsLoading(true)
   165→   165→      try {
   166→   166→        const results = await Promise.all(
   167→   167→          courses.map(async (c: any) => {
   168→   168→            try {
   169→   169→              const res = await fetch(`/api/lessons?course_id=${c.id}`)
   170→   170→              const data = await res.json()
   171→   171→              const lessons = Array.isArray(data) ? data : []
   172→   172→              return lessons.filter((l: any) => l.lesson_type === 'exam').length
   173→   173→            } catch {
   174→   174→              return 0
   175→   175→            }
   176→   176→          })
   177→   177→        )
   178→   178→        if (cancelled) return
   179→   179→        setExamsCount(results.reduce((a, b) => a + b, 0))
   180→   180→      } catch {
   181→   181→        if (!cancelled) setExamsCount(0)
   182→   182→      } finally {
   183→   183→        if (!cancelled) setExamsLoading(false)
   184→   184→      }
   185→   185→    }
   186→   186→    load()
   187→   187→    return () => { cancelled = true }
   188→   188→  }, [courses])
   189→   189→
   190→   190→  return (
   191→   191→    <div className="space-y-6">
   192→   192→      <div className="bg-fia-gradient rounded-2xl p-6 lg:p-8 text-white">
   193→   193→        <h2 className="text-2xl font-bold mb-1">مرحباً، {user?.name} 👋</h2>
   194→   194→        <p className="text-white/80">نظرة عامة على نشاطك التدريبي</p>
   195→   195→      </div>
   196→   196→
   197→   197→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   198→   198→        <StatCard icon={BookOpen} label="الدورات التدريبية" value={courses.length} color="navy" />
   199→   199→        <StatCard icon={Users} label="إجمالي المتدربين" value={String(students.length)} color="gold" />
   200→   200→        <StatCard icon={Star} label="متوسط التقييم" value={avgRating} color="green" />
   201→   201→        <StatCard
   202→   202→          icon={FileText}
   203→   203→          label="الاختبارات النشطة"
   204→   204→          value={examsLoading ? '—' : examsCount}
   205→   205→          color="blue"
   206→   206→        />
   207→   207→      </div>
   208→   208→
   209→   209→      <SectionCard title="أحدث الدورات" description="الدورات التي تدرّسها">
   210→   210→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   211→   211→          {courses.slice(0, 3).map((c: any) => (
   212→   212→            <Card key={c.id} className="overflow-hidden card-hover">
   213→   213→              <div className="h-32 bg-navy">
   214→   214→                {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   215→   215→              </div>
   216→   216→              <div className="p-4">
   217→   217→                <h3 className="font-bold text-navy text-sm mb-1 line-clamp-1">{c.title}</h3>
   218→   218→                <div className="text-xs text-muted-foreground">{c.enrolled || 0} متدرب • {c.rating || 0} ⭐</div>
   219→   219→              </div>
   220→   220→            </Card>
   221→   221→          ))}
   222→   222→        </div>
   223→   223→      </SectionCard>
   224→   224→    </div>
   225→   225→  )
   226→   226→}

   228→   228→function TrainerCourses({ courses, onAdd, onEdit, onRefresh, onManageContent }: any) {
   229→   229→  const [deleting, setDeleting] = useState<string | null>(null)
   230→   230→  // Course currently awaiting delete confirmation. Null means the dialog is closed.
   231→   231→  const [deleteTarget, setDeleteTarget] = useState<any | null>(null)
   232→   232→
   233→   233→  const handleDelete = async () => {
   234→   234→    if (!deleteTarget) return
   235→   235→    const id = deleteTarget.id
   236→   236→    setDeleting(id)
   237→   237→    try {
   238→   238→      const res = await fetch('/api/courses', {
   239→   239→        method: 'DELETE',
   240→   240→        headers: { 'Content-Type': 'application/json' },
   241→   241→        body: JSON.stringify({ id }),
   242→   242→      })
   243→   243→      if (res.ok) {
   244→   244→        toast.success('تم حذف الدورة')
   245→   245→        setDeleteTarget(null)
   246→   246→        onRefresh()
   247→   247→      } else {
   248→   248→        const err = await res.json().catch(() => ({}))
   249→   249→        toast.error(err.error || 'فشل الحذف')
   250→   250→      }
   251→   251→    } catch {
   252→   252→      toast.error('خطأ في الاتصال')
   253→   253→    } finally {
   254→   254→      setDeleting(null)
   255→   255→    }
   256→   256→  }
   257→   257→
   258→   258→  return (
   259→   259→    <SectionCard
   260→   260→      title="دوراتي التدريبية"
   261→   261→      description="إدارة الدورات التي تدرّسها"
   262→   262→      action={
   263→   263→        <Button onClick={onAdd} className="bg-navy hover:bg-navy-light">
   264→   264→          <Plus className="w-4 h-4 ml-1" /> دورة جديدة
   265→   265→        </Button>
   266→   266→      }
   267→   267→    >
   268→   268→      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   269→   269→        {courses.map((c: any) => (
   270→   270→          <Card key={c.id} className="overflow-hidden card-hover">
   271→   271→            <div className="h-32 bg-navy relative">
   272→   272→              {c.cover && <img src={c.cover} alt={c.title} className="w-full h-full object-cover" />}
   273→   273→              <Badge className="absolute top-2 right-2 bg-navy/80 text-gold border-gold/30">{c.level}</Badge>
   274→   274→            </div>
   275→   275→            <div className="p-4">
   276→   276→              <h3 className="font-bold text-navy text-sm mb-2 line-clamp-1">{c.title}</h3>
   277→   277→              <div className="flex justify-between text-xs text-muted-foreground mb-3">
   278→   278→                <span>{c.enrolled || 0} متدرب</span>
   279→   279→                <span>{c.price_type === 'مجانية' ? 'مجانية' : `${c.price.toLocaleString()} ${c.currency || 'ر.ي'}`}</span>
   280→   280→              </div>
   281→   281→              <div className="space-y-2">
   282→   282→                <Button size="sm" className="w-full bg-gold hover:bg-gold-light text-navy" onClick={() => onManageContent(c.id)}>
   283→   283→                  <FileText className="w-3.5 h-3.5 ml-1" /> إدارة المحتوى
   284→   284→                </Button>
   285→   285→                <div className="flex gap-2">
   286→   286→                  <Button size="sm" variant="outline" className="flex-1" onClick={() => onEdit(c)}>
   287→   287→                    <Edit className="w-3.5 h-3.5 ml-1" /> تعديل
   288→   288→                  </Button>
   289→   289→                  <Button size="sm" variant="outline" className="border-red-500/30 text-red-500 hover:bg-red-500/10" onClick={() => setDeleteTarget(c)} disabled={deleting === c.id}>
   290→   290→                    <Trash2 className="w-3.5 h-3.5" />
   291→   291→                  </Button>
   292→   292→                </div>
   293→   293→              </div>
   294→   294→            </div>
   295→   295→          </Card>
   296→   296→        ))}
   297→   297→      </div>
   298→   298→
   299→   299→      {/* Professional delete-confirmation card */}
   300→   300→      <Dialog
   301→   301→        open={!!deleteTarget}
   302→   302→        onOpenChange={(open) => { if (!open && !deleting) setDeleteTarget(null) }}
   303→   303→      >
   304→   304→        <DialogContent className="sm:max-w-md">
   305→   305→          <DialogHeader>
   306→   306→            <DialogTitle className="flex items-center gap-2 text-red-600">
   307→   307→              <AlertTriangle className="w-5 h-5" />
   308→   308→              تأكيد الحذف
   309→   309→            </DialogTitle>
   310→   310→          </DialogHeader>
   311→   311→          <div className="py-2 space-y-3">
   312→   312→            <div className="flex items-start gap-3 p-3 bg-red-500/5 border border-red-500/20 rounded-lg">
   313→   313→              <Trash2 className="w-5 h-5 text-red-500 flex-shrink-0 mt-0.5" />
   314→   314→              <div className="min-w-0">
   315→   315→                <div className="text-xs text-muted-foreground mb-1">سيتم حذف الدورة التالية:</div>
   316→   316→                <div className="font-bold text-navy line-clamp-2 break-words">{deleteTarget?.title}</div>
   317→   317→              </div>
   318→   318→            </div>
   319→   319→            <p className="text-sm text-muted-foreground text-center">
   320→   320→              هل أنت متأكد من الحذف؟ لا يمكن التراجع عن هذا الإجراء.
   321→   321→            </p>
   322→   322→          </div>
   323→   323→          <DialogFooter>
   324→   324→            <Button
   325→   325→              variant="outline"
   326→   326→              onClick={() => setDeleteTarget(null)}
   327→   327→              disabled={!!deleting}
   328→   328→            >
   329→   329→              إلغاء
   330→   330→            </Button>
   331→   331→            <Button
   332→   332→              onClick={handleDelete}
   333→   333→              disabled={!!deleting}
   334→   334→              className="bg-red-600 hover:bg-red-700 text-white"
   335→   335→            >
   336→   336→              {deleting
   337→   337→                ? <><Loader2 className="w-4 h-4 ml-1 animate-spin" /> جاري الحذف...</>
   338→   338→                : <><Trash2 className="w-4 h-4 ml-1" /> حذف الدورة</>}
   339→   339→            </Button>
   340→   340→          </DialogFooter>
   341→   341→        </DialogContent>
   342→   342→      </Dialog>
   343→   343→    </SectionCard>
   344→   344→  )
   345→   345→}

   347→   347→function CourseDialog({ open, onOpenChange, course, trainerId, onSaved }: any) {
   348→   348→  const [form, setForm] = useState<any>(
   349→   349→    course || {
   350→   350→      title: '',
   351→   351→      short_description: '',
   352→   352→      description: '',
   353→   353→      level: 'متوسط',
   354→   354→      price_type: 'مجانية',
   355→   355→      price: 0,
   356→   356→      currency: 'ر.ي',
   357→   357→      duration_hours: 0,
   358→   358→      pass_score: 60,
   359→   359→      cover: '',
   360→   360→      status: 'منشورة',
   361→   361→    }
   362→   362→  )
   363→   363→  const [saving, setSaving] = useState(false)
   364→   364→  const [uploadingCover, setUploadingCover] = useState(false)
   365→   365→
   366→   366→  // Keep the form in sync with the `course` prop whenever the dialog opens or
   367→   367→  // the course being edited changes. Without this, a dialog that stays mounted
   368→   368→  // across course switches would show stale state (and could overwrite the
   369→   369→  // wrong course's `cover` field).
   370→   370→  useEffect(() => {
   371→   371→    if (open) {
   372→   372→      setForm(
   373→   373→        course || {
   374→   374→          title: '',
   375→   375→          short_description: '',
   376→   376→          description: '',
   377→   377→          level: 'متوسط',
   378→   378→          price_type: 'مجانية',
   379→   379→          price: 0,
   380→   380→          currency: 'ر.ي',
   381→   381→          duration_hours: 0,
   382→   382→          pass_score: 60,
   383→   383→          cover: '',
   384→   384→          status: 'منشورة',
   385→   385→        }
   386→   386→      )
   387→   387→    }
   388→   388→  }, [open, course])
   389→   389→
   390→   390→  const handleUploadCover = async (file: File) => {
   391→   391→    setUploadingCover(true)
   392→   392→    try {
   393→   393→      const fd = new FormData()
   394→   394→      fd.append('file', file)
   395→   395→      fd.append('folder', 'courses')
   396→   396→      const res = await fetch('/api/upload/image', { method: 'POST', body: fd })
   397→   397→      const data = await res.json().catch(() => ({}))
   398→   398→      if (data.url) {
   399→   399→        // Store the Telegram-hosted URL on form.cover — it is later persisted
   400→   400→        // to the courses.cover column by the save() call below.
   401→   401→        setForm((prev: any) => ({ ...prev, cover: data.url }))
   402→   402→        toast.success('تم رفع صورة الغلاف')
   403→   403→      } else {
   404→   404→        toast.error('فشل رفع الصورة: ' + (data.error || ''))
   405→   405→      }
   406→   406→    } catch {
   407→   407→      toast.error('خطأ في الاتصال')
   408→   408→    } finally {
   409→   409→      setUploadingCover(false)
