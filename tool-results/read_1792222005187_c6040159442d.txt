    48→// Bio may embed structured JSON after a marker (e.g. "===FIA_EXTRA===").
    49→// Show only the human-readable text in the UI, but preserve the extra data on save.
    50→const FIA_BIO_EXTRA_MARKER = '===FIA_EXTRA==='
    51→const splitBio = (raw: string) => {
    52→  const i = raw.indexOf(FIA_BIO_EXTRA_MARKER)
    53→  return i === -1 ? raw : raw.slice(0, i).trimEnd()
    54→}
    55→const joinBio = (display: string, original?: string) => {
    56→  if (!original) return display
    57→  const i = original.indexOf(FIA_BIO_EXTRA_MARKER)
    58→  if (i === -1) return display
    59→  return `${display.trimEnd()}\n\n${original.slice(i)}`
    60→}

    62→// Returns a normalized status info object for an enrollment's payment_status.
    63→// payment_status can be one of: 'pending' | 'confirmed' | 'completed' | 'free'.
    64→// - 'pending'    -> student paid, waiting for admin approval (yellow)
    65→// - 'confirmed'  -> admin approved the payment, course in progress (green)
    66→// - 'completed'  -> payment confirmed AND course finished (green)
    67→// - 'free'       -> free enrollment (blue)
    68→export function getPaymentStatusInfo(paymentStatus: string | undefined | null): {
    69→  label: string
    70→  badgeClass: string
    71→} {
    72→  switch (paymentStatus) {
    73→    case 'completed':
    74→      return { label: 'مكتمل', badgeClass: 'bg-green-500 text-white' }
    75→    case 'confirmed':
    76→      return { label: 'تم تأكيد الدفع', badgeClass: 'bg-green-500 text-white' }
    77→    case 'free':
    78→      return { label: 'مجانية', badgeClass: 'bg-blue-500 text-white' }
    79→    case 'pending':
    80→    default:
    81→      return { label: 'بانتظار تأكيد الدفع', badgeClass: 'bg-yellow-500 text-white' }
    82→  }
    83→}
    84→
    85→// Returns a status info object for a raw payment record's `status` field
    86→// (used in the Payments section). Raw status can be: 'مكتمل' | 'completed' |
    87→// 'قيد الانتظار' | 'pending' | 'بانتظار' | 'ملغي' | 'failed' | 'فشل' | etc.
    88→export function getPaymentRecordStatusInfo(status: string | undefined | null): {
    89→  label: string
    90→  badgeClass: string
    91→} {
    92→  const s = (status || '').trim()
    93→  const norm = s.toLowerCase()
    94→  if (s === 'مكتمل' || norm === 'completed' || norm === 'paid' || norm === 'confirmed') {
    95→    return { label: s || 'مكتمل', badgeClass: 'bg-green-500/15 text-green-600 border-green-500/30' }
    96→  }
    97→  if (s === 'ملغي' || norm === 'failed' || s === 'فشل' || norm === 'cancelled' || norm === 'canceled') {
    98→    return { label: s || 'فشل', badgeClass: 'bg-red-500/15 text-red-600 border-red-500/30' }
    99→  }
   100→  if (s === 'قيد الانتظار' || s === 'بانتظار' || norm === 'pending' || s === '') {
   101→    return { label: s || 'بانتظار', badgeClass: 'bg-yellow-500/15 text-yellow-600 border-yellow-500/30' }
   102→  }
   103→  // Unknown status — show as neutral
   104→  return { label: s, badgeClass: 'bg-muted text-muted-foreground border-border' }
   105→}

   361→function BrowseCourses({ courses, enrollments, userId, onEnrolled }: any) {
   362→  const [paymentMethods, setPaymentMethods] = useState<any[]>([])
   363→  const [payDialog, setPayDialog] = useState<{ course: any; open: boolean } | null>(null)
   364→  const [enrolling, setEnrolling] = useState<string | null>(null)
   365→
   366→  useEffect(() => {
   367→    fetch('/api/payment-methods').then((r) => r.json()).then((d) => setPaymentMethods(Array.isArray(d) ? d.filter((p:any)=>p.enabled) : [])).catch(() => {})
   368→  }, [])
   369→
   370→  const isEnrolled = (courseId: string) => enrollments?.some((e: any) => e.course_id === courseId)
   371→
   372→  const enrollFree = async (course: any) => {
   373→    // Double-check client-side before calling API
   374→    if (isEnrolled(course.id)) {
   375→      toast.info('أنت مسجّل في هذه الدورة')
   376→      return
   377→    }
   378→    setEnrolling(course.id)
   379→    try {
   380→      const res = await fetch('/api/enrollments', {
   381→        method: 'POST',
   382→        headers: { 'Content-Type': 'application/json' },
   383→        body: JSON.stringify({ user_id: userId, course_id: course.id, payment_status: 'free' }),
   384→      })
   385→      const data = await res.json()
   386→      if (data.already_enrolled) {
   387→        toast.info('أنت مسجّل في هذه الدورة')
   388→      } else if (res.ok) {
   389→        toast.success(`تم التسجيل في دورة: ${course.title}`)
   390→        onEnrolled()
   391→      } else toast.error('فشل التسجيل')
   392→    } catch {
   393→      toast.error('خطأ')
   394→    } finally {
   395→      setEnrolling(null)
   396→    }
   397→  }
   398→
   399→  const enrollPaid = async (course: any, method: any, reference: string, proofUrl?: string | null, couponCode?: string) => {
   400→    if (isEnrolled(course.id)) { toast.info('أنت مسجّل في هذه الدورة'); return }
   401→    setEnrolling(course.id)
   402→    try {
   403→      let fullReference = proofUrl ? (reference + ' | PROOF: ' + proofUrl) : reference
   404→      // If coupon was applied, validate it server-side and get discounted price
   405→      let finalAmount = course.price
   406→      if (couponCode) {
   407→        try {
   408→          const couponRes = await fetch(`/api/coupons?code=${encodeURIComponent(couponCode.toUpperCase())}`)
   409→          const couponData = await couponRes.json()
   410→          if (couponData.valid && couponData.coupon) {
   411→            const c = couponData.coupon
   412→            finalAmount = c.discount_type === 'percentage'
   413→              ? Math.max(0, course.price - Math.round((course.price * c.discount_value) / 100))
   414→              : Math.max(0, course.price - c.discount_value)
   415→            // Add coupon info to the reference
   416→            fullReference = fullReference + ' | COUPON:' + couponCode
   417→          }
   418→        } catch {}
   419→      }
   420→      const res = await fetch('/api/enrollments', {
   421→        method: 'POST',

   683→function Exams({ userId, courses }: any) {

   685→  const [loading, setLoading] = useState(true)

   690→      if (!userId) {
   691→        setLoading(false)
   692→        return
   693→      }
   694→      try {
   695→        // 1. Fetch the student's enrollments (only their own).

   697→        const enrollments = await enrRes.json()
   698→        const validEnrollments: any[] = Array.isArray(enrollments) ? enrollments : []
   699→
   700→        // 2. For each enrolled course, fetch its lessons and filter exam-type lessons.
   701→        //    Only show exams for courses the student actually has access to
   702→        //    (i.e., payment_status is NOT 'pending').
   703→        const accessible = validEnrollments.filter(
   704→          (e: any) => e.payment_status !== 'pending'
   705→        )
   706→
   707→        const results = await Promise.all(
   708→          accessible.map(async (e: any) => {
   709→            try {
   710→              const lessonsRes = await fetch(`/api/lessons?course_id=${e.course_id}`)
   711→              const lessons = await lessonsRes.json()
   712→              const allLessons: any[] = Array.isArray(lessons) ? lessons : []
   713→              const examLessons = allLessons.filter(
   714→                (l: any) => l.lesson_type === 'exam'
   715→              )
   716→              const courseTitle =
   717→                courses?.find((c: any) => c.id === e.course_id)?.title || null

   719→                id: l.id,
   720→                name: l.title || 'اختبار بدون اسم',

   732→        if (cancelled) return

   736→      } finally {
   737→        if (!cancelled) setLoading(false)

   819→function Certificates({ certificates }: any) {

   823→
   824→  const handleDownload = (cert: any) => {
   825→    setDownloading(cert.id)
   826→    // Open PDF in new tab (browser will download or preview)
   827→    window.open(`/api/certificate-pdf?number=${encodeURIComponent(cert.certificate_number)}&format=pdf`, '_blank')
   828→    setTimeout(() => setDownloading(null), 2000)

   921→function Payments({ payments, userId, courses }: any) {

   927→    const da = a.date ? new Date(a.date).getTime() : 0

   929→    return db - da

   931→
   932→  const getCourseTitle = (courseId: string | null | undefined) => {

  1134→function Notifications({ userId, enrollments, courses }: any) {

  1156→              year: 'numeric',
  1157→              month: 'long',
  1158→              day: 'numeric',

  1161→            return null

  1183→              iconBg: 'bg-green-500/15 text-green-600',

  1187→            })

  1190→              id: `enr-completed-${e.id}`,
  1191→              icon: Award,

  1209→        // 2. "New lesson added" notifications: compare each enrolled course's

  1245→        } catch {}
  1246→
  1247→        if (!cancelled) setNotifs(items)
  1248→      } catch {
